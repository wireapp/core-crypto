17/// Implements the `Entity` trait and related traits for the given struct.
18///
19/// Intended to be used only within `core-crypto-keystore`.
20///
21/// All fields in an entity using this derive macro must have one of four types:
22///
23/// - `Vec<u8>`
24/// - `String`
25/// - `Option<Vec<u8>>`
26/// - `u32`
27///
28/// There is one exception: the identity field can only be a `Vec<u8>` or `String`; the optional
29/// variant is forbidden in that context.
30///
31/// This is intentionally restrictive. If your entity has other field types, implement these traits manually.
32///
33/// ## Attributes
34///
35/// Attributes modify the code generated by this proc macro by adding annotations to the whole struct
36/// or to distinct fields. All attributes accepted by this proc macro are keyed with `#[entity]`.
37/// The attribute itself appears in the following parenthesis.
38///
39/// It is acceptable to combine multiple entity attributes together, simply by combining them
40/// within the parenthesis, separated by commas: `#[entity(id, column_name="column")]`, for example.
41///
42/// ### `#[entity(table_name = "my_table")]`
43///
44/// This is an **outer attribute** which modifies the entire item. Its parameters:
45///
46/// - `table_name`: Optional. When set, this specifies the SQL table name which will be used. Otherwise, one will be
47///   generated by transforming the item name to snake case and adding an `s`.
48/// - `no_upsert`: Optional flag. By default the generated `save` emits `INSERT OR REPLACE`. When this flag is present
49///   it emits a plain `INSERT` instead, and a uniqueness violation is mapped to `CryptoKeystoreError::AlreadyExists`.
50///
51/// ### `#[entity(id)]`
52///
53/// This is a **field attribute** which can be applied to at most 1 field in the item.
54///
55/// The field to which this attribute is applied must have type `Vec<u8>` or `String`.
56/// Otherwise, you must implement all relevant entity traits manually instead of relying on this macro.
57///
58/// If this attribute does not appear, the macro will search for a field whose name is `id` and assume
59/// that is an implicit id.
60///
61/// This attribute does not accept a value.
62///
63/// ### `#[entity(column = "my_column")]`
64///
65/// This is a **field attribute** which can be applied to any number of fields in the struct.
66///
67/// The `column` value will be used as the SQL column name for this field. If this is unset, the field's name
68/// will be used as the sql column name.
69///
70/// ## Trait Implementations
71///
72/// ### Implements
73///
74/// - `PrimaryKey`
75/// - `BorrowPrimaryKey`
76/// - `Entity`
77/// - `EntityGetBorrowed`
78/// - `EntityDatabaseMutation`
79/// - `EntityDeleteBorrowed`
80///
81/// ### `PrimaryKey` and `BorrowPrimaryKey`
82///
83/// The primary key is the type of the id field: `Vec<u8>` or `String`, borrowed as `[u8]` or `str`
84/// respectively. For a primary key which is not simply one of the entity's own fields, implement these
85/// traits manually.
86///
87/// ### `Entity`
88///
89/// `TABLE_NAME` comes from the `table_name` attribute, or from the item name if that attribute is absent.
90/// `get` delegates to `EntityGetBorrowed::get_borrowed`; `count` and `load_all` defer to the helper
91/// functions in `core_crypto_keystore::entities::helpers`.
92///
93/// No fields are autogenerated: `AutoGeneratedFields` is `()`. If any fields should be autogenerated,
94/// implement `EntityDatabaseMutation` manually.
95///
96/// ### `EntityDatabaseMutation` and `EntityDeleteBorrowed`
97///
98/// The derived `save` method generates sql like:
99///
100/// ```sql
101/// INSERT OR REPLACE INTO entities (id, foo, bar) VALUES (?, ?, ?)
102/// ```
103///
104/// The `OR REPLACE` is omitted when the `no_upsert` flag is set; see the attribute documentation above.
105///
106/// `delete` delegates to `EntityDeleteBorrowed::delete_borrowed`, which deletes by the id column.
107#[proc_macro_derive(Entity, attributes(entity))]
108pub fn derive_entity(input: TokenStream) -> TokenStream {
109    let input = parse_macro_input!(input as DeriveInput);
110    match entity_derive::Entity::from_derive_input(&input) {
111        Ok(entity) => quote! { #entity },
112        Err(err) => err.write_errors(),
113    }
114    .into()
115}

117/// Will drop current MLS group in memory and replace it with the one in the keystore.
118/// This simulates an application crash. Once restarted, everything has to be loaded from the
119/// keystore, memory is lost.
120///
121/// Requires the `MlsConversation` method to have a parameter exactly like `backend: &MlsCryptoProvider`
122///
123/// This helps spotting:
124/// * when one has forgotten to call `persist_group_when_changed`
125/// * if persisted fields are sufficient to pursue normally after a crash
126///
127/// **IF** you mark a method `#[durable]`, remove its call to
128/// `persist_group_when_changed` and tests still pass, you either:
129/// * have unit tests not covering the method enough
130/// * do not require this method to be durable
131#[proc_macro_attribute]
132pub fn durable(_args: TokenStream, item: TokenStream) -> TokenStream {
133    durable::durable(item)
134}

136/// !!! Not literally idempotent !!!
137///
138/// Marker for methods on 'core_crypto::Client' which leave the number of entities in the keystore even.
139/// They can create/destroy some but always compensate.
140/// So they are not idempotent, they cannot be safely replayed and they might leave the keystore in
141/// a different state.
142#[proc_macro_attribute]
143pub fn idempotent(_args: TokenStream, item: TokenStream) -> TokenStream {
144    idempotent::idempotent(item)
145}
146
147/// Neologism to mean the opposite of idempotent. Methods of 'core_crypto::Client' marked with this have to
148/// insert/delete an entity in the keystore and change the number of entities persisted.
149#[proc_macro_attribute]
150pub fn dispotent(_args: TokenStream, item: TokenStream) -> TokenStream {
151    idempotent::dispotent(item)
152}

154pub(crate) fn doc_attributes(ast: &ItemFn) -> Vec<Attribute> {
155    ast.attrs
156        .iter()
157        .filter(|attr| attr.path().is_ident("doc"))
158        .cloned()
159        .collect::<Vec<syn::Attribute>>()
160}

162pub(crate) fn compile_error(mut item: TokenStream, err: syn::Error) -> TokenStream {
163    let compile_err = TokenStream::from(err.to_compile_error());
164    item.extend(compile_err);
165    item
166}

168#[allow(clippy::type_complexity)]
169pub(crate) fn items(
170    ast: &ItemFn,
171) -> (
172    &ReturnType,
173    &Ident,
174    &Punctuated<FnArg, Comma>,
175    &Box<Block>,
176    &Vec<Attribute>,
177    &Visibility,
178) {
179    let ret = &ast.sig.output;
180    let name = &ast.sig.ident;
181    let inputs = &ast.sig.inputs;
182    let body = &ast.block;
183    let attrs = &ast.attrs;
184    let vis = &ast.vis;
185    (ret, name, inputs, body, attrs, vis)
186}
