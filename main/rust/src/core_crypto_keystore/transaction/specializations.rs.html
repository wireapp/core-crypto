13impl Transaction {
14    pub(crate) async fn remove_pending_messages_by_conversation_id(&self, conversation_id: impl AsRef<[u8]> + Send) {
15        let conversation_id = conversation_id.as_ref().to_vec().into();
16        self.bulk_remove::<MlsPendingMessage, ConversationId>(conversation_id)
17            .await;
18    }
19
20    pub(crate) async fn find_pending_messages_by_conversation_id(
21        &self,
22        conversation_id: &[u8],
23        persisted_records: impl IntoIterator<Item = Arc<MlsPendingMessage>>,
24    ) -> CryptoKeystoreResult<Vec<Arc<MlsPendingMessage>>> {
25        let conversation_id = conversation_id.to_vec().into();
26        Ok(self.search(persisted_records, &conversation_id).await.collect())
27    }

29    /// Find a free proteus prekey id.
30    ///
31    /// Prefers an id freed by a deletion in this transaction, which is not necessarily the lowest
32    /// free id overall but is quite fast to compute. Otherwise returns the lowest id which is free
33    /// in the database and not already claimed by this transaction, or 1 if no prekey exists at all.
34    ///
35    /// NOTE: repeated calls to this method will produce the same value until and unless someone
36    /// adds or deletes another prekey.
37    #[cfg(feature = "proteus-keystore")]
38    pub(crate) async fn free_proteus_prekey_id(&self) -> CryptoKeystoreResult<u16> {
39        fn as_prekey_id(entity_id: &EntityId) -> Option<u16> {
40            entity_id.primary_key::<ProteusPrekey>().map(Arc::unwrap_or_clone)
41        }
42
43        // if we can find a deleted id, we don't need to touch the DB at all;
44        // that's worth a scan through the operations
45        {
46            let operations = self.operations.read().await;
47            if let Some(free) = operations
48                .delete_indices_for_type::<ProteusPrekey>()
49                .find_map(|delete_idx| {
50                    let entity_id = operations[delete_idx]
51                        .as_delete::<ProteusPrekey>()
52                        .expect("delete_indices_for_type has correct indices for delete operations");
53                    operations
54                        .last_upsert_idx_for(&entity_id)
55                        .is_none_or(|upsert_idx| upsert_idx < delete_idx)
56                        .then(|| as_prekey_id(&entity_id))?
57                })
58            {
59                return Ok(free);
60            }
61        }
62
63        // assumption: nobody is going to be assigning auto prekeys while also assigning manual prekeys, so if multiple
64        // prekeys exist in our cache, they'll exist in a continuous run.
65        // Technically this assumption means that this is a partial function, in the sense that it never returns
66        // a wrong free prekey id, but may incorrectly report that no free prekey id exists.
67        // Practically, that should never actually happen.
68        let conn = self.database.conn().await;
69        let db_free_id = ProteusPrekey::free_id_greater_than(&conn, 0)?;
70        let Some((low, high)) = ({
71            use itertools::Itertools as _;
72
73            let operations = self.operations.read().await;
74            operations.upsert_indices_for_type::<ProteusPrekey>()
75                .filter(|&(entity_id, idx)|
76                    operations.last_delete_idx_for(entity_id)
77                        .is_none_or(|delete_idx| delete_idx < idx))
78                .map(|(_entity_id, idx)| {
79                    let entity = operations[idx]
80                        .as_upsert::<ProteusPrekey>()
81                        .expect("upsert_indices_for_type has correct data");
82                    entity.id
83                })
84                // exclude the last resort prekey, if we happen to have set that
85                .filter(|id| *id != u16::MAX)
86                .minmax()
87                .into_option()
88        }) else {
89            // no prekeys already added to the cache, so no possible conflict
90            return Ok(db_free_id);
91        };
92
93        if db_free_id < low || db_free_id > high {
94            // db produced a value which is outside the range of values in the cache;
95            // no conflict is possible
96            return Ok(db_free_id);
97        }
98
99        // the database's next free id is in the range within our cache.
100        // As we assume that the cached prekeys are contiguous, we just ask the database
101        // for the next one.
102        ProteusPrekey::free_id_greater_than(&conn, high)
103    }
104}

106#[cfg(all(test, feature = "proteus-keystore", not(target_os = "unknown")))]
107mod tests {
108    use crate::{Database, entities::ProteusPrekey};
109
110    fn prekey(id: u16) -> ProteusPrekey {
111        ProteusPrekey::from_raw(id, vec![id as u8; 8])
112    }
113
114    /// A database whose prekey table holds `1..=3`, so its own next free id is 4.
115    async fn store_with_three_prekeys() -> std::sync::Arc<Database> {
116        let store = Database::open_in_memory().unwrap();
117        store
118            .transactionally(async |tx| {
119                for id in 1..=3 {
120                    tx.save(prekey(id)).await?;
121                }
122                Ok(())
123            })
124            .await
125            .unwrap();
126        store
127    }
128
129    /// An id freed by a deletion in this transaction wins over the database's next free id.
130    #[macro_rules_attribute::apply(smol_macros::test)]
131    async fn free_prekey_id_prefers_an_id_deleted_in_this_transaction() {
132        let store = store_with_three_prekeys().await;
133        let tx = store.new_transaction().await.unwrap();
134
135        assert_eq!(
136            tx.free_proteus_prekey_id().await.unwrap(),
137            4,
138            "with no deletions staged, the id has to come from the database"
139        );
140
141        tx.remove::<ProteusPrekey>(&2).await.unwrap();
142
143        assert_eq!(
144            tx.free_proteus_prekey_id().await.unwrap(),
145            2,
146            "the staged deletion frees id 2, which is preferred over the database's id 4"
147        );
148    }
149
150    /// Saving a prekey under an id staged for deletion nops that deletion, so the id stops being
151    /// offered and the next call moves on to the other staged deletion.
152    ///
153    /// The deleted set is unordered, so which of the two ids comes back first is unspecified;
154    /// the property under test is that a claimed id is not handed out twice.
155    #[macro_rules_attribute::apply(smol_macros::test)]
156    async fn claiming_a_freed_prekey_id_uncovers_the_next_one() {
157        let store = store_with_three_prekeys().await;
158        let tx = store.new_transaction().await.unwrap();
159
160        tx.remove::<ProteusPrekey>(&2).await.unwrap();
161        tx.remove::<ProteusPrekey>(&3).await.unwrap();
162
163        let first = tx.free_proteus_prekey_id().await.unwrap();
164        assert!(
165            [2, 3].contains(&first),
166            "one of the two staged deletions must be offered, got {first}"
167        );
168        tx.save(prekey(first)).await.unwrap();
169
170        let second = tx.free_proteus_prekey_id().await.unwrap();
171        assert_ne!(
172            second, first,
173            "claiming {first} must nop its deletion so it is no longer free"
174        );
175        assert!(
176            [2, 3].contains(&second),
177            "the remaining staged deletion must be offered, got {second}"
178        );
179        tx.save(prekey(second)).await.unwrap();
180
181        assert_eq!(
182            tx.free_proteus_prekey_id().await.unwrap(),
183            4,
184            "with both staged deletions claimed, the id has to come from the database again"
185        );
186    }
187
188    /// The same, for a prekey which was saved and then deleted within this one transaction:
189    /// the deletion is still the last operation for that id, so it must still be nop'd on re-save.
190    #[macro_rules_attribute::apply(smol_macros::test)]
191    async fn claiming_a_prekey_id_deleted_after_being_saved_in_the_same_transaction() {
192        let store = Database::open_in_memory().unwrap();
193        let tx = store.new_transaction().await.unwrap();
194
195        tx.save(prekey(1)).await.unwrap();
196        tx.remove::<ProteusPrekey>(&1).await.unwrap();
197
198        assert_eq!(
199            tx.free_proteus_prekey_id().await.unwrap(),
200            1,
201            "the staged deletion frees id 1 again"
202        );
203
204        tx.save(prekey(1)).await.unwrap();
205
206        assert_eq!(
207            tx.free_proteus_prekey_id().await.unwrap(),
208            2,
209            "claiming 1 must nop its deletion, so 1 is no longer free"
210        );
211    }
