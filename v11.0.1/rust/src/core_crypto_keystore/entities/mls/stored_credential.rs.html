11/// This type exists so that we can efficiently search for credentials by a variety of metrics at the database level.
12///
13/// This includes some but not all of the fields in `core_crypto::CredentialFindFilters`: those that are actually stored
14/// in the database, and do not require deserializing the `credential` field.
15#[derive(Debug, Default, Clone, Copy, serde::Serialize)]
16pub struct CredentialFindFilters<'a> {
17    /// Hash of public key to search for.
18    pub hash: Option<Sha256Hash>,
19    /// Credential type to search for.
20    pub credential_type: Option<u16>,
21    /// Public key to search for
22    pub public_key: Option<&'a [u8]>,
23    /// Session / Client id to search for
24    pub session_id: Option<&'a [u8]>,
25    /// Ciphersuite to search for
26    pub ciphersuite: Option<u16>,
27    /// unix timestamp (seconds) of point of earliest validity to search for
28    pub earliest_validity: Option<u64>,
29}

31/// Entity representing a persisted `Credential`
32#[derive(core_crypto_macros::Debug, Clone, PartialEq, Eq, Zeroize, serde::Serialize, serde::Deserialize)]
33#[zeroize(drop)]
34pub struct StoredCredential {
35    /// Note: this is not a unique identifier, but the session id this credential belongs to.
36    #[sensitive]
37    pub session_id: Vec<u8>,
38    #[sensitive]
39    pub credential: Vec<u8>,
40    pub created_at: u64,
41    pub ciphersuite: u16,
42    #[sensitive]
43    pub public_key: Vec<u8>,
44    pub credential_type: u16,
45    /// Unix timestamp in seconds of the leaf certificate's `not_before` for an X509 credential, and 0 for any other.
46    ///
47    /// Part of the primary key: it tells apart X509 credentials which certify the same signature key, and can be
48    /// read from a leaf node. [`credential_not_before`] computes it from the credential.
49    pub not_before: u64,
50    #[sensitive]
51    pub private_key: Vec<u8>,
52}
53
54impl StoredCredential {
55    /// The `not_before` of a credential without a certificate, such as a basic credential.
56    pub const NOT_BEFORE_WITHOUT_CERTIFICATE: u64 = 0;
57
58    const PRIMARY_KEY_COLUMN_NAMES: [&str; 3] = ["public_key_sha256", "credential_type", "not_before"];
59
60    fn from_row(row: &rusqlite::Row<'_>) -> rusqlite::Result<Self> {
61        Ok(Self {
62            session_id: row.get("session_id")?,
63            credential: row.get("credential")?,
64            created_at: row.get("created_at")?,
65            ciphersuite: row.get("ciphersuite")?,
66            public_key: row.get("public_key")?,
67            credential_type: row.get("credential_type")?,
68            not_before: row.get("not_before")?,
69            private_key: row.get("private_key")?,
70        })
71    }
72
73    /// Update `self.created_at` to the current time.
74    pub fn pre_save(&mut self) -> CryptoKeystoreResult<()> {
75        #[cfg(not(target_os = "unknown"))]
76        use std::time::{SystemTime, UNIX_EPOCH};
77
78        #[cfg(target_os = "unknown")]
79        use web_time::{SystemTime, UNIX_EPOCH};
80
81        self.created_at = SystemTime::now()
82            .duration_since(UNIX_EPOCH)
83            .map_err(|_| crate::CryptoKeystoreError::TimestampError)?
84            .as_secs();
85
86        Ok(())
87    }
88}

90/// The `not_before` part of a stored credential's primary key.
91///
92/// For an X509 credential this is the leaf certificate's `not_before`, as a unix timestamp in seconds. Any other
93/// credential has no certificate, and gets [`StoredCredential::NOT_BEFORE_WITHOUT_CERTIFICATE`].
94pub fn credential_not_before(credential: &MlsCredential) -> CryptoKeystoreResult<u64> {
95    let MlsCredentialType::X509(certificate) = credential.mls_credential() else {
96        return Ok(StoredCredential::NOT_BEFORE_WITHOUT_CERTIFICATE);
97    };
98    let leaf = certificate
99        .certificates
100        .first()
101        .ok_or(CryptoKeystoreError::InvalidX509Credential(
102            "the certificate chain is empty".into(),
103        ))?;
104    let leaf = x509_cert::Certificate::from_der(leaf.as_slice())
105        .map_err(|e| CryptoKeystoreError::InvalidX509Credential(format!("decoding the leaf certificate: {e}")))?;
106    Ok(leaf
107        .tbs_certificate()
108        .validity()
109        .not_before
110        .to_unix_duration()
111        .as_secs())
112}

114#[derive(Debug, Clone, Hash, PartialEq, Eq, derive_more::Constructor)]
115pub struct StoredCredentialPk {
116    pub public_key_hash: Sha256Hash,
117    pub credential_type: u16,
118    /// See [`StoredCredential::not_before`].
119    pub not_before: u64,
120}
121
122impl StoredCredentialPk {
123    /// The primary key of the stored credential holding this signature key and credential.
124    ///
125    /// This is how a leaf node finds its credential: both parts are in the leaf node. Fails if the credential is
126    /// X509 and its leaf certificate can't be decoded.
127    pub fn from_mls_credential(signature_key: &[u8], credential: &MlsCredential) -> CryptoKeystoreResult<Self> {
128        Ok(Self {
129            public_key_hash: Sha256Hash::hash_from(signature_key),
130            credential_type: credential.credential_type().into(),
131            not_before: credential_not_before(credential)?,
132        })
133    }
134}

149impl crate::traits::Entity for StoredCredential {
150    const TABLE_NAME: &'static str = "mls_credentials";
151
152    fn get(conn: &rusqlite::Connection, key: &Self::PrimaryKey) -> crate::CryptoKeystoreResult<Option<Self>> {
153        use rusqlite::OptionalExtension as _;
154
155        conn.prepare_cached(
156            "SELECT session_id, credential, unixepoch(created_at) AS created_at, ciphersuite, public_key, \
157                    credential_type, not_before, private_key \
158             FROM mls_credentials WHERE public_key_sha256 = ? AND credential_type = ? AND not_before = ?",
159        )?
160        .query_row(
161            rusqlite::params![key.public_key_hash, key.credential_type, key.not_before],
162            Self::from_row,
163        )
164        .optional()
165        .map_err(Into::into)
166    }
167
168    fn count(conn: &rusqlite::Connection) -> crate::CryptoKeystoreResult<u32> {
169        crate::ancillary::helpers::count_helper::<Self>(conn)
170    }
171
172    fn load_all(conn: &rusqlite::Connection) -> crate::CryptoKeystoreResult<Vec<Self>> {
173        let mut statement = conn.prepare_cached(
174            "SELECT session_id, credential, unixepoch(created_at) AS created_at, ciphersuite, public_key, \
175                    credential_type, not_before, private_key \
176             FROM mls_credentials",
177        )?;
178        statement
179            .query_map([], Self::from_row)?
180            .collect::<Result<_, _>>()
181            .map_err(Into::into)
182    }
183}

185impl crate::traits::EntityDatabaseMutation for StoredCredential {
186    fn save<'a, Tx>(&self, tx: &'a Tx) -> crate::CryptoKeystoreResult<()>
187    where
188        &'a Tx: Into<Transactionlike<'a>>,
189    {
190        let conn = tx.into().conn()?;
191        let mut stmt = conn.prepare_cached(
192            "INSERT INTO mls_credentials \
193             (public_key_sha256, credential_type, not_before, public_key, session_id, credential, created_at, ciphersuite, private_key) \
194             VALUES \
195             (:public_key_sha256, :credential_type, :not_before, :public_key, :session_id, :credential, datetime(:created_at, 'unixepoch'), :ciphersuite, :private_key)",
196        )?;
197        stmt.execute(rusqlite::named_params![
198            ":public_key_sha256": Sha256Hash::hash_from(&self.public_key),
199            ":credential_type": self.credential_type,
200            ":not_before": self.not_before,
201            ":public_key": self.public_key,
202            ":session_id": self.session_id,
203            ":credential": self.credential,
204            ":created_at": self.created_at,
205            ":ciphersuite": self.ciphersuite,
206            ":private_key": self.private_key,
207        ])
208        .map_err(CryptoKeystoreError::map_already_exists(StoredCredential::TABLE_NAME))?;
209
210        Ok(())
211    }
212
213    fn delete<'a, Tx>(tx: &'a Tx, id: &Self::PrimaryKey) -> crate::CryptoKeystoreResult<bool>

226impl<'a> crate::traits::SearchableEntity<CredentialFindFilters<'a>> for StoredCredential {
227    fn find_all_matching(
228        conn: &rusqlite::Connection,
229        filters: &CredentialFindFilters<'a>,
230    ) -> crate::CryptoKeystoreResult<Vec<Self>> {
231        let hash = filters
232            .hash
233            .or_else(|| filters.public_key.map(crate::Sha256Hash::hash_from));
234        let mut query =
235            "SELECT session_id, credential, unixepoch(created_at) AS created_at, ciphersuite, public_key, credential_type, \
236                    not_before, private_key \
237             FROM mls_credentials \
238             WHERE (true OR :hash OR :credential_type OR :ciphersuite OR :created_at OR :session_id) "
239                .to_owned();
240
241        if hash.is_some() {
242            query.push_str("AND public_key_sha256 = :hash ");
243        }
244        if filters.credential_type.is_some() {
245            query.push_str("AND credential_type = :credential_type ");
246        }
247        if filters.ciphersuite.is_some() {
248            query.push_str("AND ciphersuite = :ciphersuite ");
249        }
250        if filters.earliest_validity.is_some() {
251            query.push_str("AND unixepoch(created_at) = :created_at ");
252        }
253        if filters.session_id.is_some() {
254            query.push_str("AND session_id = :session_id ");
255        }
256
257        let mut stmt = conn.prepare(&query)?;
258        stmt.query_map(
259            rusqlite::named_params![
260                ":hash": hash,
261                ":credential_type": filters.credential_type,
262                ":ciphersuite": filters.ciphersuite,
263                ":created_at": filters.earliest_validity,
264                ":session_id": filters.session_id,
265            ],
266            Self::from_row,
267        )?
268        .collect::<Result<_, _>>()
269        .map_err(Into::into)
270    }
271
272    fn matches(&self, filters: &CredentialFindFilters<'a>) -> bool {
273        filters
274            .hash
275            .is_none_or(|hash| hash == Sha256Hash::hash_from(&self.public_key))
276            && filters
277                .credential_type
278                .is_none_or(|credential_type| credential_type == self.credential_type)
279            && filters.public_key.is_none_or(|pk| pk == self.public_key)
280            && filters.session_id.is_none_or(|sid| sid == self.session_id)
281            && filters.ciphersuite.is_none_or(|cs| cs == self.ciphersuite)
282            && filters.earliest_validity.is_none_or(|ev| ev == self.created_at)
283    }
