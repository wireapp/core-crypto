9/// Identifies one database record: which table it lives in, plus its primary key within that table.
10///
11/// This is the type-erased counterpart to an [`Entity`]'s primary key. Equality and hashing cover
12/// both fields, so two entities of different types never collide even if their key bytes agree, and
13/// an id built from a borrowed key compares equal to one built from the owned form.
14///
15/// ## Invariant
16///
17/// [`Entity::TABLE_NAME`] has to identify the Rust type uniquely. This type only ever compares table
18/// names ([`Self::matches_type`]), while the operations which act on the entity behind an id recover
19/// the type by downcasting, so the two notions of identity have to agree. Were two entity types ever
20/// to share a table name, `matches_type` would answer `true` for both while the downcast succeeded
21/// for only one — turning the `expect`s which pair the two into panics, and making bulk deletions
22/// silently match nothing. Adding an entity therefore means giving it a table name no other entity
23/// uses.
24#[derive(Debug, Clone, PartialEq, Eq, Hash)]
25pub(crate) struct EntityId {
26    /// the table this record lives in, which also identifies its Rust type
27    table_name: &'static str,
28    /// the record's primary key, in the byte encoding given by [`KeyType::bytes`]
29    id: Vec<u8>,
30}
31
32impl fmt::Display for EntityId {
33    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
34        let Self { table_name, id } = self;
35        write!(f, "{table_name:?}: {}", hex::encode(id))
36    }
37}
38
39impl EntityId {
40    /// `true` when this id refers to a record of type `E`
41    ///
42    /// This is a table name comparison, which stands in for a type comparison by the invariant
43    /// documented on [`EntityId`].
44    pub(crate) fn matches_type<E>(&self) -> bool
45    where
46        E: Entity,
47    {
48        self.table_name == E::TABLE_NAME
49    }
50
51    /// Recover the typed primary key from this id.
52    ///
53    /// Errors with [`CryptoKeystoreError::InvalidPrimaryKeyBytes`] if the stored bytes are not a valid
54    /// encoding of `E::PrimaryKey`, which for an id built by one of the constructors below can only
55    /// happen if `E` is the wrong type.
56    pub(crate) fn primary_key<E>(&self) -> CryptoKeystoreResult<E::PrimaryKey>
57    where
58        E: Entity,
59    {
60        // we'd prefer not to pay the cost for a runtime check, but we want more than 0 checks
61        debug_assert!(
62            self.matches_type::<E>(),
63            "well-constructed code will never call this for a non-matching type"
64        );
65        E::PrimaryKey::from_bytes(&self.id).ok_or(CryptoKeystoreError::InvalidPrimaryKeyBytes(self.table_name))
66    }
67
68    /// Build an id for a record of type `E` from the byte encoding of its primary key.
69    ///
70    /// The other constructors all funnel through here; prefer them where one fits, as they derive the
71    /// encoding rather than trusting the caller to have produced it.
72    pub(crate) fn from_key<E>(primary_key: Cow<'_, [u8]>) -> Self
73    where
74        E: Entity,
75    {
76        Self {
77            id: primary_key.into_owned(),
78            table_name: E::TABLE_NAME,
79        }
80    }
81
82    /// Build the id of an entity instance.
83    ///
84    /// Note that this asks the entity for its primary key, which for some entities means hashing
85    /// their contents. Where the id is already available — as a cache key, say — reuse it rather than
86    /// deriving it again.
87    pub(crate) fn from_entity<E>(entity: &E) -> Self
88    where
89        E: Entity,
90    {
91        Self::from_key::<E>(entity.primary_key().bytes())
92    }
93
94    /// Build the id of the record of type `E` with this primary key.
95    pub(crate) fn from_primary_key<E>(primary_key: &E::PrimaryKey) -> Self
96    where
97        E: Entity,
98    {
99        Self::from_key::<E>(primary_key.bytes())
100    }
101
102    /// Build the id of the record of type `E` with this primary key, in its borrowed form.
103    ///
104    /// Equal to what [`Self::from_primary_key`] produces for the owned form of the same key, so the
105    /// two are interchangeable for lookups.
106    pub(crate) fn from_borrowed_primary_key<E>(primary_key: &E::BorrowedPrimaryKey) -> Self
107    where
108        E: Entity + BorrowPrimaryKey,
109    {
110        Self::from_key::<E>(primary_key.to_owned().bytes())
111    }
112}
