11    /// Set up the common scenario for these tests: Alice has buffered Bob's commit, which removes
12    /// Charlie, because it refers to a proposal she never received.
13    ///
14    /// Bob's commit is merged on his side, so he is at epoch 2 while Alice and Charlie are still at epoch 1.
15    /// Returns the conversation and the serialized proposal Alice is missing.
16    async fn alice_buffers_commit_missing_proposal<'a>(
17        case: &'a TestContext,
18        [alice, bob, charlie]: [&'a SessionContext; 3],
19    ) -> (TestConversation<'a>, Vec<u8>) {
20        let conversation = case.create_conversation([alice, bob, charlie]).await;
21
22        let proposal_guard = conversation.acting_as(bob).await.remove_proposal(charlie).await;
23        let missing_proposal = proposal_guard.message().to_bytes().unwrap();
24        let conversation = proposal_guard.finish();
25
26        let commit_guard = conversation.acting_as(bob).await.commit_pending_proposals().await;
27        let (commit_guard, result) = commit_guard.notify_member_fallible(alice).await;
28        assert!(
29            matches!(result.unwrap().into_single_deferred(), DeferralReason::MissingProposal),
30            "a commit referencing an unknown proposal must be deferred until the proposal arrives"
31        );
32        let conversation = commit_guard.finish();
33        assert_eq!(
34            alice.transaction.count_entities().await.buffered_commits,
35            1,
36            "the commit Alice could not apply must have been buffered"
37        );
38
39        (conversation, missing_proposal)
40    }

42    /// A proposal which doesn't complete the buffered commit must leave it buffered.
43    ///
44    /// Every incoming proposal retries the buffered commit. When the retry reports that proposals are
45    /// still missing, the commit must stay where it is, so that it can apply once the proposal it is
46    /// actually waiting for arrives. Discarding it instead would silently lose that epoch.
47    #[apply(all_cred_cipher)]
48    async fn unrelated_proposal_must_not_discard_buffered_commit(case: TestContext) {
49        let [alice, bob, charlie] = case.sessions().await;
50        let (conversation, missing_proposal) =
51            alice_buffers_commit_missing_proposal(&case, [&alice, &bob, &charlie]).await;
52
53        // Charlie, still at epoch 1, sends an unrelated proposal. Alice accepts it, but it doesn't
54        // complete Bob's commit.
55        let proposal_guard = conversation.acting_as(&charlie).await.remove_proposal(&bob).await;
56        let (proposal_guard, result) = proposal_guard.notify_member_fallible(&alice).await;
57        assert!(
58            matches!(
59                result.as_deref(),
60                Ok([DecryptionOutcome::Decrypted(DecryptedMessage::Proposal(_))])
61            ),
62            "the unrelated proposal must decrypt as a proposal, and nothing else: {result:?}"
63        );
64        let conversation = proposal_guard.finish();
65        assert_eq!(
66            alice.transaction.count_entities().await.buffered_commits,
67            1,
68            "the commit is still missing a proposal, so it must still be buffered"
69        );
70
71        // The proposal Bob's commit is waiting for arrives, and the commit applies.
72        let decrypted = conversation
73            .guard_of(&alice)
74            .await
75            .decrypt_message(&missing_proposal)
76            .await
77            .unwrap();
78        assert!(
79            matches!(
80                decrypted.as_slice(),
81                [
82                    DecryptionOutcome::Decrypted(DecryptedMessage::Proposal(_)),
83                    DecryptionOutcome::Decrypted(DecryptedMessage::Commit(_))
84                ]
85            ),
86            "the missing proposal must decrypt, then apply the buffered commit: {decrypted:?}"
87        );
88    }

163    /// A message in the inbox which can't be decoded must not stop a buffered commit from applying.
164    ///
165    /// When a buffered commit applies, it replays whatever was waiting for its new epoch. A message
166    /// there which can't even be parsed can never be replayed, so it is dropped, and the commit goes
167    /// ahead. If it failed the commit instead, the epoch would be lost for the sake of a message
168    /// which could never be read anyway.
169    #[apply(all_cred_cipher)]
170    async fn undecodable_inbox_message_must_not_block_buffered_commit(case: TestContext) {
171        let [alice, bob, charlie] = case.sessions().await;
172        let (conversation, missing_proposal) =
173            alice_buffers_commit_missing_proposal(&case, [&alice, &bob, &charlie]).await;
174        let conversation_id = conversation.id().clone();
175
176        // A protocol version and a public-message wire format, then a group id claiming five bytes
177        // which aren't there: decoding the message fails after the wire format check.
178        // It waits for epoch 2, which Bob's buffered commit moves Alice into, so that the commit replays it.
179        {
180            let transaction = alice.transaction.inner().unwrap();
181            let undecodable = InboxMessage {
182                conversation_id: conversation_id.as_ref().into(),
183                message: vec![0, 1, 0, 1, 5],
184                content_type: ContentType::Application,
185                waiting_for: WaitingFor::Epoch,
186                epoch: 2,
187                arrival_sequence: transaction.transaction().inbox_message_next_arrival_sequence(),
188            };
189            undecodable.save(transaction.transaction()).unwrap();
190        }
191
192        {
193            let mut alice_conversation = conversation.guard_of(&alice).await;
194            let decrypted = alice_conversation.decrypt_message(&missing_proposal).await;
195            assert!(
196                matches!(
197                    decrypted.as_deref(),
198                    Ok([
199                        DecryptionOutcome::Decrypted(DecryptedMessage::Proposal(_)),
200                        DecryptionOutcome::Decrypted(DecryptedMessage::Commit(_))
201                    ])
202                ),
203                "the proposal completes the buffered commit, which must apply, \
204                 and the undecodable message must not be reported as replayed: {decrypted:?}"
205            );
206            assert_eq!(
207                alice_conversation.epoch().await,
208                2,
209                "the in-memory group must have moved to the commit's epoch"
210            );
211        }
212
213        // Load the group afresh from the database.
214        alice
215            .session()
216            .await
217            .conversation_cache
218            .lock()
219            .await
220            .remove(&conversation_id);
221        {
222            let reloaded = alice.transaction.conversation(&conversation_id).await.unwrap();
223            assert_eq!(
224                reloaded.epoch().await,
225                2,
226                "the stored group must have moved to the commit's epoch"
227            );
228        }
229
230        let counts = alice.transaction.count_entities().await;
231        assert_eq!(
232            counts.buffered_commits, 0,
233            "the applied commit must have left the inbox"
234        );
235        assert_eq!(
236            counts.pending_messages, 0,
237            "the undecodable message must have been dropped from the inbox"
238        );
239    }

241    /// A buffered commit must be discarded once the group moves past its epoch.
242    ///
243    /// A commit waiting for a missing proposal is only ever valid in the epoch it was created for.
244    /// If the group bypasses the commit, the old commit must be discarded as stale.
245    #[apply(all_cred_cipher)]
246    async fn stale_buffered_commit_is_discarded_on_epoch_advance(case: TestContext) {
247        let [alice, bob, charlie] = case.sessions().await;
248        let conversation = case.create_conversation([&alice, &bob, &charlie]).await;
249
250        // Bob proposes removing Charlie at epoch 1, but nobody else is told about the proposal.
251        let conversation = conversation
252            .acting_as(&bob)
253            .await
254            .remove_proposal(&charlie)
255            .await
256            .finish();
257
258        // Bob then commits it. The commit refers to the proposal by reference, so Alice, who never
259        // received that proposal, cannot apply the commit, and buffers it to retry once she does.
260        let commit_guard = conversation
261            .acting_as(&bob)
262            .await
263            .commit_pending_proposals_unmerged()
264            .await;
265        let (commit_guard, result) = commit_guard.notify_member_fallible(&alice).await;
266        assert!(
267            matches!(result.unwrap().into_single_deferred(), DeferralReason::MissingProposal),
268            "a commit referencing an unknown proposal must be deferred until the proposal arrives"
269        );
270        let conversation = commit_guard.finish();
271        assert_eq!(
272            alice.transaction.count_entities().await.buffered_commits,
273            1,
274            "the commit Alice could not apply must have been buffered"
275        );
276
277        // Before the proposal arrives, Alice updates her key material, and her commit wins:
278        // everyone moves to epoch 2, and Bob's commit is discarded. Bob's buffered commit belongs
279        // to epoch 1, so it can never apply now.
280        conversation
281            .acting_as(&alice)
282            .await
283            .update()
284            .await
285            .notify_member(&bob)
286            .await
287            .notify_member(&charlie)
288            .await
289            .finish();
290
291        assert_eq!(
292            alice.transaction.count_entities().await.buffered_commits,
293            0,
294            "the stale buffered commit can never apply, so it must have been discarded"
295        );
296    }
