1//! Populate `mls_inbox` from `mls_pending_messages` and `mls_buffered_commits`.
2//!
3//! The old tables don't record why a message is waiting, nor its epoch or content type. All three have to be
4//! recovered by parsing each message and comparing it with the current state of its group.
5

18const TRANSIENT_WIRE_FORMAT: u16 = 0xF000;
19const TARGETED_WIRE_FORMAT: u16 = 0xF001;
20const TRANSIENT_TARGETED_WIRE_FORMAT: u16 = 0xF002;
21const TNT_WIRE_FORMATS: std::ops::RangeInclusive<u16> = TRANSIENT_WIRE_FORMAT..=TRANSIENT_TARGETED_WIRE_FORMAT;
22
23pub(crate) fn meta_migration(tx: &rusqlite::Transaction<'_>) -> CryptoKeystoreResult<()> {
24    // the inbox table was just created
25    let mut arrival_sequence = 0;
26    let mut write_inbox_message =
27        move |tx, conversation_id, message, content_type, waiting_for, epoch| -> CryptoKeystoreResult<()> {
28            arrival_sequence += 1;
29            InboxMessage {
30                conversation_id,
31                message,
32                content_type,
33                waiting_for,
34                epoch,
35                arrival_sequence,
36            }
37            .save(tx)?;
38            Ok(())
39        };
40
41    let mut group_data_stmt = tx.prepare_cached("SELECT is_pending, epoch FROM mls_groups WHERE id = ?")?;
42
43    // handle pending messages
44    {
45        let mut stmt = tx.prepare("SELECT conversation_id, message FROM mls_pending_messages ORDER BY rowid")?;
46        let rows = stmt.query_map([], |row| {
47            Ok((
48                row.get::<_, ConversationId>("conversation_id")?,
49                row.get::<_, Vec<u8>>("message")?,
50            ))
51        })?;
52        for maybe_row in rows {
53            let (conversation_id, message) = maybe_row?;
54            let (group_is_pending, group_epoch) = group_data_stmt.query_one([&conversation_id], |row| {
55                Ok((row.get::<_, bool>(0)?, row.get::<_, u64>(1)?))
56            })?;
57            let Some((content_type, msg_epoch)) = parse_message(&message) else {
58                // anything we can't parse gets skipped, which ends up dropping it
59                warn!(conversation_id, group_epoch, group_is_pending; "migration failed to parse a pending message; dropping");
60                continue;
61            };
62
63            let waiting_for = if group_is_pending {
64                WaitingFor::PendingJoin
65            } else if msg_epoch > group_epoch {
66                WaitingFor::Epoch
67            } else {
68                // we can't do anything about this message; it's either stale or failed to pull out when we joined the
69                // current epoch, and nothing will pull it anymore. So just skip it.
70                warn!(msg_epoch, group_epoch; "migration found a message in the pending queue which should already have been replayed; dropping");
71                continue;
72            };
73
74            write_inbox_message(tx, conversation_id, message, content_type, waiting_for, msg_epoch)?;
75        }
76    }
77
78    // handle buffered commits
79    {
80        let mut stmt = tx.prepare("SELECT conversation_id, commit_data FROM mls_buffered_commits")?;
81        let rows = stmt.query_map([], |row| {
82            Ok((
83                row.get::<_, ConversationId>("conversation_id")?,
84                // commit data is nullable for some reason
85                row.get::<_, Option<Vec<u8>>>("commit_data")?,
86            ))
87        })?;
88        for maybe_row in rows {
89            let (conversation_id, message) = maybe_row?;
90            let Some(message) = message else {
91                // if `message` is `None`, no point in keeping this around
92                warn!(conversation_id; "migration found a buffered commit with no commit data; dropping");
93                continue;
94            };
95            let (group_is_pending, group_epoch) = group_data_stmt.query_one([&conversation_id], |row| {
96                Ok((row.get::<_, bool>(0)?, row.get::<_, u64>(1)?))
97            })?;
98            let Some((content_type, msg_epoch)) = parse_message(&message) else {
99                // anything we can't parse gets skipped, which ends up dropping it
100                warn!(conversation_id, group_epoch, group_is_pending; "migration failed to parse a buffered commit; dropping");
101                continue;
102            };
103
104            let waiting_for = if group_is_pending {
105                WaitingFor::PendingJoin
106            } else if content_type == ContentType::Commit && msg_epoch == group_epoch {
107                WaitingFor::Proposal
108            } else {
109                // we can't do anything about this message; it's either stale or weirdly from the future
110                warn!(msg_epoch, group_epoch; "migration found a message in the buffered commits which doesn't match the current epoch; dropping");
111                continue;
112            };
113
114            write_inbox_message(tx, conversation_id, message, content_type, waiting_for, msg_epoch)?;
115        }
116    }
117
118    Ok(())
119}

121/// Recover the content type and epoch of a message without interpreting it any further.
122fn parse_message(message: &[u8]) -> Option<(ContentType, u64)> {
123    // Bytes 2 and 3 are the wire format.
124    let wire_format = u16::from_be_bytes(message.get(2..4)?.try_into().ok()?);
125    if TNT_WIRE_FORMATS.contains(&wire_format) {
126        let epoch = parse_tnt_epoch(wire_format, message)?;
127        return Some((ContentType::Application, epoch));
128    }
129
130    let protocol_message: ProtocolMessage = match MlsMessageIn::tls_deserialize_exact(message).ok()?.extract() {
131        MlsMessageInBody::PublicMessage(message) => message.into(),
132        MlsMessageInBody::PrivateMessage(message) => message.into(),
133        _ => return None,
134    };
135
136    Some((protocol_message.content_type(), protocol_message.epoch().as_u64()))
137}

139/// compute the number of bytes occupied by a vlbytes field starting at `initial_offset`
140///
141/// returns `None` if the field can't be read or if it would overrun the `message`.
142fn vlbytes_field_length(initial_offset: usize, message: &[u8]) -> Option<usize> {
143    let length_of_length = match message.get(initial_offset)? >> 6 {
144        0b00 => 1,
145        0b01 => 2,
146        0b10 => 4,
147        _ => return None,
148    };
149    let mut length_encoded = [0; 4];
150    length_encoded[4 - length_of_length..]
151        .copy_from_slice(message.get(initial_offset..initial_offset + length_of_length)?);
152    length_encoded[4 - length_of_length] &= 0b11_1111; // clear the length_of_length bits from the leading byte
153    let field_length = length_of_length + u32::from_be_bytes(length_encoded) as usize;
154    (message.len() >= initial_offset + field_length).then_some(field_length)
155}

157/// Recover the epoch of a TNT message.
158///
159/// The keystore can't see `core-crypto`'s `TntMessage`, so this reads the wire format directly.
160///
161/// - Transient field layout: an MLS `GroupContext`, i.e. protocol version (u16), ciphersuite (u16), group id (variable
162///   length), then the epoch (u64).
163/// - Targeted and transient targeted field layout: counter (u32), sender (u32), recipient (u32), recipient id (variable
164///   length), then the epoch (u64).
165fn parse_tnt_epoch(wire_format: u16, message: &[u8]) -> Option<u64> {
166    let mut offset = 2 + 2; // protocol_version, wire_format
167    match wire_format {
168        TRANSIENT_WIRE_FORMAT => {
169            offset += 2 + 2; // protocol_version, ciphersuite
170            offset += vlbytes_field_length(offset, message)?; // group_id
171        }
172        TARGETED_WIRE_FORMAT | TRANSIENT_TARGETED_WIRE_FORMAT => {
173            offset += 4 + 4 + 4; // counter, sender_leaf, recipient_leaf
174            offset += vlbytes_field_length(offset, message)?; // recipient_id
175        }
176        _ => return None, // unknown wire format
177    }
178    message.get(offset..offset + 8)?.try_into().ok().map(u64::from_be_bytes)
179}
