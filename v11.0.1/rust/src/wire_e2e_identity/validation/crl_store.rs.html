14type IssuerMap = BTreeMap<String, Vec<usize>>;
15type SkidMap = BTreeMap<Vec<u8>, Vec<usize>>;

18pub(super) struct CrlStore {
19    crls: Arc<Mutex<Vec<CertificateList<Raw>>>>,
20    crl_info: Arc<Mutex<Vec<CrlInfo>>>,
21    issuers: Arc<Mutex<IssuerMap>>,
22    sk_ids: Arc<Mutex<SkidMap>>,
23    dps: Arc<Mutex<DpMap>>,
24}

26impl From<&[CertificateList<Raw>]> for CrlStore {
27    fn from(value: &[CertificateList<Raw>]) -> Self {
28        Self {
29            crls: Mutex::new(value.to_vec()).into(),
30            crl_info: Default::default(),
31            issuers: Default::default(),
32            sk_ids: Default::default(),
33            dps: Default::default(),
34        }
35    }
36}

38impl CrlStore {
39    /// Update maps that are used when searching for CRLs matching a given certificate.
40    fn populate_maps(&self, index: usize, info: &CrlInfo) {
41        if let Some(dp) = info.idp_blob.clone() {
42            self.dps
43                .lock()
44                .expect("not poisoned")
45                .entry(dp)
46                .or_default()
47                .push(index);
48        } else {
49            if let Some(akid) = info.skid.clone() {
50                self.sk_ids
51                    .lock()
52                    .expect("not poisoned")
53                    .entry(akid)
54                    .or_default()
55                    .push(index);
56            }
57
58            if info.type_info.scope == CrlScope::Complete {
59                self.issuers
60                    .lock()
61                    .expect("not poisoned")
62                    .entry(info.issuer_name.clone())
63                    .or_default()
64                    .push(index);
65            }
66        }
67    }
68}

70impl CrlSource for CrlStore {
71    fn get_all_crls(&self) -> certval::Result<Vec<Vec<u8>>> {
72        let crls = self.crls.lock().map_err(|_| certval::Error::Unrecognized)?;
73        crls.iter().try_fold(Vec::with_capacity(crls.len()), |mut acc, crl| {
74            acc.push(crl.to_der()?);
75            Ok(acc)
76        })
77    }
78
79    fn get_crls(&self, cert: &PDVCertificate) -> certval::Result<Vec<Vec<u8>>> {
80        let crls = self.crls.lock().map_err(|_| certval::Error::Unrecognized)?;
81        // DistributionPoint matching
82        if let Some(dps) = get_dps_from_cert(cert) {
83            let source_dps = self.dps.lock().map_err(|_| certval::Error::Unrecognized)?;
84            for dp in dps {
85                if let Some(indices) = source_dps.get(&dp) {
86                    let mut retval = vec![];
87                    for index in indices {
88                        if let Some(crl) = crls.get(*index) {
89                            retval.push(crl.to_der()?);
90                        }
91                    }
92                    return Ok(retval);
93                }
94            }
95        }
96
97        // AKI matching
98        if let Ok(Some(PDVExtension::AuthorityKeyIdentifier(akid))) =
99            cert.get_extension(&ID_CE_AUTHORITY_KEY_IDENTIFIER)
100            && let Some(kid) = &akid.key_identifier
101        {
102            let skids = self.sk_ids.lock().map_err(|_| certval::Error::Unrecognized)?;
103            let kid_bytes = kid.as_bytes();
104            if let Some(indices) = skids.get(kid_bytes) {
105                let mut retval = vec![];
106                for index in indices {
107                    if let Some(crl) = crls.get(*index) {
108                        retval.push(crl.to_der()?);
109                    }
110                }
111                return Ok(retval);
112            }
113        }
114
115        // Issuer fallback
116        let issuer_name = name_to_string(cert.decoded().tbs_certificate().issuer());
117        let issuers = self.issuers.lock().map_err(|_| certval::Error::Unrecognized)?;
118        if let Some(indices) = issuers.get(&issuer_name) {
119            let mut retval = vec![];
120            for index in indices {
121                if let Some(crl) = crls.get(*index) {
122                    retval.push(crl.to_der()?);
123                }
124            }
125            return Ok(retval);
126        }
127
128        Err(certval::Error::NotFound)
129    }
130
131    fn add_crl(&self, _: &[u8], crl: &CertificateList<Raw>, _: &str) -> certval::Result<()> {
132        let info = get_crl_info(crl)?;
133
134        let mut crls = self.crls.lock().expect("not poisoned");
135        let mut crl_infos = self.crl_info.lock().expect("not poisoned");
136
137        // see if we already have a matching CRL
138        let pos = crl_infos
139            .iter()
140            .position(|ci| ci.type_info == info.type_info && ci.idp_name == info.idp_name && ci.uri == info.uri);
141
142        match pos {
143            Some(idx) => {
144                // update the existing CRL and CRL info, but only if the new CRL is actually newer
145                if info.this_update < crl_infos[idx].this_update {
146                    return Err(certval::Error::Unrecognized);
147                }
148                crls[idx] = crl.clone();
149                crl_infos[idx] = info;
150                return Ok(());
151            }
152            None => {
153                crls.push(crl.clone());
154                crl_infos.push(info.clone());
155                self.populate_maps(crls.len() - 1, &info);
156            }
157        };
158
159        Ok(())
160    }
161}
