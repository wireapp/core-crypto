39/// The version of the Transient and Targeted Messages protocol.
40#[derive(Clone, Copy, PartialEq, Eq, TlsSize, TlsSerialize, TlsDeserialize)]
41#[repr(transparent)]
42struct ProtocolVersion(u16);
43
44impl ProtocolVersion {
45    /// Transient and Targeted Messages Version 1
46    const V1: Self = Self(1);
47
48    const CURRENT: Self = Self::V1;
49
50    const fn as_u16(&self) -> u16 {
51        self.0
52    }
53}

55/// The body of [TntMessageTBS], with currently three variants. The wire format is encoded in the leading two
56/// bytes of the TLS-serialized version of this struct. The wire-format values are contained in the range reserved for
57/// private use in RFC 9420.
58#[derive(TlsSize, TlsSerialize, TlsDeserialize)]
59#[repr(u16)]
60#[non_exhaustive]
61enum TntMessageBody {
62    /// A transient message.
63    #[tls_codec(discriminant = 0xF000)]
64    Transient(TransientMessage),
65    /// A targeted message.
66    #[tls_codec(discriminant = 0xF001)]
67    Targeted(TargetedMessage),
68    /// A transient, targeted message.
69    #[tls_codec(discriminant = 0xF002)]
70    TransientTargeted(TargetedMessage),
71}
72
73impl TntMessageBody {
74    /// Get the epoch to which this message applies
75    pub(crate) fn epoch(&self) -> u64 {
76        match self {
77            Self::Transient(transient) => transient.epoch().as_u64(),
78            Self::Targeted(targeted) | Self::TransientTargeted(targeted) => targeted.epoch.as_u64(),
79        }
80    }
81
82    /// `true` when this message is transient.
83    pub(crate) fn is_transient(&self) -> bool {
84        matches!(
85            self,
86            TntMessageBody::Transient(_) | TntMessageBody::TransientTargeted(_)
87        )
88    }
89}

91#[derive(Clone, Copy, TlsSize, TlsDeserialize, TlsSerialize, derive_more::From, PartialEq, Eq)]
92pub(crate) struct TntWireFormat(u16);
93
94impl TntWireFormat {
95    pub(crate) const TRANSIENT_MESSAGE: Self = Self(0xF000);
96
97    pub(crate) const TARGETED_MESSAGE: Self = Self(0xF001);
98    pub(crate) const TRANSIENT_TARGETED_MESSAGE: Self = Self(0xF002);
99
100    pub(crate) const MIN: u16 = Self::TRANSIENT_MESSAGE.0;
101    pub(crate) const MAX: u16 = Self::TRANSIENT_TARGETED_MESSAGE.0;
102
103    pub(crate) const ALL: std::ops::RangeInclusive<u16> = Self::MIN..=Self::MAX;
104
105    /// The current rx counter value of the given message type, which depends on the given sender, conversation, and
106    /// epoch.
107    ///
108    /// Implementation note: because transient targeted message and targeted message share a type but don't share a
109    /// counter table, we're implementing this here instead of on the types themselves.
110    async fn current_rx_counter(
111        &self,
112        database: &impl FetchFromDatabase,
113        primary_key: MessageRxCounterPkRef<'_>,
114    ) -> Result<u32> {
115        let count = match *self {
116            // Both transient message types share an rx counter table because they are sent over the
117            // same channel.
118            Self::TRANSIENT_MESSAGE | Self::TRANSIENT_TARGETED_MESSAGE => database
119                .get_borrowed::<TransientMessageRxCounter>(primary_key)
120                .await
121                .map_err(KeystoreError::wrap("getting TransientMessageRxCounter"))?
122                .map(|counter| counter.count)
123                .unwrap_or_default(),
124            // Targeted messages have their own rx counter type.
125            Self::TARGETED_MESSAGE => database
126                .get_borrowed::<TargetedMessageRxCounter>(primary_key)
127                .await
128                .map_err(KeystoreError::wrap("getting TargetedMessageRxCounter"))?
129                .map(|counter| counter.count)
130                .unwrap_or_default(),
131
132            _ => panic!("All TntWireFormats map to a MessageRxCounter implementation"),
133        };
134
135        Ok(count)
136    }
137
138    /// Perist the receiver counter for this message type.
139    ///
140    /// Implementation note: because transient targeted message and targeted message share a type but don't share a
141    /// counter table, we're implementing this here instead of on the types themselves.
142    fn persist_rx_message_counter(
143        &self,
144        group_epoch: u64,
145        message_sender: LeafNodeIndex,
146        message_counter: TntMessageCounter,
147        tx: &Transaction,
148        conversation_id: KeystoreConversationId,
149    ) -> Result<(), Error> {
150        match *self {
151            TntWireFormat::TRANSIENT_MESSAGE | TntWireFormat::TRANSIENT_TARGETED_MESSAGE => TransientMessageRxCounter {
152                conversation_id,
153                sender: message_sender.u32(),
154                epoch: group_epoch,
155                count: message_counter.into(),
156            }
157            .save(tx)
158            .map_err(KeystoreError::wrap("persisting TransientMessageRxCounter"))?,
159            TntWireFormat::TARGETED_MESSAGE => TargetedMessageRxCounter {
160                conversation_id,
161                sender: message_sender.u32(),
162                epoch: group_epoch,
163                count: message_counter.into(),
164            }
165            .save(tx)
166            .map_err(KeystoreError::wrap("persisting TargetedMessageRxCounter"))?,
167
168            _ => panic!("All TntWireFormats map to a MessageRxCounter implementation"),
169        }
170        Ok(())
171    }
172}

181impl TntMessageTBS {
182    fn new_transient(transient: TransientMessage) -> Self {
183        Self {
184            protocol_version: ProtocolVersion::CURRENT,
185            body: TntMessageBody::Transient(transient),
186        }
187    }
188
189    fn new_targeted(targeted: TargetedMessage) -> Self {
190        Self {
191            protocol_version: ProtocolVersion::CURRENT,
192            body: TntMessageBody::Targeted(targeted),
193        }
194    }
195
196    fn new_transient_targeted(targeted: TargetedMessage) -> Self {
197        Self {
198            protocol_version: ProtocolVersion::CURRENT,
199            body: TntMessageBody::TransientTargeted(targeted),
200        }
201    }

203    pub(super) fn sender_index(&self) -> LeafNodeIndex {
204        match &self.body {
205            TntMessageBody::Transient(transient_message) => transient_message.sender(),
206            TntMessageBody::Targeted(targeted_message) | TntMessageBody::TransientTargeted(targeted_message) => {
207                targeted_message.sender()
208            }
209        }
210    }

263impl ConversationMut {
264    /// Verify the [TntMessage] signature, then proceed with decryption.
265    pub(super) async fn decrypt_tnt_message(&self, message: TntMessage) -> Result<DecryptedMessage> {
266        let protocol_version = message.content.protocol_version;
267        // Currently, we only support `ProtocolVersion::V1`, so we're throwing an error if anyone from a future version
268        // sends us a message.
269        if protocol_version != ProtocolVersion::V1 {
270            return Err(Error::MlsMessageInvalidState("Unsupported tnt protocol version"));
271        }
272
273        let message_epoch = message.content.body.epoch();
274        let group_epoch = self.epoch().await;
275        let sender = if message_epoch == group_epoch {
276            // for current-epoch messages regardless of type it is ok to inspect the current group state to find the
277            // sender
278            self.group()
279                .await
280                .members()
281                .find(|member| member.index == message.content.sender_index())
282                .ok_or(Error::MlsMessageInvalidState(
283                    "tnt sender is not a group member in the current epoch",
284                ))?
285        } else if message_epoch > group_epoch {
286            if message_epoch <= group_epoch.saturating_add(MAX_FUTURE_EPOCHS) {
287                // returning the error here is sufficient to cause the buffering to happen,
288                // and should also be ok for transient messages; transience is a backend property
289                return Err(Error::BufferedFutureMessage { message_epoch });
290            }
291            return Err(Error::UnbufferedFarFutureMessage);
292        } else if message.content.body.is_transient() {
293            return Err(Error::InvalidTransientMessageEpoch {
294                group_epoch,
295                message_epoch,
296            });
297        } else if message_epoch < group_epoch.saturating_sub(MAX_PAST_EPOCHS as _) {
298            return Err(Error::MessageEpochTooOld);
299        } else {
300            // for older targeted messages we have to be able to look into the past and get old member data
301            // (FetchFromDatabase impl for Database arc-wraps internally, so this should never clone)
302            let oemd = Arc::unwrap_or_clone(
303                self.database()?
304                    .get_borrowed::<OldEpochMemberData>(OldEpochMemberDataPkRef::new(
305                        self.id.keystore(),
306                        message_epoch,
307                        message.content.sender_index().u32(),
308                    ))
309                    .await
310                    .map_err(KeystoreError::wrap("fetching old epoch member data"))?
311                    .ok_or(Error::MlsMessageInvalidState(
312                        "old epoch member data not found for the sender",
313                    ))?,
314            );
315            let credential = MlsCredential::tls_deserialize_exact(&oemd.credential)
316                .map_err(TlsCodecError::deserialize("deserializing old epoch member credential"))?;
317            Member {
318                index: oemd.leaf_node_index.into(),
319                signature_key: oemd.signature_key,
320                credential,
321                // we don't attempt to encrypt anything with this member, only decrypt
322                encryption_key: Default::default(),
323            }
324        };
325
326        let crypto_provider = self.crypto_provider().await?;
327        let signature_algorithm = self.cipher_suite().signature_algorithm();
328
329        let sender_public_key =
330            OpenMlsSignaturePublicKey::new(sender.signature_key.clone().into(), signature_algorithm)
331                .map_err(OpenMlsError::wrap("constructing tnt sender signature key"))?;
332        message
333            .verify_no_out(crypto_provider.crypto(), &sender_public_key)
334            .map_err(OpenMlsError::wrap("verifying tnt message signature"))?;
335
336        match message.content.body {
337            TntMessageBody::Transient(transient_message) => {
338                self.decrypt_transient(transient_message, protocol_version, &sender)
339                    .await
340            }
341            TntMessageBody::Targeted(targeted_message) => {
342                self.decrypt_targeted(
343                    targeted_message,
344                    TargetedMessagePolicy::Persisted,
345                    protocol_version,
346                    &sender,
347                )
348                .await
349            }
350            TntMessageBody::TransientTargeted(targeted_message) => {
351                self.decrypt_targeted(
352                    targeted_message,
353                    TargetedMessagePolicy::Transient,
354                    protocol_version,
355                    &sender,
356                )
357                .await
358            }
359        }
360    }

362    /// The final steps of each tnt message encryption procedure: sign, format as [TntMessage] and serialize.
363    async fn sign_tnt_message(&self, tbs: TntMessageTBS) -> Result<Vec<u8>> {
364        let credential = self.find_current_credential().await?;
365        let signature_key = credential.signature_key();
366        let signed_message: TntMessage = tbs
367            .sign(signature_key)
368            .map_err(OpenMlsError::wrap("signing TntMessageTBS"))?;
369
370        signed_message
371            .tls_serialize_detached()
372            .map_err(TlsCodecError::serialize("TntMessage"))
373            .map_err(Into::into)
374    }

376    /// Shared function for all tnt message types, parameterized by message type and sender, because that's how the
377    /// receiver counters are split.
378    /// The additional complexity introduced by this split is worthwhile, because each message type may be delivered in
379    /// a unique way. We don't want to assume that messages arrive in order across different message types. This is why
380    /// we're just checking the counter of the same message type, which is enough for duplicate message detection.
381    async fn ensure_no_duplicate_message(
382        &self,
383        group_epoch: u64,
384        message_type: TntWireFormat,
385        message_sender: LeafNodeIndex,
386        message_counter: TntMessageCounter,
387    ) -> Result<()> {
388        let database = self.database()?;
389        let counter_pk = MessageRxCounterPkRef::new(self.id.as_ref().into(), message_sender.u32(), group_epoch);
390        let existing_counter = message_type.current_rx_counter(database.as_ref(), counter_pk).await?;
391
392        if u32::from(message_counter) > existing_counter {
393            Ok(())
394        } else {
395            Err(Error::DuplicateMessage)
396        }
397    }

399    fn persist_rx_message_counter(
400        &self,
401        group_epoch: u64,
402        message_type: TntWireFormat,
403        message_sender: LeafNodeIndex,
404        message_counter: TntMessageCounter,
405    ) -> Result<()> {
406        let tx = self
407            .tx_context
408            .inner()
409            .map_err(RecursiveError::transaction("getting inner context"))?;
410        let tx = tx.transaction();
411        let conversation_id = self.id().into();
412
413        message_type.persist_rx_message_counter(group_epoch, message_sender, message_counter, tx, conversation_id)?;
414        Ok(())
415    }

417    /// The final step after decrypting a tnt message: extract the sender client id and format as [DecryptedBytes].
418    async fn extract_sender_id(&self, sender: &Member, plaintext: Vec<u8>) -> Result<DecryptedBytes> {
419        let crypto_provider = self.crypto_provider().await?;
420        let sender_credential_with_key = CredentialWithKey {
421            credential: sender.credential.clone(),
422            signature_key: sender.signature_key.clone().into(),
423        };
424        let pki_env = crypto_provider.authentication_service().pki_env().await;
425        let identity = sender_credential_with_key
426            .extract_identity(self.cipher_suite().signature_algorithm(), pki_env.as_deref())
427            .await
428            .map_err(RecursiveError::mls_credential("extracting identity"))?;
429        let sender_client_id = sender
430            .credential
431            .identity()
432            .try_into()
433            .map_err(RecursiveError::mls_client("client id from credential"))?;
434        let decrypted_bytes = DecryptedBytes {
435            plaintext,
436            sender_client_id,
437            identity,
438        };
439        Ok(decrypted_bytes)
440    }

443/// Used for plaintext padding before encrypting tnt messages.
444fn tls_serialize_padded(message: &[u8]) -> Result<Vec<u8>, tls_codec::Error> {
445    // VLBytes includes a length prefix, so whoever decrypts the ciphertext will know the payload length.
446    let mut payload = VLBytes::from(message).tls_serialize_detached()?;
447    payload.resize(payload.len().next_multiple_of(PADDING_SIZE), 0);
448    Ok(payload)
449}

461    async fn unrelated_mutation_after_reload_preserves_tnt_counter(case: TestContext) {
462        let [mut alice, bob] = case.sessions().await;
463
464        let (conversation_id, epoch) = {
465            let conversation = case.create_conversation([&alice, &bob]).await;
466
467            // Encrypt a transient message, bumping the tx counter to 1.
468            let first = conversation
469                .guard()
470                .await
471                .encrypt_transient(b"first".to_vec())
472                .await
473                .unwrap();
474            conversation.guard_of(&bob).await.decrypt_message(&first).await.unwrap();
475            (conversation.id().to_owned(), conversation.guard().await.epoch().await)
476        };
477        alice.commit_transaction().await;
478
479        // load the session and mutate its group by sending a normal message
480        let session = alice.session().await;
481        session.conversation_cache.lock().await.clear();
482        let loaded = Conversation::load(&session, conversation_id.as_ref())
483            .await
484            .unwrap()
485            .unwrap();
486
487        let second = {
488            let loaded = session.conversation_cache.lock().await.insert(loaded);
489            let mut loaded = ConversationMut::new(loaded, alice.transaction.clone());
490            loaded.encrypt_message(b"ordinary MLS message").await.unwrap();
491            assert_eq!(loaded.epoch().await, epoch);
492
493            let counter = alice
494                .database()
495                .get_borrowed::<PersistedMlsGroup>(conversation_id.keystore())
496                .await
497                .unwrap()
498                .unwrap()
499                .tnt_tx_counter;
500            assert_eq!(
501                counter, 1,
502                "persisting unrelated group state must preserve the stored TNT counter"
503            );
504
505            loaded.encrypt_transient(b"second".to_vec()).await.unwrap()
506        };
507        alice.commit_transaction().await;
508
509        let counter = alice
510            .database()
511            .get_borrowed::<PersistedMlsGroup>(conversation_id.keystore())
512            .await
513            .unwrap()
514            .unwrap()
515            .tnt_tx_counter;
516        assert_eq!(
517            counter, 2,
518            "the next transient message must advance the persisted counter"
519        );
520
521        let second = bob
522            .transaction
523            .conversation(&conversation_id)
524            .await
525            .unwrap()
526            .decrypt_message(second)
527            .await
528            .unwrap();
529        assert_eq!(second.into_transient().unwrap().plaintext, b"second");
530    }

532    #[apply(all_cred_cipher)]
533    async fn epoch_change_resets_tnt_counter_after_reload(case: TestContext) {
534        let [mut alice, bob] = case.sessions().await;
535        let conversation = case.create_conversation([&alice, &bob]).await;
536        let id = conversation.id().clone();
537        let epoch = conversation.guard().await.epoch().await;

540            let encrypted = conversation
541                .guard()
542                .await
543                .encrypt_transient(message.to_vec())
544                .await
545                .unwrap();
546            conversation
547                .guard_of(&bob)
548                .await
549                .decrypt_message(encrypted)
550                .await
551                .unwrap();
552        }
553        drop(conversation);
554        alice.commit_transaction().await;
555        alice.pretend_crash().await;

559        assert_eq!(conversation.guard().await.epoch().await, epoch + 1);
560        let counter = alice

563            .await
564            .unwrap()
565            .unwrap()
566            .tnt_tx_counter;

577            .unwrap();

589            .unwrap()
590            .unwrap()
591            .tnt_tx_counter;
592        assert_eq!(counter, 1);
593    }
