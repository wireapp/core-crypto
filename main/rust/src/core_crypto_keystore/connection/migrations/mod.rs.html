12#[derive(Default)]
13pub(crate) enum MigrationTarget {
14    /// Perform all migrations
15    #[default]
16    Latest,
17    /// Perform all migrations up to the specified version
18    Version(u16),
19    /// Execute the composite schema, producing the final form of the database without going through each individual
20    /// migration.
21    Composite,
22}
23
24pub(super) fn run_migrations(conn: &mut rusqlite::Connection, target: MigrationTarget) -> CryptoKeystoreResult<()> {
25    conn.create_scalar_function("sha256_blob", 1, FunctionFlags::SQLITE_DETERMINISTIC, |ctx| {
26        let input_blob = ctx.get::<Vec<u8>>(0)?;
27        Ok(crate::sha256(&input_blob))
28    })?;
29
30    let mut runner = migrations::runner();
31    let Some(latest_migration_version) = runner
32        .get_migrations()
33        .iter()
34        .map(|migration| migration.version())
35        .max()
36    else {
37        // No migrations means nothing to do.
38        return Ok(());
39    };
40
41    let target_version = match target {
42        MigrationTarget::Latest => latest_migration_version,
43        MigrationTarget::Version(target_argument) => (latest_migration_version).min(target_argument as i32),
44        MigrationTarget::Composite => {
45            conn.execute_batch(COMPOSITE_SCHEMA)?;
46            conn.pragma_update(None, "user_version", latest_migration_version)?;
47            return Ok(());
48        }
49    };
50
51    // This version is known to have an additional newline in some releases, but the actual migration work is
52    // identical. Ensure Refinery sees the checksum of the embedded migration when it validates the history.
53    const BROKEN_MIGRATION_FILE_VERSION: i32 = 16;
54    let expected_checksum = runner
55        .get_migrations()
56        .iter()
57        .find(|migration| migration.version() == BROKEN_MIGRATION_FILE_VERSION)
58        .expect("V16 is embedded")
59        .checksum()
60        .to_string();
61
62    // Adjust the schema history only if it already exists.
63    let has_refinery_schema_history = conn.query_row(
64        "SELECT EXISTS(SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'refinery_schema_history')",
65        [],
66        |row| row.get::<_, bool>(0),
67    )?;
68    if has_refinery_schema_history {
69        conn.execute(
70            "UPDATE refinery_schema_history SET checksum = ?1 WHERE version = ?2",
71            (expected_checksum, BROKEN_MIGRATION_FILE_VERSION),
72        )?;
73    }
74
75    for version in 1..=target_version {
76        runner = runner.set_target(Target::Version(version));
77        let report = runner.run(conn).map_err(Box::new)?;
78
79        let Some(updated_version) = report.applied_migrations().iter().map(|m| m.version()).max() else {
80            continue;
81        };
82
83        // If the version has been updated by the runner, first run the meta migration, then update the schema
84        // version.
85        run_meta_migration(updated_version, conn)?;
86        conn.pragma_update(None, "user_version", updated_version)?;
87    }
88
89    Ok(())
90}

92/// Add a new match arm here if you want to run a meta migration (i.e., addtional work implemented in rust)
93/// after a regular SQL migration.
94fn run_meta_migration(sql_migration_version: i32, conn: &mut rusqlite::Connection) -> CryptoKeystoreResult<()> {
95    match sql_migration_version {
96        meta_migrations::v16::VERSION => meta_migrations::v16::meta_migration(conn),
97        meta_migrations::v18::VERSION => meta_migrations::v18::meta_migration(conn),
98        meta_migrations::v19::VERSION => meta_migrations::v19::meta_migration(conn),
99        meta_migrations::v28::VERSION => meta_migrations::v28::meta_migration(conn),
100        meta_migrations::v31::VERSION => meta_migrations::v31::meta_migration(conn),
101        meta_migrations::v34::VERSION => meta_migrations::v34::meta_migration(conn),
102        meta_migrations::v37::VERSION => meta_migrations::v37::meta_migration(conn),
103        meta_migrations::v39::VERSION => meta_migrations::v39::meta_migration(conn),
104        _ => Ok(()),
105    }
106}

108/// Migrate a database encrypted with a string key to the new raw-bytes [`DatabaseKey`].
109///
110/// This is intended to be called only once, when migrating from CoreCrypto 5.x to 6.x, before
111/// opening the database via [`super::Database::open`].
112pub async fn migrate_db_key_type_to_bytes(
113    path: &str,
114    old_key: &str,
115    new_key: &DatabaseKey,
116) -> CryptoKeystoreResult<()> {
117    // On WASM the legacy data lives in an IndexedDB database. This rekeys that legacy store in place,
118    // re-encrypting every entity from the old string-derived cipher to the new bytes key. The
119    // subsequent [`super::Database::open`] then copies the data into the unified rusqlite database.
120    #[cfg(target_os = "unknown")]
121    {
122        super::idb_migration::migrate_legacy_idb_key_type_to_bytes(path, old_key, new_key).await
123    }
124
125    // On native platforms we implement the migration directly.
126    #[cfg(not(target_os = "unknown"))]
127    {
128        use crate::CryptoKeystoreError;
129
130        let mut conn = rusqlite::Connection::open(path)?;
131
132        conn.pragma_update(None, "key", old_key)?;
133
134        // ? iOS WAL journaling fix; see details here: https://github.com/sqlcipher/sqlcipher/issues/255
135        #[cfg(target_os = "ios")]
136        super::ios_wal_compat::handle_ios_wal_compat(&conn, path)?;
137
138        /// This is the latest schema version our test db dump is compatible with.
139        const MAX_SUPPORTED_SCHEMA_VERSION: u8 = 15;
140
141        let version = conn.query_row("PRAGMA user_version;", [], |row| row.get::<_, i32>(0))?;
142        if version >= MAX_SUPPORTED_SCHEMA_VERSION as i32 {
143            return Err(CryptoKeystoreError::MigrationFailed(
144                "key type migration from string to bytes can and should only be done once and on database versions
145                    corresponding to a core crypto version <= 9."
146                    .to_string(),
147            ));
148        }
149
150        // Enable WAL journaling mode
151        conn.pragma_update(None, "journal_mode", "wal")?;
152
153        // Disable FOREIGN KEYs - The 2 step blob writing process invalidates foreign key checks unfortunately
154        conn.pragma_update(None, "foreign_keys", "OFF")?;
155
156        // Now update the database to the latest compatible schema version. The other, following migrations
157        // will be run when the database is opened regularly.
158        run_migrations(&mut conn, MigrationTarget::Version(MAX_SUPPORTED_SCHEMA_VERSION as u16))?;
159
160        // Rekey the database.
161        super::encryption::rekey(&mut conn, new_key)
162    }
163}

452    /// A pending message which predates V31 is reachable by primary key once V31 has run.
453    ///
454    /// V31 backfills `hash_sha256` in SQL while [`MlsPendingMessage`] computes it in Rust. Those two
455    /// have to agree on both the hash and its storage encoding, or migrated rows become unreachable
456    /// by every primary-key operation: `get`, `delete`, and the transaction cache alike.
457    #[test]
458    fn v31_backfills_primary_keys_which_the_entity_can_use() {
459        const CONVERSATION_ID: &[u8] = b"a conversation predating the migration";
460        const MESSAGE: &[u8] = b"a message buffered before the migration ran";
461
462        let db_file = NamedTempFile::new().unwrap();
463        let path = db_file
464            .path()
465            .to_str()
466            .expect("tmpfile path is representable in unicode");
467        let key = DatabaseKey::generate();
468
469        smol::block_on(async {
470            // write a pending message the way the schema looked before V31
471            {
472                let db = Database::open_at_schema_version(path, &key, MigrationTarget::Version(30))
473                    .await
474                    .unwrap();
475                let conn = db.conn().await;
476                // The pre-V31 table still carries V7's foreign key onto `mls_pending_groups`, so the
477                // parent row has to exist before the message can be inserted here. V31 drops that
478                // constraint; see `v31_migrates_a_pending_message_whose_conversation_is_not_a_pending_group`
479                // for the case which the constraint used to make impossible.
480                conn.execute(
481                    "INSERT INTO mls_pending_groups (id, state, cfg) VALUES (?, ?, ?)",
482                    (CONVERSATION_ID, b"group state", b"custom configuration"),
483                )
484                .expect("inserting the pending group the message refers to");
485                conn.execute(
486                    "INSERT INTO mls_pending_messages (id, message) VALUES (?, ?)",
487                    (CONVERSATION_ID, MESSAGE),
488                )
489                .expect("inserting a pre-migration pending message");
490            }
491
492            // reopening runs V31, which must backfill a primary key the entity agrees with
493            let db = Database::open(path, &key).await.unwrap();
494            let conn = db.conn().await;
495
496            let migrated = MlsPendingMessage::load_all(&conn).expect("loading migrated pending messages");
497            assert_eq!(migrated.len(), 1, "the migration must not lose the pending message");
498            let migrated = migrated.into_iter().next().unwrap();
499            assert_eq!(migrated.conversation_id.bytes(), CONVERSATION_ID);
500            assert_eq!(migrated.message, MESSAGE);
501
502            // spelled out rather than taken from `primary_key()`, so that this pins the derivation
503            // for both the entity and the migration rather than just mirroring one of them
504            let expected_key = Sha256Hash::hash_from_many([
505                (CONVERSATION_ID.len() as u64).to_be_bytes().as_slice(),
506                CONVERSATION_ID,
507                MESSAGE,
508            ]);
509            assert_eq!(
510                migrated.primary_key(),
511                expected_key,
512                "the entity must derive its primary key from the same inputs the migration hashed"
513            );
514
515            assert!(
516                MlsPendingMessage::get(&conn, &expected_key)
517                    .expect("getting the migrated pending message by primary key")
518                    .is_some(),
519                "the primary key V31 wrote must be byte-identical to the one the entity binds when querying"
520            );
521        });
522    }

524    /// V31 migrates a pending message whose conversation has no `mls_pending_groups` row.
525    ///
526    /// This is the state V31's backfill would have rejected had it recreated V7's foreign key: an
527    /// `INSERT ... SELECT` into a constrained table fails on the first orphan row, and a migration which
528    /// fails leaves the database unopenable. Such rows are reachable in practice, because
529    /// [`migrate_db_key_type_to_bytes`] runs the early migrations with `foreign_keys` off and the legacy
530    /// IndexedDB import writes buffered messages before the pending groups they refer to.
531    ///
532    /// The seed plants one the same way, by turning foreign keys off for the insert. There is no other way
533    /// to create a row which the pre-V31 schema forbids, which is the point: what the schema forbade is
534    /// exactly what the code above it was trying to store.
535    #[test]
536    fn v31_migrates_a_pending_message_whose_conversation_is_not_a_pending_group() {
537        const CONVERSATION_ID: &[u8] = b"an established conversation, absent from mls_pending_groups";
538        const MESSAGE: &[u8] = b"a message from epoch n + 1, buffered before the migration ran";
539
540        let (db_file, key) = temp_db();
541        let path = db_file.path().to_str().unwrap();
542
543        smol::block_on(async {
544            let db = seed_then_migrate(path, &key, 30, |conn| {
545                conn.pragma_update(None, "foreign_keys", "OFF")
546                    .expect("disabling foreign keys so that the orphan row can be planted at all");
547                conn.execute(
548                    "INSERT INTO mls_pending_messages (id, message) VALUES (?, ?)",
549                    (CONVERSATION_ID, MESSAGE),
550                )
551                .expect("inserting a pending message whose conversation is not a pending group");
552            })
553            .await;
554            let conn = db.conn().await;
555
556            let migrated = MlsPendingMessage::load_all(&conn).expect("loading migrated pending messages");
557            assert_eq!(
558                migrated.len(),
559                1,
560                "V31 must carry an orphan pending message across, not drop it or fail the migration"
561            );
562            let migrated = migrated.into_iter().next().unwrap();
563            assert_eq!(migrated.conversation_id.bytes(), CONVERSATION_ID);
564            assert_eq!(migrated.message, MESSAGE);
565        });
566    }

568    // The tests below all cover the same defect and its repair, one entity at a time.
569    //
570    // `sha256_blob` is named for its input, not its output: it returns `crate::sha256`, which is a
571    // 64-character hex string. Every migration which backfills a sha256 key column therefore writes
572    // TEXT, while `impl ToSql for Sha256Hash` binds the 32 raw digest bytes. SQLite never compares a
573    // blob equal to text, so a row such a migration wrote cannot be found by the code which owns it:
574    // the data survives `load_all` but is invisible to `get` and `delete`, and so to the transaction
575    // cache as well.
576    //
577    // V31 avoids this in its own backfill by wrapping it in `unhex(...)`. V12 and V20 shipped
578    // without it, so the V31 meta migration rewrites what they left behind.

726    /// A hex key which collides with a key already stored as bytes is dropped, not rewritten.
727    ///
728    /// This is the state any long-lived database is actually in: V12 left a hex-keyed row behind,
729    /// and because that row was invisible to `get`, the code went on to save the same key again —
730    /// correctly, as bytes. The byte-keyed row is therefore the live one, and rewriting the hex row
731    /// would both resurrect stale data and violate the column's uniqueness constraint.
732    #[test]
733    fn v31_meta_migration_drops_hex_keys_superseded_by_byte_keys() {
734        const PK: &[u8] = b"a public key saved once before V12 and once after";
735        const STALE_SK: &[u8] = b"the secret key stored alongside the hex-keyed row";
736        const LIVE_SK: &[u8] = b"the secret key stored alongside the byte-keyed row";
737
738        let (db_file, key) = temp_db();
739        let path = db_file.path().to_str().unwrap();
740
741        smol::block_on(async {
742            let hash = Sha256Hash::hash_from(PK);
743
744            // Seed after V12 has already run, so both encodings can be planted side by side. They
745            // coexist because the uniqueness constraint sees a text and a blob as distinct values.
746            let db = seed_then_migrate(path, &key, 30, |conn| {
747                conn.execute(
748                    "INSERT INTO mls_encryption_keypairs (pk_sha256, pk, sk) VALUES (?, ?, ?)",
749                    (hash.to_string(), PK, STALE_SK),
750                )
751                .expect("inserting the hex-keyed row V12 would have left behind");
752                conn.execute(
753                    "INSERT INTO mls_encryption_keypairs (pk_sha256, pk, sk) VALUES (?, ?, ?)",
754                    (hash, PK, LIVE_SK),
755                )
756                .expect("inserting the byte-keyed row the entity would have written since");
757            })
758            .await;
759            let conn = db.conn().await;
760
761            assert_eq!(
762                StoredEncryptionKeyPair::count(&conn).unwrap(),
763                1,
764                "the superseded hex-keyed row must be dropped rather than rewritten"
765            );
766
767            let survivor = StoredEncryptionKeyPair::get_borrowed(&conn, PK)
768                .unwrap()
769                .expect("the byte-keyed row must survive and stay gettable");
770            assert_eq!(
771                survivor.sk, LIVE_SK,
772                "the surviving row must be the byte-keyed one, not the stale hex-keyed one"
773            );
774        });
775    }

836    /// V39 unifies `mls_groups`/`mls_pending_groups`, but can't populate `epoch`, `ciphersuite`,
837    /// `own_leaf_index`, `credential_id`, or `credential_type` in SQL: those values only exist
838    /// inside the postcard-serialized `MlsGroup` in `state`. The v39 meta migration backfills them
839    /// by deserializing `state` in Rust; this test pins that backfill against an independent parse
840    /// of the same `state` blob, using the one real group the bundled legacy dump carries.
841    #[test]
842    fn v39_meta_migration_backfills_group_columns_from_state() {
843        let mut db_file = NamedTempFile::new().unwrap();
844        db_file.write_all(DB).unwrap();
845        let path = db_file
846            .path()
847            .to_str()
848            .expect("tmpfile path is representable in unicode");
849
850        let new_key = DatabaseKey::generate();
851        smol::block_on(migrate_db_key_type_to_bytes(path, OLD_KEY, &new_key)).unwrap();
852
853        smol::block_on(async {
854            let db = Database::open(path, &new_key).await.unwrap();
855            let conn = db.conn().await;
856
857            struct BackfilledGroupRow {
858                state: Vec<u8>,
859                epoch: Option<i64>,
860                ciphersuite: Option<u16>,
861                own_leaf_index: Option<u32>,
862                credential_id: Option<Vec<u8>>,
863                credential_type: Option<u16>,
864                is_pending: bool,
865            }
866
867            let BackfilledGroupRow {
868                state,
869                epoch,
870                ciphersuite,
871                own_leaf_index,
872                credential_id,
873                credential_type,
874                is_pending,
875            } = conn
876                .query_row(
877                    "SELECT state, epoch, ciphersuite, own_leaf_index, credential_id, credential_type, is_pending FROM mls_groups",
878                    [],
879                    |row| {
880                        Ok(BackfilledGroupRow {
881                            state: row.get(0)?,
882                            epoch: row.get(1)?,
883                            ciphersuite: row.get(2)?,
884                            own_leaf_index: row.get(3)?,
885                            credential_id: row.get(4)?,
886                            credential_type: row.get(5)?,
887                            is_pending: row.get(6)?,
888                        })
889                    },
890                )
891                .expect("the bundled database carries exactly one real mls group");
892
893            assert!(
894                !is_pending,
895                "a row copied from mls_groups, not mls_pending_groups, must not be marked pending"
896            );
897
898            // parsed independently of the meta migration's own implementation, so this pins the
899            // backfill against the state blob rather than against itself
900            let group =
901                crate::deser::<openmls::group::MlsGroup>(&state).expect("the bundled group's state must deserialize");
902
903            assert_eq!(epoch, Some(group.epoch().as_u64() as i64));
904            assert_eq!(ciphersuite, Some(group.ciphersuite() as u16));
905            assert_eq!(own_leaf_index, Some(group.own_leaf_index().u32()));
906
907            let own_leaf = group.own_leaf().expect("the bundled group must have an own leaf node");
908            let expected_credential_id = Sha256Hash::hash_from(own_leaf.signature_key().as_slice());
909            let expected_credential_type: u16 = own_leaf.credential().credential_type().into();
910            assert_eq!(credential_id, Some(expected_credential_id.as_ref().to_vec()));
911            assert_eq!(credential_type, Some(expected_credential_type));
912
913            // and the backfilled (credential_id, credential_type) pair must actually resolve to a
914            // stored credential, since that's the whole point of the columns
915            let credential_exists: bool = conn
916                .query_row(
917                    "SELECT EXISTS(SELECT 1 FROM mls_credentials WHERE public_key_sha256 = ? AND credential_type = ?)",
918                    rusqlite::params![expected_credential_id, expected_credential_type],
919                    |row| row.get(0),
920                )
921                .expect("checking for the credential");
922            assert!(
923                credential_exists,
924                "(credential_id, credential_type) must reference a credential which is actually in mls_credentials"
925            );
926        });
