1use std::cmp::Ordering;
2
3use const_format::formatcp;
4use log::info;
5use rusqlite::named_params;
6
7use crate::{
8    CryptoKeystoreError, CryptoKeystoreResult, Sha256Hash, migrations::StoredCredentialV36, traits::Entity as _,
9};
10
11pub(crate) const VERSION: i32 = 19;
12
13pub(crate) fn meta_migration(tx: &rusqlite::Transaction<'_>) -> CryptoKeystoreResult<()> {
14    let mut credential_stmt = tx.prepare(formatcp!(
15        "SELECT ciphersuite, public_key, credential, unixepoch(created_at) AS created_at, rowid FROM {table}",
16        table = StoredCredentialV36::TABLE_NAME,
17    ))?;
18
19    let mut credentials = credential_stmt
20        .query_map([], |row| {
21            let rowid = row.get::<_, i64>("rowid")?;
22            let credential = StoredCredentialV36 {
23                ciphersuite: row.get("ciphersuite")?,
24                public_key: row.get("public_key")?,
25                credential: row.get("credential")?,
26                created_at: row.get("created_at")?,
27                session_id: Vec::new(),  // not relevant for this application
28                private_key: Vec::new(), // not relevant for this application
29            };
30            let ctype = credential.credential_type().ok();
31            Ok((rowid, ctype, credential))
32        })?
33        .collect::<Result<Vec<_>, _>>()
34        .map_err(|err| {
35            CryptoKeystoreError::MigrationFailed(format!("could not load credential for v19 meta-migration: {err}"))
36        })?;
37
38    // group by public key
39    credentials.sort_by_cached_key(|(_, _, credential)| credential.public_key.clone());
40    // consider groups which have the same public key
41    for group in credentials.chunk_by_mut(|(_, _, a), (_, _, b)| a.public_key == b.public_key) {
42        // in this sort higher is better
43        group.sort_unstable_by(|(a_rowid, a_type, a_cred), (b_rowid, b_type, b_cred)| {
44            // first sort by credential type: basic is 1 and x509 is 2;
45            // we always want x509 if available over basic, but if one was unparseable,
46            // keep the one which did parse
47            let type_ordering = match (a_type, b_type) {
48                (Some(a), Some(b)) => a.cmp(b),
49                (Some(_), None) => Ordering::Greater,
50                (None, Some(_)) => Ordering::Less,
51                (None, None) => Ordering::Equal,
52            };
53            // after credential type we take the most recent credential
54            type_ordering
55                .then(a_cred.created_at.cmp(&b_cred.created_at))
56                // after that we take the highest rowid (the last one created; guaranteed nonequal)
57                .then(a_rowid.cmp(b_rowid))
58        });
59
60        // we retain the "highest" member of this group, which is the best
61        let (rowid, _, credential) = group.last().expect("group is guaranteed to have at least one member");
62        let mut stmt = tx.prepare_cached(formatcp!(
63            "DELETE FROM {table} WHERE public_key = :public_key AND rowid != :rowid",
64            table = StoredCredentialV36::TABLE_NAME
65        ))?;
66        let affected_rows = stmt.execute(named_params! {":public_key": &credential.public_key, ":rowid": rowid})?;
67        if affected_rows > 0 {
68            let public_key_hash = Sha256Hash::hash_from(&credential.public_key);
69            info!(affected_rows, public_key_hash:%; "deleted extraneous credentials as part of v19 meta-migration");
70        }
71    }
72
73    Ok(())
74}
