6/// Extend an [`Entity`] with db-mutating operations which can be performed when provided with a transaction.
7pub trait EntityDatabaseMutation: Entity {
8    /// Save this entity to the database in the context of this transaction.
9    ///
10    /// The transaction is required instead of a simple connection to give us type-level assurances
11    /// that we only mutate the database within the context of a transaction; every mutation can
12    /// be rolled back if necessary.
13    ///
14    /// Unlike in earlier iterations of this trait, saving makes no automatic adjustment to the
15    /// entity being saved. Entities requiring a `pre_save` method must implement this manually.
16    fn save<'a, Tx>(&self, tx: &'a Tx) -> CryptoKeystoreResult<()>
17    where
18        &'a Tx: Into<Transactionlike<'a>>;
19
20    /// Use the transaction's inteface to delete this entity from the database.
21    ///
22    /// Returns `true` if at least one entity was deleted, or `false` if the id was not found in the database.
23    ///
24    /// For entites whose primary key has a distinct borrowed type, it is best to implement this as a direct
25    /// passthrough:
26    ///
27    /// ```rust,ignore
28    /// fn delete<'a, Tx>(tx: &'a Tx, id: &Self::PrimaryKey) -> CryptoKeystoreResult<bool>
29    /// where
30    ///     &'a Tx: Into<Transactionlike<'a>>,
31    /// {
32    ///     <Self as EntityDeleteBorrowed>::delete_borrowed(tx, id)
33    /// }
34    /// ```
35    fn delete<'a, Tx>(tx: &'a Tx, id: &Self::PrimaryKey) -> CryptoKeystoreResult<bool>
36    where
37        Transactionlike<'a>: From<&'a Tx>;
38}

40/// Extend an [`Entity`] with db-mutating operations which can be performed when provided with a transaction.
41pub trait EntityDeleteBorrowed: EntityDatabaseMutation + BorrowPrimaryKey {
42    /// Delete an entity by a borrowed form of its primary key.
43    fn delete_borrowed<'a, Tx>(tx: &'a Tx, id: Self::BorrowedPrimaryKey<'_>) -> CryptoKeystoreResult<bool>
44    where
45        &'a Tx: Into<Transactionlike<'a>>;
46}
