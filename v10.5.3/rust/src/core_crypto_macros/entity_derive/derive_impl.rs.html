7impl quote::ToTokens for Entity {
8    fn to_tokens(&self, tokens: &mut TokenStream) {
9        tokens.extend(self.impl_primary_key());
10        tokens.extend(self.impl_entity());
11        tokens.extend(self.impl_entity_get_borrowed());
12        tokens.extend(self.impl_entity_database_mutation());
13        tokens.extend(self.impl_entity_delete_borrowed());
14    }
15}

17impl Entity {
18    /// `impl PrimaryKey for MyEntity` and `impl BorrowPrimaryKey for MyEntity`
19    fn impl_primary_key(&self) -> TokenStream {
20        let Self {
21            struct_name, id_column, ..
22        } = self;
23
24        let primary_key = id_column.column_type.owned();
25        let borrowed_primary_key = id_column.column_type.borrowed();
26        let pk_field_name = &id_column.field_name;
27
28        quote! {
29            impl crate::traits::PrimaryKey for #struct_name {
30                type PrimaryKey = #primary_key;
31
32                fn primary_key(&self) -> Self::PrimaryKey {
33                    self.#pk_field_name.clone()
34                }
35            }
36
37            impl crate::traits::BorrowPrimaryKey for #struct_name {
38                type BorrowedPrimaryKey<'a> = &'a #borrowed_primary_key;
39
40                fn borrow_primary_key(&self) -> Self::BorrowedPrimaryKey<'_> {
41                    &self.#pk_field_name
42                }
43            }
44        }
45    }
46
47    /// Returns `(sql_statement, fields_params, sql_map_err)` for the generated `save` impl.
48    fn sql_insert_parts(&self) -> (String, TokenStream, Option<TokenStream>) {
49        let Self {
50            upsert,
51            table_name,
52            id_column,
53            other_columns,
54            ..
55        } = self;
56
57        let sql_column_names = std::iter::once(id_column.sql_name())
58            .chain(other_columns.iter().map(|column| column.sql_name()))
59            .join(", ");
60        let sql_field_placeholders = std::iter::repeat_n("?", other_columns.len() + 1).join(", ");
61
62        let on_conflict = if !upsert {
63            String::new()
64        } else {
65            let id_name = id_column.sql_name();
66            let assignments = other_columns
67                .iter()
68                .map(|column| {
69                    let name = column.sql_name();
70                    format!("{name} = excluded.{name}")
71                })
72                .join(", ");
73            if assignments.is_empty() {
74                format!(" ON CONFLICT ({id_name}) DO NOTHING")
75            } else {
76                format!(" ON CONFLICT ({id_name}) DO UPDATE SET {assignments}")
77            }
78        };
79
80        let sql_statement =
81            format!("INSERT INTO {table_name} ({sql_column_names}) VALUES ({sql_field_placeholders}){on_conflict}");
82        let fields = std::iter::once(id_column.store_expression())
83            .chain(other_columns.iter().map(|column| column.store_expression()))
84            .map(|tokens| quote!(#tokens,))
85            .collect::<TokenStream>();
86
87        // `TABLE_NAME` is qualified rather than written as `Self::TABLE_NAME` because entity
88        // modules are not obliged to import `Entity`, and an associated const of a trait which is
89        // not in scope cannot be named through `Self`.
90        let sql_map_err = (!upsert).then_some(quote! {
91            .map_err(crate::CryptoKeystoreError::map_already_exists(<Self as crate::traits::Entity>::TABLE_NAME))
92        });
93
94        (sql_statement, fields, sql_map_err)
95    }

97    /// `impl Entity for MyEntity`
98    fn impl_entity(&self) -> TokenStream {
99        let Self {
100            table_name,
101            struct_name,
102            id_column,
103            other_columns,
104            ..
105        } = self;
106
107        let field_assignments = std::iter::once(id_column.field_assignment())
108            .chain(other_columns.iter().map(|column| column.field_assignment()));
109
110        quote! {
111            impl crate::traits::Entity for #struct_name {
112                const TABLE_NAME: &'static str = #table_name;
113
114                fn get(conn: &rusqlite::Connection, key: &Self::PrimaryKey) -> crate::CryptoKeystoreResult<Option<Self>> {
115                    <Self as crate::traits::EntityGetBorrowed>::get_borrowed(conn, key)
116                }
117
118                fn count(conn: &rusqlite::Connection) -> crate::CryptoKeystoreResult<u32> {
119                    crate::entities::helpers::count_helper::<Self>(conn)
120                }
121
122                fn load_all(conn: &rusqlite::Connection) -> crate::CryptoKeystoreResult<Vec<Self>> {
123                    crate::entities::helpers::load_all_helper::<Self, _>(conn, |row| {
124                        Ok(Self {
125                            #( #field_assignments, )*
126                        })
127                    })
128                }
129            }
130        }
131    }
132
133    /// `impl EntityGetBorrowed for MyEntity`
134    fn impl_entity_get_borrowed(&self) -> TokenStream {
135        let Self {
136            struct_name,
137            id_column,
138            other_columns,
139            ..
140        } = self;
141
142        let pk_column_name = id_column
143            .column_name
144            .clone()
145            .unwrap_or_else(|| id_column.field_name.to_string());
146
147        let field_assignments = std::iter::once(id_column.field_assignment())
148            .chain(other_columns.iter().map(|column| column.field_assignment()));
149
150        quote! {
151            impl crate::traits::EntityGetBorrowed for #struct_name {
152                fn get_borrowed(conn: &rusqlite::Connection, key: Self::BorrowedPrimaryKey<'_>)
153                    -> crate::CryptoKeystoreResult<Option<Self>>
154                {
155                    // Bind the borrowed key directly, exactly as the insert generated by
156                    // `store_expression` binds the id field: `&str` binds as `TEXT` and `&[u8]` as
157                    // `BLOB`, so the lookup and the insert cannot disagree about storage class.
158                    // Converting the key here is what once made a string key bind as a `BLOB`,
159                    // which silently matched zero rows, since SQLite never compares a `BLOB` equal
160                    // to a `TEXT` value and column affinity does not coerce a `BLOB` argument.
161                    crate::entities::helpers::get_helper::<Self, _>(conn, #pk_column_name, key, |row| {
162                        Ok(Self {
163                            #( #field_assignments, )*
164                        })
165                    })
166                }
167            }
168        }
169    }
170
171    /// `impl EntityDatabaseMutation for MyEntity`
172    fn impl_entity_database_mutation(&self) -> TokenStream {
173        let Self { struct_name, .. } = self;
174
175        let (sql_statement, fields, sql_map_err) = self.sql_insert_parts();
176
177        quote! {
178            impl crate::traits::EntityDatabaseMutation for #struct_name {
179                fn save<'a, Tx>(&self, tx: &'a Tx) -> crate::CryptoKeystoreResult<()>
180                where
181                    &'a Tx: Into<crate::Transactionlike<'a>>
182                {
183                    let conn = tx.into().conn()?;
184                    let mut stmt = conn.prepare_cached(#sql_statement)?;
185                    stmt.execute(rusqlite::params![#fields])#sql_map_err?;
186                    Ok(())
187                }
188
189                fn delete<'a, Tx>(tx: &'a Tx, id: &Self::PrimaryKey) -> crate::CryptoKeystoreResult<bool>
190                where
191                    &'a Tx: Into<crate::Transactionlike<'a>>
192                {
193                    <Self as crate::traits::EntityDeleteBorrowed>::delete_borrowed(tx, id)
194                }
195            }
196        }
197    }
198
199    /// `impl EntityDeleteBorrowed for MyEntity`
200    fn impl_entity_delete_borrowed(&self) -> TokenStream {
201        let Self {
202            struct_name, id_column, ..
203        } = self;
204
205        let id_column_name = id_column.sql_name();
206
207        quote! {
208            impl crate::traits::EntityDeleteBorrowed for #struct_name {
209                fn delete_borrowed<'a, Tx>(
210                    tx: &'a Tx,
211                    id: <Self as crate::traits::BorrowPrimaryKey>::BorrowedPrimaryKey<'_>,
212                ) -> crate::CryptoKeystoreResult<bool>
213                where
214                    &'a Tx: Into<crate::Transactionlike<'a>>
215                {
216                    // Bind the borrowed key directly, for the same reason as in `get_borrowed`.
217                    crate::entities::helpers::delete_helper::<Self, _>(tx, #id_column_name, id)
218                }
219            }
220        }
221    }
