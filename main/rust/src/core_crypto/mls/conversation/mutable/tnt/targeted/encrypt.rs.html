16/// The policy to encrypt the targeted message with.
17#[derive(Debug, Clone, Copy, PartialEq, Eq, TlsSize, TlsSerialize)]
18#[repr(u8)]
19pub enum TargetedMessagePolicy {
20    /// Won't be persisted and will only visible to currently online members who immediately process it.
21    Transient,
22    /// May be persisted and buffered, will also be delivered and processed by currently offline members.
23    Persisted,
24}
25
26impl ConversationMut {
27    /// Targeted messages are messages distributed to individual clients. They don't mutate regular group state (i.e, as
28    /// defined by RFC 9420). That is because many or most clients in a group will never receive those messages, and
29    /// will not have the appropriate cryptographic state to decrypt the messages.
30    ///
31    /// Any feature using a targeted, transient or transient targeted message MUST specify why the lower security
32    /// guarantees (compared to MLS application messages) are acceptable and/or how they are mitigated.
33    pub async fn encrypt_targeted(
34        &mut self,
35        recipient: impl Borrow<ClientIdRef>,
36        policy: TargetedMessagePolicy,
37        message: Vec<u8>,
38    ) -> Result<Vec<u8>> {
39        let cipher_suite = self.cipher_suite();
40        let crypto_provider = self
41            .tx_context
42            .crypto_provider()
43            .await
44            .map_err(RecursiveError::context("obtaining crypto provider"))?;
45        let database = self.database().await?;
46
47        let targeted = self
48            .mutate_group(async |_, group_state, _| {
49                let recipient = group_state
50                    .mls_group()
51                    .members()
52                    .find(|member| ClientIdRef::new(member.credential.identity()) == recipient.borrow())
53                    .ok_or_else(|| Error::MemberNotFound(recipient.borrow().to_owned()))?;
54                Self::create_targeted_message(
55                    &database,
56                    policy,
57                    group_state,
58                    &cipher_suite,
59                    &crypto_provider,
60                    &recipient,
61                    &message,
62                )
63                .await
64            })
65            .await?;
66
67        let tbs = match policy {
68            TargetedMessagePolicy::Transient => TntMessageTBS::new_transient_targeted(targeted),
69            TargetedMessagePolicy::Persisted => TntMessageTBS::new_targeted(targeted),
70        };
71
72        let credential = self.find_current_credential().await?;
73        let signature_key = credential.signature_key();
74        let signed_message: TntMessage = tbs
75            .sign(signature_key)
76            .map_err(OpenMlsError::wrap("signing TntMessageTBS"))?;
77
78        signed_message
79            .tls_serialize_detached()
80            .map_err(TlsCodecError::serialize("TntMessage: targeted"))
81            .map_err(Into::into)
82    }
83
84    async fn create_targeted_message(
85        database: &Database,
86        policy: TargetedMessagePolicy,
87        group_state: &mut MlsGroupState,
88        cipher_suite: &Ciphersuite,
89        crypto_provider: &CryptoProvider,
90        recipient: &Member,
91        message: &[u8],
92    ) -> Result<TargetedMessage, Error> {
93        let nonce = group_state
94            .obtain_targeted_message_tx_counter(recipient.index, database)
95            .await?;
96        let mls_group = group_state.mls_group();
97        let aad = nonce
98            .tls_serialize_detached()
99            .map_err(TlsCodecError::serialize("TntMessageCounter"))?;
100        let context = TargetedMessageContext::new(
101            policy,
102            mls_group.export_group_context(),
103            mls_group.own_leaf_index(),
104            recipient.index,
105        );
106        let info = context
107            .tls_serialize_detached()
108            .map_err(TlsCodecError::serialize("TargetedMessageContext"))?;
109
110        // We can use an empty context because we're using a unique label.
111        let psk = mls_group
112            .export_secret(
113                crypto_provider,
114                TargetedMessage::PSK_LABEL,
115                &[],
116                cipher_suite.hash_length(),
117            )
118            .map_err(OpenMlsError::wrap("exporting targeted message psk"))?;
119        let psk_id = PskId::new(mls_group.group_id().clone(), mls_group.epoch());
120        let psk_id = psk_id
121            .tls_serialize_detached()
122            .map_err(TlsCodecError::serialize("PskId"))?;
123        let message = tls_serialize_padded(message).map_err(TlsCodecError::serialize("TargetedMessageContent"))?;
124        let payload = crypto_provider
125            .hpke_seal_psk(
126                cipher_suite.hpke_config(),
127                &recipient.encryption_key,
128                &info,
129                &aad,
130                &psk,
131                &psk_id,
132                &message,
133            )
134            .map_err(OpenMlsError::wrap("encrypting targeted message"))?;
135
136        let sender = mls_group.own_leaf_index();
137        let group_id = mls_group.group_id();
138        let epoch = mls_group.epoch();
139
140        let targeted = TargetedMessage::new(nonce, sender, recipient.index, epoch, group_id.clone(), payload);
141        Ok(targeted)
142    }
143}

145fn tls_serialize_padded(message: &[u8]) -> Result<Vec<u8>, tls_codec::Error> {
146    // VLBytes includes a length prefix, so whoever decrypts the ciphertext will know the payload length.
147    let mut payload = VLBytes::from(message).tls_serialize_detached()?;
148    payload.resize(payload.len().next_multiple_of(TargetedMessage::PADDING_SIZE), 0);
149    Ok(payload)
150}
