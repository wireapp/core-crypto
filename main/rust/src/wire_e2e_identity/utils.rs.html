18pub fn generate_key(sign_alg: JwsAlgorithm) -> E2eIdentityResult<Pem> {
19    let pem = match sign_alg {
20        JwsAlgorithm::P256 => ES256KeyPair::generate().to_pem()?,
21        JwsAlgorithm::P384 => ES384KeyPair::generate().to_pem()?,
22        JwsAlgorithm::P521 => ES512KeyPair::generate().to_pem()?,
23        JwsAlgorithm::Ed25519 => Ed25519KeyPair::generate().to_pem(),
24    };
25    Ok(pem.into())
26}

28pub fn pem_from_bytes(bytes: &[u8], sign_alg: JwsAlgorithm) -> E2eIdentityResult<Pem> {
29    let pem = match sign_alg {
30        JwsAlgorithm::P256 => ES256KeyPair::from_bytes(bytes)?.to_pem()?,
31        JwsAlgorithm::P384 => ES384KeyPair::from_bytes(bytes)?.to_pem()?,
32        JwsAlgorithm::P521 => ES512KeyPair::from_bytes(bytes)?.to_pem()?,
33        JwsAlgorithm::Ed25519 => Ed25519KeyPair::from_bytes(bytes)?.to_pem(),
34    };
35    Ok(pem.into())
36}

38pub(crate) fn public_jwk_from_pem_keypair(alg: JwsAlgorithm, keypair: &Pem) -> E2eIdentityResult<Jwk> {
39    let jwk = match alg {
40        JwsAlgorithm::P256 => ES256KeyPair::from_pem(keypair)?.public_key().try_into_jwk()?,
41        JwsAlgorithm::P384 => ES384KeyPair::from_pem(keypair)?.public_key().try_into_jwk()?,
42        JwsAlgorithm::P521 => ES512KeyPair::from_pem(keypair)?.public_key().try_into_jwk()?,
43        JwsAlgorithm::Ed25519 => Ed25519KeyPair::from_pem(keypair)?.public_key().try_into_jwk()?,
44    };
45    Ok(jwk)
46}

48pub(crate) fn public_key_bytes(alg: JwsAlgorithm, keypair: &Pem) -> E2eIdentityResult<Vec<u8>> {
49    let bytes = match alg {
50        JwsAlgorithm::P256 => ES256KeyPair::from_pem(keypair)?
51            .public_key()
52            .public_key()
53            .to_bytes_uncompressed(),
54        JwsAlgorithm::P384 => ES384KeyPair::from_pem(keypair)?
55            .public_key()
56            .public_key()
57            .to_bytes_uncompressed(),
58        JwsAlgorithm::P521 => ES512KeyPair::from_pem(keypair)?
59            .public_key()
60            .public_key()
61            .to_bytes_uncompressed(),
62        JwsAlgorithm::Ed25519 => Ed25519KeyPair::from_pem(keypair)?.public_key().to_bytes(),
63    };
64    Ok(bytes)
65}

67pub(crate) fn jws_alg_to_x509_identifier(alg: JwsAlgorithm) -> AlgorithmIdentifierOwned {
68    match alg {
69        JwsAlgorithm::Ed25519 => AlgorithmIdentifierOwned {
70            oid: const_oid::db::rfc8410::ID_ED_25519,
71            parameters: None,
72        },
73        JwsAlgorithm::P256 => AlgorithmIdentifierOwned {
74            oid: const_oid::db::rfc5912::ID_EC_PUBLIC_KEY,
75            parameters: Some(const_oid::db::rfc5912::SECP_256_R_1.into()),
76        },
77        JwsAlgorithm::P384 => AlgorithmIdentifierOwned {
78            oid: const_oid::db::rfc5912::ID_EC_PUBLIC_KEY,
79            parameters: Some(const_oid::db::rfc5912::SECP_384_R_1.into()),
80        },
81        JwsAlgorithm::P521 => AlgorithmIdentifierOwned {
82            oid: const_oid::db::rfc5912::ID_EC_PUBLIC_KEY,
83            parameters: Some(const_oid::db::rfc5912::SECP_521_R_1.into()),
84        },
85    }
86}

88pub(crate) fn extract_ski_aki_from_cert(
89    cert: &x509_cert::Certificate,
90) -> RustyX509CheckResult<(String, Option<String>)> {
91    let cert = certval::PDVCertificate::try_from(cert.clone())?;
92
93    let ski = cert
94        .get_extension(&const_oid::db::rfc5912::ID_CE_SUBJECT_KEY_IDENTIFIER)?
95        .ok_or(RustyX509CheckError::MissingSki)?;
96    let ski = match ski {
97        certval::PDVExtension::SubjectKeyIdentifier(ski) => hex::encode(ski.0.as_bytes()),
98        _ => return Err(RustyX509CheckError::ImplementationError),
99    };
100
101    let aki = cert
102        .get_extension(&const_oid::db::rfc5912::ID_CE_AUTHORITY_KEY_IDENTIFIER)?
103        .and_then(|ext| match ext {
104            certval::PDVExtension::AuthorityKeyIdentifier(AuthorityKeyIdentifier { key_identifier, .. }) => {
105                key_identifier.as_ref()
106            }
107            _ => None,
108        })
109        .map(|ki| hex::encode(ki.as_bytes()));
110
111    Ok((ski, aki))
112}
