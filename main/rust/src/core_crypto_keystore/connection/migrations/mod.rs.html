17pub(super) fn run_migrations(conn: &mut rusqlite::Connection, target: MigrationTarget) -> CryptoKeystoreResult<()> {
18    conn.create_scalar_function("sha256_blob", 1, FunctionFlags::SQLITE_DETERMINISTIC, |ctx| {
19        let input_blob = ctx.get::<Vec<u8>>(0)?;
20        Ok(crate::sha256(&input_blob))
21    })?;
22
23    let mut runner = migrations::runner();
24    let Some(latest_migration_version) = runner
25        .get_migrations()
26        .iter()
27        .map(|migration| migration.version())
28        .max()
29    else {
30        // No migrations means nothing to do.
31        return Ok(());
32    };
33
34    let target_version = match target {
35        MigrationTarget::Latest => latest_migration_version,
36        MigrationTarget::Version(target_argument) => (latest_migration_version).min(target_argument as i32),
37    };
38
39    for version in 1..=target_version {
40        runner = runner.set_target(Target::Version(version));
41        let report = runner.run(conn).map_err(Box::new)?;
42
43        let Some(updated_version) = report.applied_migrations().iter().map(|m| m.version()).max() else {
44            continue;
45        };
46
47        // If the version has been updated by the runner, first run the meta migration, then update the schema
48        // version.
49        run_meta_migration(updated_version, conn)?;
50        conn.pragma_update(None, "user_version", updated_version)?;
51    }
52
53    Ok(())
54}

56/// Add a new match arm here if you want to run a meta migration (i.e., addtional work implemented in rust)
57/// after a regular SQL migration.
58fn run_meta_migration(sql_migration_version: i32, conn: &mut rusqlite::Connection) -> CryptoKeystoreResult<()> {
59    match sql_migration_version {
60        meta_migrations::v16::VERSION => meta_migrations::v16::meta_migration(conn),
61        meta_migrations::v18::VERSION => meta_migrations::v18::meta_migration(conn),
62        meta_migrations::v19::VERSION => meta_migrations::v19::meta_migration(conn),
63        meta_migrations::v28::VERSION => meta_migrations::v28::meta_migration(conn),
64        _ => Ok(()),
65    }
66}

68/// Migrate a database encrypted with a string key to the new raw-bytes [`DatabaseKey`].
69///
70/// This is intended to be called only once, when migrating from CoreCrypto 5.x to 6.x, before
71/// opening the database via [`super::Database::open`].
72pub async fn migrate_db_key_type_to_bytes(
73    path: &str,
74    old_key: &str,
75    new_key: &DatabaseKey,
76) -> CryptoKeystoreResult<()> {
77    // On WASM the legacy data lives in an IndexedDB database. This rekeys that legacy store in place,
78    // re-encrypting every entity from the old string-derived cipher to the new bytes key. The
79    // subsequent [`super::Database::open`] then copies the data into the unified rusqlite database.
80    #[cfg(target_os = "unknown")]
81    {
82        super::idb_migration::migrate_legacy_idb_key_type_to_bytes(path, old_key, new_key).await
83    }
84
85    // On native platforms we implement the migration directly.
86    #[cfg(not(target_os = "unknown"))]
87    {
88        use crate::CryptoKeystoreError;
89
90        let mut conn = rusqlite::Connection::open(path)?;
91
92        conn.pragma_update(None, "key", old_key)?;
93
94        // ? iOS WAL journaling fix; see details here: https://github.com/sqlcipher/sqlcipher/issues/255
95        #[cfg(target_os = "ios")]
96        super::ios_wal_compat::handle_ios_wal_compat(&conn, path)?;
97
98        /// This is the latest schema version our test db dump is compatible with.
99        const MAX_SUPPORTED_SCHEMA_VERSION: u8 = 15;
100
101        let version = conn.query_row("PRAGMA user_version;", [], |row| row.get::<_, i32>(0))?;
102        if version >= MAX_SUPPORTED_SCHEMA_VERSION as i32 {
103            return Err(CryptoKeystoreError::MigrationFailed(
104                "key type migration from string to bytes can and should only be done once and on database versions
105                    corresponding to a core crypto version <= 9."
106                    .to_string(),
107            ));
108        }
109
110        // Enable WAL journaling mode
111        conn.pragma_update(None, "journal_mode", "wal")?;
112
113        // Disable FOREIGN KEYs - The 2 step blob writing process invalidates foreign key checks unfortunately
114        conn.pragma_update(None, "foreign_keys", "OFF")?;
115
116        // Now update the database to the latest compatible schema version. The other, following migrations
117        // will be run when the database is opened regularly.
118        run_migrations(&mut conn, MigrationTarget::Version(MAX_SUPPORTED_SCHEMA_VERSION as u16))?;
119
120        // Rekey the database.
121        super::encryption::rekey(&mut conn, new_key)
122    }
123}

161    #[test]
162    fn deduplicating_credentials() {
163        let mut db_file = NamedTempFile::new().unwrap();
164        db_file.write_all(DB).unwrap();
165        let path = db_file
166            .path()
167            .to_str()
168            .expect("tmpfile path is representable in unicode");
169
170        let new_key = DatabaseKey::generate();
171        smol::block_on(migrate_db_key_type_to_bytes(path, OLD_KEY, &new_key)).unwrap();
172
173        smol::block_on(async {
174            let db = Database::open_at_schema_version(path, &new_key, MigrationTarget::Version(18))
175                .await
176                .unwrap();
177
178            let conn = db.conn().await;
179            let mut stmt = conn
180                .prepare(&format!(
181                    "SELECT
182                        session_id,
183                        credential,
184                        unixepoch(created_at) AS created_at,
185                        ciphersuite,
186                        public_key,
187                        private_key
188                     FROM {credential_table}",
189                    credential_table = "mls_credentials_new",
190                ))
191                .expect("preparing statement");
192
193            let credential = stmt
194                .query_one([], |row| {
195                    Ok(StoredCredential {
196                        session_id: row.get("session_id")?,
197                        credential: row.get("credential")?,
198                        created_at: row.get("created_at")?,
199                        ciphersuite: row.get("ciphersuite")?,
200                        public_key: row.get("public_key")?,
201                        private_key: row.get("private_key")?,
202                    })
203                })
204                .expect("credential from row");
205
206            // Ciphersuites need to be ambiguous w.r.t their signature scheme to be a relevant duplicate
207            conn.execute(
208                "UPDATE mls_credentials_new SET ciphersuite = ?1",
209                [Ciphersuite::MLS_128_DHKEMX25519_AES128GCM_SHA256_Ed25519 as u16],
210            )
211            .expect("updating ciphersuite");
212
213            // Create a duplicate from this credential
214            conn.execute(
215                "INSERT INTO mls_credentials_new (
216                        session_id,
217                        credential,
218                        created_at,
219                        ciphersuite,
220                        public_key,
221                        private_key
222                    )
223                    VALUES (?1, ?2, datetime(?3, 'unixepoch'), ?4, ?5, ?6)",
224                (
225                    credential.session_id.clone(),
226                    credential.credential.clone(),
227                    credential.created_at,
228                    Ciphersuite::MLS_128_DHKEMX25519_CHACHA20POLY1305_SHA256_Ed25519 as u16,
229                    credential.public_key.clone(),
230                    credential.private_key.clone(),
231                ),
232            )
233            .expect("inserting duplicate");
234
235            let count = conn
236                .query_row("SELECT COUNT(*) FROM mls_credentials_new", [], |row| {
237                    row.get::<_, i32>(0)
238                })
239                .unwrap();
240
241            assert_eq!(count, 2);
242
243            drop(stmt);
244            drop(conn);
245            drop(db);
246
247            let db = Database::open(path, &new_key).await.unwrap();
248            let deduplicated_credentials =
249                StoredCredential::load_all(&*db.conn().await).expect("deduplicated credentials");
250
251            let deduplicated_count = deduplicated_credentials.len();
252
253            let deduplicated_credential = deduplicated_credentials.first().expect("first credential");
254
255            assert_eq!(deduplicated_count, 1);
256
257            // In case of equal occurence, the credential with the numerically lower ciphersuite is kept.
258            assert_eq!(
259                deduplicated_credential.ciphersuite,
260                Ciphersuite::MLS_128_DHKEMX25519_AES128GCM_SHA256_Ed25519 as u16
261            );
262        });
263    }
