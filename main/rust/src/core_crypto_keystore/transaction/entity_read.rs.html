1//! These methods allow for read-only operations on the entities in the transaction,
2//! without considering the database itself.
3//!
4//! The `*_in_cache` methods and [`Transaction::get_by_entity_id`] answer purely from the buffered
5//! operations. [`Transaction::find_all`] and [`Transaction::search`] compose those answers with
6//! database records the caller has already loaded, which is as close to the database as anything
7//! here gets.
8
9use std::sync::Arc;
10
11use super::dynamic_dispatch::EntityId;
12use crate::{
13    traits::{BorrowPrimaryKey, Entity, SearchableEntity},
14    transaction::{ReadOutcome, Transaction},
15};

17impl Transaction {
18    /// Get an entity by its id.
19    ///
20    /// This produces a [`ReadOutcome`] which has information about whether the tx
21    /// knows about an upsert or delete of that entity, and also a set of filters
22    /// which can be used to exclude matching entities produced by the underlying DB.
23    ///
24    /// ## Caution
25    ///
26    /// This correctly filters out upserted entities which were later bulk-deleted.
27    /// However, it _cannot_ indicate bulk deletions for entities which were not
28    /// previously upserted in this transaction. Other composing methods which have
29    /// context from the database have to perform their own bulk-deletion filtering!
30    async fn get_by_entity_id<E>(&self, entity_id: &EntityId) -> ReadOutcome<E>
31    where
32        E: 'static + Entity + Send + Sync,
33    {
34        let operations = self.operations.read().await;
35
36        let bulk_delete_filters = operations.bulk_delete_filters::<E>();
37        let entity = operations.last_mutation_idx_for(entity_id).map(|entity_idx| {
38            let mut stored_entity = operations[entity_idx].as_upsert::<E>();
39
40            // run each filter coming after the stored entity against that entity
41            if let Some(entity) = stored_entity.as_ref()
42                && bulk_delete_filters.applies_after(entity, entity_idx)
43            {
44                stored_entity.take();
45            }
46
47            stored_entity
48        });
49        let filters = bulk_delete_filters.into_inner();
50
51        ReadOutcome { entity, filters }
52    }
53
54    /// Get an entity by its primary key.
55    ///
56    /// This produces a [`ReadOutcome`] which has information about whether the tx
57    /// knows about an upsert or delete of that entity, and also a set of filters
58    /// which can be used to exclude matching entities produced by the underlying DB.
59    pub(crate) async fn get<E>(&self, id: &E::PrimaryKey) -> ReadOutcome<E>
60    where
61        E: 'static + Entity + Send + Sync,
62    {
63        let entity_id = EntityId::from_primary_key::<E>(id.clone());
64        self.get_by_entity_id(&entity_id).await
65    }
66
67    /// Get an entity by the borrowed form of its primary key.
68    ///
69    /// This produces a [`ReadOutcome`] which has information about whether the tx
70    /// knows about an upsert or delete of that entity, and also a set of filters
71    /// which can be used to exclude matching entities produced by the underlying DB.
72    pub(crate) async fn get_borrowed<E>(&self, id: &E::BorrowedPrimaryKey) -> ReadOutcome<E>
73    where
74        E: 'static + Entity + BorrowPrimaryKey + Send + Sync,
75    {
76        let entity_id = EntityId::from_borrowed_primary_key::<E>(id);
77        self.get_by_entity_id(&entity_id).await
78    }
79
80    /// Every entity of type `E` which this transaction has upserted and not since removed.
81    ///
82    /// This does not replay the operations in order. It starts from the most recent upsert of each
83    /// entity id and discards the ones buried by a later delete or bulk deletion, which reaches the
84    /// same answer while only visiting operations which could contribute to it.
85    ///
86    /// The order of the returned entities is unspecified, as it follows `HashMap` iteration. A caller
87    /// which needs a particular order has to impose it.
88    ///
89    /// ## Caution
90    ///
91    /// This correctly filters out upserted entities which were later bulk-deleted.
92    /// However, it _cannot_ filter out bulk deletions for entities which were not
93    /// previously upserted in this transaction. Other composing methods which have
94    /// context from the database have to perform their own bulk-deletion filtering!
95    pub(super) async fn find_all_in_cache<E>(&self) -> Vec<Arc<E>>
96    where
97        E: 'static + Entity + Send + Sync,
98    {
99        let operations = self.operations.read().await;
100
101        let bulk_delete_filters = operations.bulk_delete_filters::<E>();
102        operations
103            .upsert_indices_for_type::<E>()
104            .filter_map(|(entity_id, idx)| {
105                if operations
106                    .last_delete_idx_for(entity_id)
107                    .is_some_and(|delete_idx| delete_idx > idx)
108                {
109                    // entity was deleted individually
110                    return None;
111                }
112
113                let entity = operations[idx]
114                    .as_upsert::<E>()
115                    .expect("upsert_indices_for_type contains correct indices");
116
117                if bulk_delete_filters.applies_after(&entity, idx) {
118                    // entity was bulk-deleted
119                    return None;
120                }
121
122                Some(entity)
123            })
124            .collect::<Vec<_>>()
125    }
126
127    /// Every entity of type `E` matching `search_key` which this transaction has upserted and not
128    /// since removed.
129    ///
130    /// As with [`Self::find_all_in_cache`], which this filters, the order is unspecified.
131    ///
132    /// ## Caution
133    ///
134    /// This correctly filters out upserted entities which were later bulk-deleted.
135    /// However, it _cannot_ filter out bulk deletions for entities which were not
136    /// previously upserted in this transaction. Other composing methods which have
137    /// context from the database have to perform their own bulk-deletion filtering!
138    async fn search_in_cache<E, SearchKey>(&self, search_key: &SearchKey) -> impl Iterator<Item = Arc<E>>
139    where
140        E: 'static + Entity + SearchableEntity<SearchKey> + Send + Sync,
141    {
142        self.find_all_in_cache::<E>()
143            .await
144            .into_iter()
145            .filter(|entity| entity.matches(search_key))
146    }
147
148    /// Find all the entities of type `E` in the database as modified by the operations in this transaction.
149    pub(crate) async fn find_all<E>(
150        &self,
151        persisted_records: impl IntoIterator<Item = Arc<E>>,
152    ) -> impl Iterator<Item = Arc<E>>
153    where
154        E: 'static + Clone + Entity + Send + Sync,
155    {
156        let cached_records = self.find_all_in_cache::<E>().await;
157        self.merge_records(cached_records, persisted_records).await
158    }
159
160    /// Find all the entities of type `E` in the database which match `search_key`,
161    /// as modified by the operations in this transaction.
162    pub(crate) async fn search<E, SearchKey>(
163        &self,
164        persisted_records: impl IntoIterator<Item = Arc<E>>,
165        search_key: &SearchKey,
166    ) -> impl Iterator<Item = Arc<E>>
167    where
168        E: 'static + Clone + Entity + SearchableEntity<SearchKey> + Send + Sync,
169    {
170        let cached_records = self.search_in_cache(search_key).await;
171        self.merge_records(cached_records, persisted_records).await
172    }
173}

175/// These tests pin down the contract documented on [`Transaction::get_by_entity_id`]: an upsert
176/// covered by a later bulk deletion must read as deleted, and a bulk deletion alone must not.
177///
178/// [`MlsPendingMessage`] is the only entity for now which is bulk-deletable, so it is what these use.
179/// Nothing here ever commits, so the database stays empty throughout and every outcome reflects
180/// only the operations buffered in the transaction.
181#[cfg(all(test, not(target_os = "unknown")))]
182mod tests {
183    use futures_lite::future;
184
185    use super::*;
186    use crate::{
187        Database,
188        entities::{ConversationId, MlsPendingMessage},
189        traits::PrimaryKey as _,
190    };
191
192    const CONVERSATION_ID: &[u8] = b"conversation which buffers a message";
193
194    fn pending_message() -> MlsPendingMessage {
195        MlsPendingMessage {
196            conversation_id: CONVERSATION_ID.to_owned(),
197            message: b"a message which arrived before we could decrypt it".to_vec(),
198        }
199    }
200
201    /// Bulk-delete every pending message of [`CONVERSATION_ID`].
202    async fn clear_conversation(tx: &Transaction) {
203        tx.bulk_remove::<MlsPendingMessage, ConversationId>(CONVERSATION_ID.to_vec().into())
204            .await;
205    }
206
207    /// What `get_by_entity_id` reports for `message`.
208    async fn outcome_for(tx: &Transaction, message: &MlsPendingMessage) -> ReadOutcome<MlsPendingMessage> {
209        let entity_id = EntityId::from_primary_key::<MlsPendingMessage>(message.primary_key());
210        tx.get_by_entity_id(&entity_id).await
211    }
212
213    /// The documented property: an entity upserted in this transaction and then bulk-deleted
214    /// reads back as deleted, not as present.
215    #[test]
216    fn an_upsert_covered_by_a_later_bulk_deletion_reads_as_deleted() {
217        future::block_on(async {
218            let store = Database::open_in_memory().unwrap();
219            let tx = store.new_transaction().await.unwrap();
220            let message = pending_message();
221
222            tx.save(message.clone()).await.unwrap();
223            clear_conversation(&tx).await;
224
225            let outcome = outcome_for(&tx, &message).await;
226            assert!(
227                matches!(outcome.entity, Some(None)),
228                "the upsert was covered by a later bulk deletion, so it must read as deleted, got {:?}",
229                outcome.entity
230            );
231        });
232    }
233
234    /// The mirror image: only bulk deletions which come *after* the upsert can bury it.
235    ///
236    /// The filter is still reported, because it has to be applied to whatever the database
237    /// produces for other entities; see the note on [`ReadOutcome`].
238    #[test]
239    fn an_upsert_after_a_bulk_deletion_reads_as_present() {
240        future::block_on(async {
241            let store = Database::open_in_memory().unwrap();
242            let tx = store.new_transaction().await.unwrap();
243            let message = pending_message();
244
245            clear_conversation(&tx).await;
246            tx.save(message.clone()).await.unwrap();
247
248            let outcome = outcome_for(&tx, &message).await;
249            let found = outcome
250                .entity
251                .clone()
252                .expect("this transaction upserted the message, so the cache knows about it")
253                .expect("the upsert came after the bulk deletion, so it must not have been filtered out");
254            assert_eq!(*found, message);
255            assert!(
256                outcome.should_omit(&found),
257                "the bulk deletion still has to be reported as a filter over database records"
258            );
259        });
260    }
261
262    /// Bulk deletions on both sides of the upsert: the later one still applies.
263    ///
264    /// Reading the whole filter list without regard to position would pass
265    /// [`an_upsert_covered_by_a_later_bulk_deletion_reads_as_deleted`] and ignoring position
266    /// entirely would pass [`an_upsert_after_a_bulk_deletion_reads_as_present`]; only this case
267    /// distinguishes filtering by position from filtering by presence.
268    #[test]
269    fn an_upsert_straddled_by_bulk_deletions_reads_as_deleted() {
270        future::block_on(async {
271            let store = Database::open_in_memory().unwrap();
272            let tx = store.new_transaction().await.unwrap();
273            let message = pending_message();
274
275            clear_conversation(&tx).await;
276            tx.save(message.clone()).await.unwrap();
277            clear_conversation(&tx).await;
278
279            let outcome = outcome_for(&tx, &message).await;
280            assert!(
281                matches!(outcome.entity, Some(None)),
282                "the second bulk deletion follows the upsert, so the message must read as deleted, got {:?}",
283                outcome.entity
284            );
285        });
286    }
287
288    /// The documented limit of the property: with no upsert to attach it to, a bulk deletion is
289    /// reported only as a filter, and the outcome itself says nothing.
290    ///
291    /// This is why the doc comment tells composing methods with database context to do their own
292    /// bulk-deletion filtering.
293    #[test]
294    fn a_bulk_deletion_alone_yields_no_outcome_but_still_yields_a_filter() {
295        future::block_on(async {
296            let store = Database::open_in_memory().unwrap();
297            let tx = store.new_transaction().await.unwrap();
298            let message = pending_message();
299
300            clear_conversation(&tx).await;
301
302            let outcome = outcome_for(&tx, &message).await;
303            assert!(
304                outcome.entity.is_none(),
305                "nothing in this transaction mutated this message individually, so the cache cannot speak for it"
306            );
307            assert!(
308                outcome.should_omit(&message),
309                "the caller must still be told to omit any instance the database produces"
310            );
311        });
312    }
