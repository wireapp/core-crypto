2    entities::{OldEpochData, OldEpochDataPkRef, OldEpochMemberData},

8use super::{ConversationMut, Result};
9use crate::{KeystoreError, RecursiveError, TlsCodecError};
10
11impl ConversationMut {
12    /// Idempotently record per-epoch durable data for TNT messages.
13    pub(crate) async fn record_epoch_data(&self, group: &MlsGroup) -> Result<()> {
14        let tx = self.tx_context.inner().map_err(RecursiveError::context(
15            "getting conversation's tx context for idempotence check for recording epoch data",
16        ))?;
17        let tx = tx.transaction();
18
19        let old_epoch_data_key = OldEpochDataPkRef::new(self.id.keystore(), group.epoch().as_u64());
20        if tx
21            .get_borrowed::<OldEpochData>(old_epoch_data_key)
22            .await
23            .map_err(KeystoreError::wrap("finding old epoch data for current epoch"))?
24            .is_some()
25        {
26            return Ok(());
27        }
28
29        let tx = self.tx_context.inner().map_err(RecursiveError::context(
30            "getting conversation's tx context for recording epoch data",
31        ))?;
32        let tx = tx.transaction();
33        self.create_old_epoch_data(group)
34            .await
35            .map_err(RecursiveError::context("creating tnt secret for per-epoch persistence"))?
36            .save(tx)
37            .map_err(KeystoreError::wrap("persisting old epoch data for current epoch"))?;
38        for member in group.members() {
39            let credential = member
40                .credential
41                .tls_serialize_detached()
42                .map_err(TlsCodecError::serialize("member credential"))?;
43            OldEpochMemberData {
44                conversation_id: self.id().keystore().into(),
45                epoch: group.epoch().as_u64(),
46                leaf_node_index: member.index.u32(),
47                signature_key: member.signature_key,
48                credential,
49            }
50            .save(tx)
51            .map_err(KeystoreError::wrap(
52                "persisting old epoch member data for current epoch",
53            ))?;
54        }
55        Ok(())
56    }
57}
