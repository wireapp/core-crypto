10/// Entity representing a stored Proteus prekey.
11///
12/// This entity cannot be updated in the DB. A prekey id is published to peers in a prekey bundle,
13/// and overwriting the prekey it names would leave those peers holding a bundle no session can be
14/// established from. A save under an id which is already taken therefore reports
15/// [`CryptoKeystoreError::AlreadyExists`] rather than replacing the stored prekey.
16#[derive(core_crypto_macros::Debug, Clone, Zeroize, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
17#[zeroize(drop)]
18pub struct ProteusPrekey {
19    pub id: u16,
20    #[sensitive]
21    pub prekey: Vec<u8>,
22}
23
24impl ProteusPrekey {
25    pub fn from_raw(id: u16, prekey: Vec<u8>) -> Self {
26        Self { id, prekey }
27    }
28
29    /// Get the lowest free prekey id.
30    ///
31    /// Ids freed by deletion are handed out before the id space is extended.
32    ///
33    /// Returns [`CryptoKeystoreError::NoFreePrekeyId`] if every assignable id is taken.
34    pub fn get_free_id(conn: &Connection) -> CryptoKeystoreResult<u16> {
35        /// Lowest prekey id assignable to an ordinary prekey.
36        ///
37        /// Id 0 is technically legal in Proteus but has never been auto-assigned here, so it stays reserved.
38        const MIN_ASSIGNABLE_ID: u16 = 1;
39        /// Highest prekey id assignable to an ordinary prekey.
40        ///
41        /// `u16::MAX` is reserved for the last-resort prekey. This crate depends only on
42        /// `proteus-traits`, so it can't name `proteus_wasm::keys::MAX_PREKEY_ID` directly;
43        /// `ProteusCentral::last_resort_prekey_id` in `core-crypto` is the other half of this
44        /// invariant.
45        const MAX_ASSIGNABLE_ID: u16 = u16::MAX - 1;
46
47        // The scan walks the index on `id` in ascending order and `LIMIT 1` stops it at the first
48        // gap, so the cost tracks the position of that gap rather than the size of the table.
49        // It can only find gaps above an id which is present, hence the `WHEN`: if the lowest
50        // assignable id is itself free, no scan is needed. Bounding `t.id` from above keeps the
51        // last-resort prekey out of the scan, so it can neither yield the unassignable candidate
52        // `u16::MAX` nor overflow to 65536.
53        let mut stmt = conn.prepare_cached(
54            "
55            SELECT CASE
56                WHEN NOT EXISTS (SELECT 1 FROM proteus_prekeys WHERE id = :min_assignable)
57                    THEN :min_assignable
58                ELSE (
59                    SELECT t.id + 1
60                    FROM proteus_prekeys AS t
61                    WHERE t.id >= :min_assignable
62                      AND t.id < :max_assignable
63                      AND NOT EXISTS (SELECT 1 FROM proteus_prekeys p WHERE p.id = t.id + 1)
64                    ORDER BY t.id
65                    LIMIT 1
66                )
67            END
68            ",
69        )?;
70        // a NULL here is the `ELSE` branch finding no gap below `:max_assignable`, which is
71        // the exhausted case: an empty table takes the `WHEN` branch instead.
72        stmt.query_one(
73            rusqlite::named_params! {
74                ":min_assignable": MIN_ASSIGNABLE_ID,
75                ":max_assignable": MAX_ASSIGNABLE_ID,
76            },
77            |row| row.get::<_, Option<u16>>(0),
78        )?
79        .ok_or(CryptoKeystoreError::NoFreePrekeyId)
80    }
81}

91impl Entity for ProteusPrekey {
92    const TABLE_NAME: &'static str = "proteus_prekeys";

94    fn get(conn: &rusqlite::Connection, key: &u16) -> CryptoKeystoreResult<Option<Self>> {
95        get_helper(conn, "id", *key, |row| {
96            Ok(Self::from_raw(row.get("id")?, row.get("key")?))
97        })
98    }
99
100    fn count(conn: &rusqlite::Connection) -> CryptoKeystoreResult<u32> {
101        count_helper::<Self>(conn)
102    }
103
104    fn load_all(conn: &rusqlite::Connection) -> CryptoKeystoreResult<Vec<Self>> {
105        load_all_helper(conn, |row| Ok(Self::from_raw(row.get("id")?, row.get("key")?)))
106    }
107}

109impl EntityDatabaseMutation for ProteusPrekey {
110    fn save<'a, Tx>(&self, tx: &'a Tx) -> CryptoKeystoreResult<()>
111    where
112        &'a Tx: Into<Transactionlike<'a>>,
113    {
114        let conn = tx.into().conn()?;
115        let mut stmt = conn.prepare_cached("INSERT INTO proteus_prekeys (id, key) VALUES (?, ?)")?;
116        stmt.execute(rusqlite::params![self.id, self.prekey])
117            .map_err(CryptoKeystoreError::map_already_exists(Self::TABLE_NAME))?;
118        Ok(())
119    }
120
121    fn delete<'a, Tx>(tx: &'a Tx, id: &u16) -> CryptoKeystoreResult<bool>
122    where
123        &'a Tx: Into<Transactionlike<'a>>,
124    {
125        delete_helper::<Self, _>(tx, "id", *id)
126    }
