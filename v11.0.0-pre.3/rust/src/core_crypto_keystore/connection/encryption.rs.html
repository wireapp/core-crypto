5/// Run `PRAGMA name = 'value'` without going through rusqlite's pragma helpers.
6///
7/// Since rusqlite 0.41, those helpers first look the name up in `pragma_pragma_list`. That query reads
8/// the schema, which fails on an encrypted database before its key is set, and sqlcipher's `cipher_*`
9/// pragmas are not listed there at all. Use this for the key pragmas, for sqlcipher's own pragmas,
10/// and for anything which has to run before the key is set.
11///
12/// `name` is put into the SQL verbatim, so it must be a constant. `value` is quoted.
13///
14/// Upstream issue: <https://github.com/rusqlite/rusqlite/issues/1923>.
15pub(super) fn set_pragma_unchecked(
16    conn: &rusqlite::Connection,
17    name: &'static str,
18    value: &str,
19) -> rusqlite::Result<()> {
20    let mut sql = format!("PRAGMA {name} = '{}'", value.replace('\'', "''"));
21    let result = conn.execute_batch(&sql);
22    sql.zeroize();
23    result
24}

26/// Format a key pragma according to the raw form supported by sqlcipher
27fn set_key_pragma(
28    conn: &mut rusqlite::Connection,
29    key: &DatabaseKey,
30    pragma_name: &'static str,
31) -> CryptoKeystoreResult<()> {
32    // Make sqlite use raw key data, without key derivation. Also make sure to zeroize
33    // the string containing the key after the call.
34    let mut x_hex_key = format!("x'{}'", hex::encode(key));
35    let result = set_pragma_unchecked(conn, pragma_name, &x_hex_key);
36    x_hex_key.zeroize();
37    result.map_err(Into::into)
38}

40/// Decrypt the database by setting the key pragma appropriately
41pub(super) fn decrypt(conn: &mut rusqlite::Connection, key: &DatabaseKey) -> CryptoKeystoreResult<()> {
42    set_key_pragma(conn, key, "key")
43}
44
45/// Reencrypt the database with a new key.
46pub(super) fn rekey(conn: &mut rusqlite::Connection, new_key: &DatabaseKey) -> CryptoKeystoreResult<()> {
47    set_key_pragma(conn, new_key, "rekey")
48}

50/// Encrypt an unencrypted database with a key
51pub(super) fn key(conn: &mut rusqlite::Connection, key: &DatabaseKey) -> CryptoKeystoreResult<()> {
52    set_key_pragma(conn, key, "key")
