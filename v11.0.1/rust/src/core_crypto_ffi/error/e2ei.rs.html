10/// Errors produced by the end-to-end identity layer.
11#[derive(Debug, thiserror::Error, uniffi::Error)]
12#[expect(missing_docs)] // error variants are self-describing
13pub enum E2eiError {
14    #[error("Certificate acquisition is not supported for {cipher_suite:?}")]
15    UnsupportedCipherSuite { cipher_suite: CipherSuite },
16    /// One of the `PkiEnvironmentHooks` callbacks returned an error.
17    #[error("A PKI environment hook failed: {reason}")]
18    HookFailed { reason: String },
19    #[error("HTTP response is missing header '{header}'")]
20    MissingHeader { header: String },
21    #[error("The trust anchor certificate couldn't be loaded from the database.")]
22    NoTrustAnchor,
23    #[error("The trust anchor certificate already exists in the database.")]
24    TrustAnchorAlreadyExists,
25    #[error("Failed to fetch CRL from '{uri}': HTTP {status}")]
26    CrlFetchUnsuccessful { uri: String, status: u16 },
27    /// The ACME account is invalid, revoked or deactivated.
28    #[error("ACME account error: {msg}")]
29    AcmeAccount { msg: String },
30    /// The ACME order is invalid, not yet valid, expired, or has the wrong identifiers.
31    #[error("ACME order error: {msg}")]
32    AcmeOrder { msg: String },
33    /// An ACME authorization or challenge failed, or the challenge is still being processed.
34    #[error("ACME challenge error: {msg}")]
35    AcmeChallenge { msg: String },
36    /// The client ID, display name, handle or domain in the certificate is missing, malformed or
37    /// does not match the expected one.
38    #[error("Invalid identifier: {msg}")]
39    InvalidIdentifier { msg: String },
40    /// The certificate is malformed, or does not match the signing keypair.
41    #[error("Invalid certificate: {msg}")]
42    InvalidCertificate { msg: String },
43    #[error("{msg}")]
44    Other { msg: String },
45}
46
47impl E2eiError {
48    fn other(err: &impl ToString) -> Self {
49        Self::Other { msg: err.to_string() }
50    }
51}
52
53// The matches below deliberately have no wildcard arm: a new upstream variant should force us to
54// decide where it belongs, rather than silently ending up in `Other`.
55
56impl From<&E2eIdentityError> for E2eiError {
57    fn from(value: &E2eIdentityError) -> Self {
58        match value {
59            E2eIdentityError::UnsupportedCipherSuite(cipher_suite) => Self::UnsupportedCipherSuite {
60                cipher_suite: (*cipher_suite).into(),
61            },
62            E2eIdentityError::AcmeError(err) => err.into(),
63            E2eIdentityError::JwtError(_) | E2eIdentityError::JsonWebTokenError(_) => Self::other(value),
64            E2eIdentityError::PkiEnvError(err) => err.into(),
65            E2eIdentityError::ValidationError(err) => err.into(),
66        }
67    }
68}

70impl From<&AcquisitionError> for E2eiError {
71    fn from(value: &AcquisitionError) -> Self {
72        match value {
73            AcquisitionError::UnsupportedCipherSuite(cipher_suite) => Self::UnsupportedCipherSuite {
74                cipher_suite: (*cipher_suite).into(),
75            },
76            AcquisitionError::HookFailed(err) => err.into(),
77            AcquisitionError::MissingHeader(header) => Self::MissingHeader {
78                header: (*header).to_owned(),
79            },
80            AcquisitionError::Acme(err) => err.into(),
81            AcquisitionError::InvalidCertificate(err) => err.into(),
82            AcquisitionError::Json(_) | AcquisitionError::RustyJwtError(_) | AcquisitionError::JsonWebToken(_) => {
83                Self::other(value)
84            }
85        }
86    }
87}

97impl From<&pki_env::Error> for E2eiError {
98    fn from(value: &pki_env::Error) -> Self {
99        match value {
100            pki_env::Error::NoTrustAnchor => Self::NoTrustAnchor,
101            pki_env::Error::TrustAnchorAlreadyExists => Self::TrustAnchorAlreadyExists,
102            pki_env::Error::CrlFetchUnsuccessful { uri, status } => Self::CrlFetchUnsuccessful {
103                uri: uri.clone(),
104                status: *status,
105            },
106            pki_env::Error::HooksError(err) => err.into(),
107            pki_env::Error::Validation(err) => err.into(),
108            pki_env::Error::UrlError(_)
109            | pki_env::Error::JsonError(_)
110            | pki_env::Error::X509CertDerError(_)
111            | pki_env::Error::KeystoreError(_)
112            | pki_env::Error::Certval(_)
113            | pki_env::Error::Spki(_) => Self::other(value),
114        }
115    }
116}

118impl From<&validation::Error> for E2eiError {
119    fn from(value: &validation::Error) -> Self {
120        match value {
121            validation::Error::MissingSki => Self::InvalidCertificate { msg: value.to_string() },
122            validation::Error::LockPoisonError
123            | validation::Error::CannotDetermineCurrentTime
124            | validation::Error::CannotDetermineVerificationStatus
125            | validation::Error::ImplementationError
126            | validation::Error::DerError(_)
127            | validation::Error::PemError(_)
128            | validation::Error::CertValError(_) => Self::other(value),
129        }
130    }
131}

133impl From<&AcmeError> for E2eiError {
134    fn from(value: &AcmeError) -> Self {
135        let msg = value.to_string();
136        match value {
137            AcmeError::Account(_) => Self::AcmeAccount { msg },
138            AcmeError::Order(_) | AcmeError::Finalize(_) => Self::AcmeOrder { msg },
139            AcmeError::Authz(_) | AcmeError::Challenge(_) => Self::AcmeChallenge { msg },
140            AcmeError::Implementation
141            | AcmeError::ClientImplementation(_)
142            | AcmeError::SmallstepImplementation(_)
143            | AcmeError::Json(_)
144            | AcmeError::Jwt(_)
145            | AcmeError::X509CertBuilder(_)
146            | AcmeError::Oid(_)
147            | AcmeError::Der(_)
148            | AcmeError::Pem(_)
149            | AcmeError::Pkcs8(_)
150            | AcmeError::JsonWebToken(_)
151            | AcmeError::Signature(_)
152            | AcmeError::Utf8(_) => Self::other(value),
153        }
154    }
155}

157impl From<&CertificateError> for E2eiError {
158    fn from(value: &CertificateError) -> Self {
159        match value {
160            CertificateError::InvalidClientId
161            | CertificateError::ClientIdMismatch
162            | CertificateError::DisplayNameMismatch
163            | CertificateError::HandleMismatch
164            | CertificateError::DomainMismatch
165            | CertificateError::MissingDisplayName
166            | CertificateError::MissingHandle
167            | CertificateError::MissingDomain
168            | CertificateError::MissingClientId => Self::InvalidIdentifier { msg: value.to_string() },
169            CertificateError::InvalidFormat
170            | CertificateError::AlgorithmMismatch
171            | CertificateError::KeyMismatch
172            | CertificateError::InvalidSigningKeypair
173            | CertificateError::UnsupportedPublicKey => Self::InvalidCertificate { msg: value.to_string() },
174            CertificateError::Validation(err) => err.into(),
175            CertificateError::Acme(err) => err.into(),
176            CertificateError::Der(_)
177            | CertificateError::Utf8(_)
178            | CertificateError::RustyJwtError(_)
179            | CertificateError::JsonWebToken(_) => Self::other(value),
180        }
181    }
