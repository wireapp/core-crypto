13/// A `CoreCryptoCommand` has an `execute` method which accepts a `CoreCryptoContext` and returns nothing.
14///
15/// It is the argument to a `CoreCrypto::transaction` call.
16#[uniffi::export(with_foreign)]
17#[cfg_attr(target_os = "unknown", async_trait::async_trait(?Send))]
18#[cfg_attr(not(target_os = "unknown"), async_trait::async_trait)]
19pub trait CoreCryptoCommand: Send + Sync {
20    /// Will be called inside a transaction in CoreCrypto
21    async fn execute(&self, context: Arc<CoreCryptoContext>) -> CoreCryptoResult<()>;
22}
23
24/// When building outside WASM, any async function of appropriate signature is a `CoreCryptoCommand`.
25
26#[cfg_attr(target_os = "unknown", async_trait::async_trait(?Send))]
27#[cfg_attr(not(target_os = "unknown"), async_trait::async_trait)]
28impl<F, Fut> CoreCryptoCommand for F
29where
30    F: Fn(Arc<CoreCryptoContext>) -> Fut + Send + Sync,
31    Fut: Future<Output = CoreCryptoResult<()>> + Send,
32{
33    async fn execute(&self, context: Arc<CoreCryptoContext>) -> CoreCryptoResult<()> {
34        self(context).await
35    }
36}

38/// In uniffi, a Command is an Arc wrapping a dyn trait object
39type Command = Arc<dyn CoreCryptoCommand>;

41#[uniffi::export]
42impl CoreCryptoFfi {
43    /// Starts a new transaction in Core Crypto. If the callback succeeds, it will be committed,
44    /// otherwise, every operation performed with the context will be discarded.
45    ///
46    /// When calling this function from within Rust, async functions accepting a context
47    /// implement `CoreCryptoCommand`, so operations can be defined inline as follows:
48    ///
49    /// ```ignore
50    /// core_crypto.transaction(Arc::new(async |context| {
51    ///     // your implementation here
52    ///     Ok(())
53    /// }))?;
54    /// ```
55    pub async fn transaction_ffi(&self, command: Command) -> CoreCryptoResult<()> {
56        log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 1; "awaiting transaction semaphore");
57        let inner_context = Arc::new(self.inner.new_transaction().await?);
58        log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 2; "acquired semaphore; creating context");
59
60        let context = CoreCryptoContext {
61            inner: inner_context.clone(),
62            #[cfg(feature = "cancellable-transactions")]
63            cancellation_slot: self.cancellation_slot.clone(),
64        };
65
66        // We need one more layer of Arc-wrapping in uniffi. It's kind of silly, given the
67        // also-mandatory Arc-wrapping internally, but that's the price we have to pay in order
68        // to reuse the code in both target contexts.
69        let context = Arc::new(context);
70
71        log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 3; "created context; executing command");
72        let result = command.execute(context).await;
73        match result {
74            Ok(()) => {
75                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 4, command_success = true; "command succeeded; committing transaction");
76                inner_context.finish().await?;
77                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 5, command_success = true; "exiting successfully");
78                Ok(())
79            }
80            Err(err) => {
81                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 4, command_success = false; "command failed; aborting transaction");
82                inner_context.abort().await?;
83                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 5, command_success = false, err:err; "exiting propagating error");
84                Err(err)
85            }
86        }
87    }
88}

92impl CoreCryptoFfi {
93    /// Like `transaction_ffi`, but cancellable.
94    ///
95    /// Cancelling the token aborts the transaction and stops waiting for any
96    /// in-flight `MlsTransport` callback or `PkiEnvironment` hook associated with it.
97    ///
98    /// For implementation details, see the [`cancellation` docs][crate::cancellation].
99    pub async fn transaction_ffi_cancellable(
100        &self,
101        command: Command,
102        cancellation: Arc<CoreCryptoCancellationToken>,
103    ) -> CoreCryptoResult<()> {
104        log::info!(
105            scope = "CoreCryptoFfi::transaction_ffi", stage = 1;
106            "racing cancellation token against acquisition of transaction semaphore"
107        );
108        // Prefer cancellation so a pre-cancelled token cannot start a transaction.
109        let inner_context = futures_util::select_biased! {
110            _ = cancellation.cancelled().fuse() => return Err(CoreCryptoError::TransactionCanceled),
111            inner_context_result = self.inner.new_transaction().fuse() => inner_context_result?,
112        };
113
114        log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 2; "acquired semaphore; creating context");
115        let inner_context = Arc::new(inner_context);
116
117        // The cancellation token only needs to be published while the command is running: the
118        // commit and abort paths make no foreign calls, so nothing reads the slots there. Scoping
119        // the guards to the command means the slots are always empty by the time the transaction
120        // semaphore is released, whenever during the commit that happens.
121        let result = {
122            // Only the transaction owning the semaphore may publish its token.
123            let _cancellation_guard = self.cancellation_slot.enter(cancellation.clone())?;
124            let pki_cancellation_slot = self
125                .pki_environment
126                .read()
127                .await
128                .as_ref()
129                .map(|environment| environment.cancellation_slot.clone());
130            let _pki_cancellation_guard = pki_cancellation_slot
131                .map(|slot| slot.enter(cancellation.clone()))
132                .transpose()?;
133
134            let context = Arc::new(CoreCryptoContext {
135                inner: inner_context.clone(),
136                cancellation_slot: self.cancellation_slot.clone(),
137            });
138
139            log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 3; "created context; racing command against cancellation");
140
141            // Prefer cancellation when both futures become ready together.
142            futures_util::select_biased! {
143                _ = cancellation.cancelled().fuse() => Err(CoreCryptoError::TransactionCanceled),
144                result = command.execute(context).fuse() => result,
145            }
146        };
147
148        match result {
149            Ok(()) => {
150                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 4, command_success = true; "command succeeded; committing transaction");
151                inner_context.finish().await?;
152                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 5, command_success = true; "exiting successfully");
153                Ok(())
154            }
155            Err(err) => {
156                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 4, command_success = false; "command failed; aborting transaction");
157                inner_context.abort().await?;
158                log::info!(scope = "CoreCryptoFfi::transaction_ffi", stage = 5, command_success = false, err:err; "exiting propagating error");
159                Err(err)
160            }
161        }
162    }
163}

270    /// A transaction must be able to start while the previous transaction's epoch observer runs.
271    ///
272    /// The keystore releases the transaction semaphore as its commit completes, which is before
273    /// `finish` notifies epoch observers. So if `transaction_ffi_cancellable` held its cancellation
274    /// guards until it returned, the waiting transaction would acquire the semaphore while the
275    /// previous transaction's token was still in the slot, and `CancellationSlot::enter` would
276    /// panic.
277    #[macro_rules_attribute::apply(smol_macros::test)]
278    async fn transaction_can_start_while_previous_epoch_observer_runs() {
279        let database = Arc::new(Database::in_memory().await.unwrap());
280        let core_crypto = core_crypto_new(&database).unwrap();
281
282        let client_id = Arc::new(ClientId::new(
283            Arc::new(Uuid::new(CLIENT_UUID).unwrap()),
284            Arc::new(DeviceId::new(1)),
285            DOMAIN.to_owned(),
286        ));
287        let conversation_id = ConversationId::new(b"epoch observer conversation".to_vec());
288
289        core_crypto
290            .transaction_ffi(Arc::new(Setup {
291                client_id,
292                conversation_id: conversation_id.clone(),
293            }))
294            .await
295            .unwrap();
296
297        let release = Arc::new(Semaphore::new(0));
298        core_crypto
299            .register_epoch_observer(Arc::new(ParkingEpochObserver {
300                release: release.clone(),
301            }))
302            .await
303            .unwrap();
304
305        // A single-threaded executor polls these in order, so the interleaving is deterministic:
306        // the first transaction acquires the semaphore immediately and runs until it parks in the
307        // epoch observer, by which point it has committed and released the semaphore. Only then
308        // does the second transaction get to acquire it and reach for the cancellation slot.
309        let (epoch_change, next_transaction) = futures_util::future::join(
310            core_crypto.transaction_ffi_cancellable(
311                Arc::new(ChangeEpoch { conversation_id }),
312                Arc::new(CoreCryptoCancellationToken::new()),
313            ),
314            core_crypto.transaction_ffi_cancellable(
315                Arc::new(ReleaseObserver { release }),
316                Arc::new(CoreCryptoCancellationToken::new()),
317            ),
318        )
319        .await;
320
321        epoch_change.unwrap();
322        next_transaction.unwrap();
323    }
