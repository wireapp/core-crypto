1use std::sync::Arc;

3use crate::transaction::BulkDeleteFilter;

5/// The outcome of a read operation on a transaction
6///
7/// Transactions record not just individual entity upsert and delete operations, but also
8/// bulk deletions. A read outcome therefore includes information not just about the entity requested,
9/// but filters which can be used to exclude other entities produced by the database.
10///
11/// NOTE: it is possible for this outcome to have both `entity: Some(Some(e))` and `outcome.should_omit(e)`.
12/// This just indicates that a bulk deletion covered `e`, but then it was re-upserted afterwards.
13pub(crate) struct ReadOutcome<E> {
14    /// The outcome of this read operation as applied to the operations of this transaction:
15    ///
16    /// * `Some(Some(E))` - the transaction cache contains the record
17    /// * `Some(None)` - the deletion of the record has been cached
18    /// * `None` - there is no information about the record in the cache
19    pub(crate) entity: Option<Option<Arc<E>>>,
20    /// Each bulk delete operation produces a filter which must be applied to instances from the DB.
21    ///
22    /// If any function in this list returns `true` for an entity instance, then that entity has been
23    /// affected by a bulk deletion and should be omitted from result lists.
24    ///
25    /// The position of each bulk deletion within the transaction is deliberately not recorded here.
26    /// These filters exist to be applied to records loaded from the database, and such a record
27    /// predates every operation in the transaction, so every bulk deletion in this list necessarily
28    /// comes after it and all of them apply. Contrast `BulkDeleteFilters::applies_after`, which is
29    /// what the same filters need when the candidate entity came out of the transaction cache and so
30    /// has a position of its own to be compared against.
31    pub(crate) filters: Vec<BulkDeleteFilter<E>>,
32}
33
34impl<E> ReadOutcome<E> {
35    /// Helper function to apply all the filters to an entity instance.
36    ///
37    /// If this returns `true` then the entity should be omitted from results.
38    ///
39    /// Applying *all* of them is only correct for an entity which did not come out of this
40    /// transaction; see the note on [`Self::filters`].
41    pub(crate) fn should_omit(&self, entity: &E) -> bool {
42        self.filters.iter().any(|filter| (filter)(entity))
43    }
44}
45
46// manually derived to avoid undesired `E: Default` bound
47impl<E> Default for ReadOutcome<E> {
48    fn default() -> Self {
49        Self {
50            entity: Default::default(),
51            filters: Default::default(),
52        }
53    }
54}
