23#[derive(Debug, thiserror::Error)]
24pub enum RustyX509CheckError {
25    /// Failed mapping a DER certificate
26    #[error(transparent)]
27    DerError(#[from] x509_cert::der::Error),
28    /// PEM de/serialization error
29    #[error("PEM en/decoding error: {0}")]
30    PemError(x509_cert::der::pem::Error),
31    /// Poisoned lock error
32    #[error("A lock has been poisoned and cannot be recovered from.")]
33    LockPoisonError,
34    /// Error for when the current UNIX epoch time cannot be determined.
35    #[error("Cannot determine current UNIX epoch")]
36    CannotDetermineCurrentTime,
37    /// Certificate / revocation validation error
38    #[error("Certificate validation error: {0}")]
39    CertValError(certval::Error),
40    /// Error when we have no idea what the cert status is
41    #[error("Something went wrong, we cannot determine if this certificate is OK. You might want to ignore this")]
42    CannotDetermineVerificationStatus,
43    /// Required 'Subject Key Identifier' extension is missing
44    #[error("Required 'Subject Key Identifier' extension is missing")]
45    MissingSki,
46    /// Implementation error
47    #[error("Implementation error")]
48    ImplementationError,
49}

51impl From<x509_cert::der::pem::Error> for RustyX509CheckError {
52    fn from(value: x509_cert::der::pem::Error) -> Self {
53        RustyX509CheckError::PemError(value)
54    }
55}

65#[derive(Debug, Clone, Copy, PartialEq, Eq)]
66pub enum IdentityStatus {
67    /// All is fine
68    Valid,
69    /// The Certificate is expired
70    Expired,
71    /// The Certificate is revoked
72    Revoked,
73}
74
75impl IdentityStatus {
76    pub async fn from_cert(cert: &x509_cert::Certificate, env: &crate::pki_env::PkiEnvironment) -> Self {
77        match env.validate_cert(cert).await {
78            Err(RustyX509CheckError::CertValError(certval::Error::PathValidation(e))) => match e {
79                PathValidationStatus::InvalidNotAfterDate => IdentityStatus::Expired,
80                PathValidationStatus::CertificateRevoked
81                | PathValidationStatus::CertificateRevokedEndEntity
82                | PathValidationStatus::NoPathsFound
83                | PathValidationStatus::CertificateRevokedIntermediateCa => IdentityStatus::Revoked,
84                _ => IdentityStatus::Valid,
85            },
86            _ => IdentityStatus::Valid,
87        }
88    }
89}

91/// Extracts the CRL Distribution points that are FullName URIs from the Certificate
92pub fn extract_crl_uris(
93    cert: &x509_cert::Certificate,
94) -> RustyX509CheckResult<Option<std::collections::HashSet<String>>> {
95    use certval::validator::{PDVCertificate, PDVExtension};
96    use x509_cert::ext::pkix::name::{DistributionPointName, GeneralName};
97
98    Ok(PDVCertificate::try_from(cert.clone())?
99        .get_extension(&const_oid::db::rfc5280::ID_CE_CRL_DISTRIBUTION_POINTS)?
100        .and_then(|ext| {
101            let PDVExtension::CrlDistributionPoints(crl_distribution_points) = ext else {
102                return None;
103            };
104
105            Some(crl_distribution_points.0.iter().fold(
106                Default::default(),
107                |mut set: std::collections::HashSet<String>, dp| {
108                    if let Some(DistributionPointName::FullName(dp_full_names)) = dp.distribution_point.as_ref() {
109                        for gn in dp_full_names.iter() {
110                            if let GeneralName::UniformResourceIdentifier(uri) = gn {
111                                set.insert(uri.to_string());
112                            }
113                        }
114                    }
115
116                    set
117                },
118            ))
119        }))
120}

122fn check_cpr(cpr: CertificationPathResults) -> RustyX509CheckResult<()> {
123    if let Some(validation_status) = cpr.get_validation_status() {
124        match validation_status {
125            certval::PathValidationStatus::Valid => Ok(()),
126            // No CRL is available, this is fine
127            certval::PathValidationStatus::RevocationStatusNotDetermined
128            | certval::PathValidationStatus::RevocationStatusNotAvailable => Ok(()),
129            validation_status => Err(RustyX509CheckError::CertValError(certval::Error::PathValidation(
130                validation_status,
131            ))),
132        }
133    } else {
134        Err(RustyX509CheckError::CannotDetermineVerificationStatus)
135    }
136}

138pub(crate) fn now() -> RustyX509CheckResult<u64> {
139    Ok(web_time::SystemTime::now()
140        .duration_since(web_time::SystemTime::UNIX_EPOCH)
141        .map_err(|_| RustyX509CheckError::CannotDetermineCurrentTime)?
142        .as_secs())
143}

145/// Initializes a certval PkiEnvironment using the provided params
146pub(crate) fn prepare_environment(
147    trust_roots: &[x509_cert::anchor::TrustAnchorChoice],
148    intermediates: &[x509_cert::Certificate],
149    crls: &[x509_cert::crl::CertificateList<Raw>],
150) -> RustyX509CheckResult<certval::environment::PkiEnvironment> {
151    let toi = TimeOfInterest::from_unix_secs(now()?)?;
152
153    let mut cps = CertificationPathSettings::new();
154    cps.set_time_of_interest(toi);
155
156    // Make a Certificate source for intermediate CA certs
157    let mut cert_source = CertSource::new();
158    for (i, cert) in intermediates.iter().enumerate() {
159        cert_source.push(certval::CertFile {
160            filename: format!("Intermediate CA #{i} [{}]", cert.tbs_certificate().subject()),
161            bytes: cert.to_der()?,
162        });
163    }
164
165    cert_source.initialize(&cps)?;
166
167    // Make a TrustAnchor source
168    let mut trust_anchors = TaSource::new();
169    for (i, root) in trust_roots.iter().enumerate() {
170        trust_anchors.push(certval::CertFile {
171            filename: format!("TrustAnchor #{i}"),
172            bytes: root.to_der()?,
173        });
174    }
175
176    trust_anchors.initialize()?;
177
178    let revocation_cache = RevocationCache::default();
179
180    // Make a CRL source
181    let crl_source = CrlStore::from(crls);
182    crl_source.index_crls(toi)?;
183
184    let mut pe = certval::environment::PkiEnvironment::default();
185    pe.populate_5280_pki_environment();
186    pe.add_trust_anchor_source(Box::new(trust_anchors));
187    pe.add_crl_source(Box::new(crl_source));
188    pe.add_revocation_cache(Box::new(revocation_cache));
189
190    cert_source.find_all_partial_paths(&pe, &cps);
191
192    pe.add_certificate_source(Box::new(cert_source));
193    Ok(pe)
194}

196pub(crate) fn validate_trust_anchor_cert(
197    pe: &certval::environment::PkiEnvironment,
198    cert: &x509_cert::Certificate,
199) -> RustyX509CheckResult<()> {
200    let toi = TimeOfInterest::from_unix_secs(now()?)?;
201
202    let mut cps = CertificationPathSettings::default();
203    cps.set_time_of_interest(toi);
204
205    let mut cert = PDVCertificate::try_from(cert.clone())?;
206    cert.parse_extensions(EXTS_OF_INTEREST);
207
208    let ta = PDVTrustAnchorChoice::try_from(x509_cert::anchor::TrustAnchorChoice::Certificate(
209        cert.decoded().clone(),
210    ))?;
211    let mut certification_path = CertificationPath::new(ta, vec![], cert);
212
213    check_validity(pe, &cps, &mut certification_path, &mut CertificationPathResults::new())?;
214    verify_signatures(pe, &cps, &mut certification_path, &mut CertificationPathResults::new())?;
215
216    Ok(())
217}

219pub(crate) fn validate_cert(
220    pe: &certval::environment::PkiEnvironment,
221    end_identity_cert: &x509_cert::Certificate,
222    perform_revocation_check: bool,
223) -> RustyX509CheckResult<()> {
224    let toi = TimeOfInterest::from_unix_secs(now()?)?;
225
226    let mut cps = CertificationPathSettings::default();
227    cps.set_time_of_interest(toi);
228    cps.set_require_ta_store(true);
229    cps.set_forbid_self_signed_ee(true);
230
231    let mut end_identity_cert = PDVCertificate::try_from(end_identity_cert.clone())?;
232    end_identity_cert.parse_extensions(EXTS_OF_INTEREST);
233
234    let mut paths = vec![];
235    pe.get_paths_for_target(&end_identity_cert, &mut paths, 0, toi)?;
236
237    if paths.is_empty() {
238        return Err(RustyX509CheckError::CertValError(certval::Error::PathValidation(
239            certval::PathValidationStatus::NoPathsFound,
240        )));
241    }
242
243    let mut result = Ok(());
244
245    let any_path_validates = paths.into_iter().any(|mut path| {
246        let mut cpr = CertificationPathResults::new();
247        let _ = validate_path_rfc5280(pe, &cps, &mut path, &mut cpr);
248        let r = check_cpr(cpr);
249        if r.is_err() {
250            result = r;
251            return false;
252        }
253
254        if perform_revocation_check {
255            cps.set_check_crls(true);
256            cps.set_revocation_max_age(Duration::from_hours(24));
257            let mut cpr = CertificationPathResults::new();
258            let _ = check_revocation(pe, &cps, &mut path, &mut cpr);
259            let r = check_cpr(cpr);
260            if r.is_err() {
261                result = r;
262                return false;
263            }
264        }
265
266        true
267    });
268
269    if any_path_validates { Ok(()) } else { result }
270}

272pub(crate) fn validate_crl(
273    pe: &certval::environment::PkiEnvironment,
274    crl_raw: &[u8],
275) -> RustyX509CheckResult<x509_cert::crl::CertificateList<Raw>> {
276    let crl = x509_cert::crl::CertificateList::from_der(crl_raw)?;
277
278    let mut spki_list = vec![];
279    if let Some(aki) = crl.tbs_cert_list.crl_extensions.as_ref().and_then(|extensions| {
280        extensions
281            .iter()
282            .find(|ext| ext.extn_id == x509_cert::ext::pkix::AuthorityKeyIdentifier::OID)
283    }) {
284        let akid = aki.extn_value.as_bytes();
285        if let Ok(ta) = pe.get_trust_anchor(akid) {
286            spki_list.push(certval::source::ta_source::get_subject_public_key_info_from_trust_anchor(&ta.decoded_ta));
287        } else if let Ok(intermediates) = pe.get_intermediates_by_skid(akid) {
288            spki_list.extend(
289                intermediates
290                    .into_iter()
291                    .map(|c| c.decoded().tbs_certificate().subject_public_key_info()),
292            );
293        }
294    }
295
296    if let Ok(ta) = pe.get_trust_anchor_by_name(&crl.tbs_cert_list.issuer) {
297        let spki = certval::source::ta_source::get_subject_public_key_info_from_trust_anchor(&ta.decoded_ta);
298        if !spki_list.contains(&spki) {
299            spki_list.push(spki);
300        }
301    }
302
303    spki_list.extend(
304        pe.get_cert_by_name(&crl.tbs_cert_list.issuer)
305            .into_iter()
306            .map(|c| c.decoded().tbs_certificate().subject_public_key_info()),
307    );
308
309    spki_list.dedup();
310
311    let crl_defer = DeferDecodeSigned::from_der(crl_raw)?;
312
313    let any_spki_verifies = spki_list.into_iter().any(|spki| {
314        pe.verify_signature_message(
315            pe,
316            &crl_defer.tbs_field,
317            crl.signature.raw_bytes(),
318            &crl.signature_algorithm,
319            spki,
320        )
321        .is_ok()
322    });
323
324    if any_spki_verifies {
325        Ok(crl)
326    } else {
327        Err(RustyX509CheckError::CertValError(certval::Error::PathValidation(
328            certval::PathValidationStatus::SignatureVerificationFailure,
329        )))
330    }
