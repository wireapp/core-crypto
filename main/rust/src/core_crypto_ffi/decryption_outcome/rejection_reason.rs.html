5/// A reason why this message could not be decrypted.
6#[derive(Debug, uniffi::Enum)]
7pub enum RejectionReason {
8    /// This message has already been processed, or replays a message which has.
9    DuplicateMessage,
10    /// This commit is for an epoch which this conversation has already moved past.
11    StaleCommit,
12    /// This proposal is for an epoch which this conversation has already moved past.
13    StaleProposal,
14    /// This message is from an epoch whose secrets are no longer available.
15    ///
16    /// This includes transient messages from any epoch older than the current one.
17    MessageEpochTooOld {
18        /// Epoch of the message
19        message_epoch: u64,
20        /// Epoch of the group
21        group_epoch: u64,
22    },
23    /// This message is from an epoch too far in the future to buffer.
24    UnbufferedFarFutureMessage {
25        /// Epoch of the message
26        message_epoch: u64,
27        /// Epoch of the group
28        group_epoch: u64,
29    },
30    /// This message's ciphertext failed authentication, so it could not be decrypted.
31    ///
32    /// The ciphertext may have been corrupted or tampered with, or encrypted with keys we don't have.
33    /// Authenticated encryption deliberately doesn't distinguish between these.
34    InvalidAuthentication,
35    /// This message could not be deserialized.
36    Unparseable {
37        /// What specifically could not be parsed.
38        reason: String,
39    },
40    /// This message is valid MLS, but not a public or private message, so it cannot be decrypted.
41    ///
42    /// For example, a welcome or a key package.
43    InappropriateMessageBodyType,
44    /// This message uses a protocol version we don't support.
45    UnsupportedProtocolVersion,
46    /// This message uses a wire format which the conversation's wire format policy doesn't allow.
47    IncompatibleWireFormat,
48    /// This message's signature doesn't match its sender's signature key.
49    InvalidSignature,
50    /// The sender's X509 certificate chain could not be parsed.
51    MalformedCredential {
52        /// What went wrong when attempting to parse the certificate chain.
53        error: String,
54    },
55    /// The sender's X509 certificate chain is not correctly signed.
56    UntrustedCredential {
57        /// What went wrong when validating the signatures of the certificate chain.
58        error: String,
59    },
60    /// This message belongs to a different conversation.
61    WrongGroupId,
62    /// This message's sender is not a member of the conversation in the message's epoch.
63    UnknownMember,
64    /// This targeted message is addressed to a different member of the conversation.
65    WrongRecipient,
66    /// This public message's membership tag is missing or invalid.
67    InvalidMembershipTag,
68    /// This message comes from an external sender not authorized in this conversation.
69    UnauthorizedExternalSender,
70    /// This external proposal is of a type which external senders may not send.
71    UnsupportedProposalType,
72    /// This commit's confirmation tag doesn't match the one we computed.
73    ///
74    /// Either the commit is invalid, or our view of the conversation has diverged from the sender's.
75    ConfirmationTagMismatch,
76}

78impl From<CcRejectionReason> for RejectionReason {
79    fn from(value: CcRejectionReason) -> Self {
80        match value {
81            CcRejectionReason::Unparseable(tls_codec_error) => {
82                let reason = format!(
83                    "{tls_codec_error}: {}",
84                    tls_codec_error.source().expect("tls codec error always has a source")
85                );
86                Self::Unparseable { reason }
87            }
88            CcRejectionReason::MalformedCredential(credential_error) => Self::MalformedCredential {
89                error: credential_error.to_string(),
90            },
91            CcRejectionReason::UntrustedCredential(crypto_error) => Self::UntrustedCredential {
92                error: crypto_error.to_string(),
93            },
94            CcRejectionReason::DuplicateMessage => Self::DuplicateMessage,
95            CcRejectionReason::StaleCommit => Self::StaleCommit,
96            CcRejectionReason::StaleProposal => Self::StaleProposal,
97            CcRejectionReason::MessageEpochTooOld {
98                message_epoch,
99                group_epoch,
100            } => Self::MessageEpochTooOld {
101                message_epoch,
102                group_epoch,
103            },
104            CcRejectionReason::UnbufferedFarFutureMessage {
105                message_epoch,
106                group_epoch,
107            } => Self::UnbufferedFarFutureMessage {
108                message_epoch,
109                group_epoch,
110            },
111            CcRejectionReason::InvalidAuthentication => Self::InvalidAuthentication,
112            CcRejectionReason::InappropriateMessageBodyType => Self::InappropriateMessageBodyType,
113            CcRejectionReason::UnsupportedProtocolVersion => Self::UnsupportedProtocolVersion,
114            CcRejectionReason::IncompatibleWireFormat => Self::IncompatibleWireFormat,
115            CcRejectionReason::InvalidSignature => Self::InvalidSignature,
116            CcRejectionReason::WrongGroupId => Self::WrongGroupId,
117            CcRejectionReason::UnknownMember => Self::UnknownMember,
118            CcRejectionReason::WrongRecipient => Self::WrongRecipient,
119            CcRejectionReason::InvalidMembershipTag => Self::InvalidMembershipTag,
120            CcRejectionReason::UnauthorizedExternalSender => Self::UnauthorizedExternalSender,
121            CcRejectionReason::UnsupportedProposalType => Self::UnsupportedProposalType,
122            CcRejectionReason::ConfirmationTagMismatch => Self::ConfirmationTagMismatch,
123        }
124    }
125}
