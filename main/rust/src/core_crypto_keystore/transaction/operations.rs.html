8/// A list of operations with some caches intended to ease traversal of that list.
9///
10/// These are all logically grouped together: if we want to be able to read from
11/// one of these things, we want to read all of them. If we want to be able to
12/// write one, we want to be able to write all. So we group them together in this
13/// struct so they can all live under a single lock in the transaction.
14///
15/// While this doesn't implement all of the `Vec<Operation>` interface, it does
16/// implement `Deref`, `IntoIterator`, and `.push`, so it covers the basic use
17/// cases transparently.
18///
19/// ## Invariants
20///
21/// The caches are only correct as long as every mutation of the operations list goes through
22/// [`Self::push`] or [`Self::make_nop`]. That is what the `Deref` impl is for: it hands out
23/// `&Vec<Operation>`, so callers get the entire read-only `Vec` API and no way to append, remove,
24/// or reorder behind the caches' back. A `DerefMut` impl, or any other method which touches
25/// `operations` without maintaining the tables alongside it, would silently break every reader.
26///
27/// Every cached `usize` is an index into `operations`. That list is only ever appended to, and
28/// entries within it are only ever replaced in place by [`Operation::Nop`], so an index stays valid
29/// for the life of the transaction and comparing two of them tells you which operation came first.
30/// Several readers depend on exactly that: "was this entity deleted after it was upserted" is
31/// answered by comparing two cached indices.
32#[derive(Default, derive_more::Deref, derive_more::IntoIterator)]
33pub(super) struct Operations {
34    #[deref]
35    #[into_iterator(owned, ref)]
36    operations: Vec<Operation>,
37    /// index in the operations list of the *most recent* upsert of each entity id
38    ///
39    /// Only the most recent one: earlier upserts of the same entity id are not reachable from here.
40    last_upserts: HashMap<EntityId, usize>,
41    /// index in the operations list of the *most recent* delete of each entity id
42    ///
43    /// Only the most recent one: earlier deletes of the same entity id are not reachable from here.
44    last_deletes: HashMap<EntityId, usize>,
45    /// indices in the operations list of *every* bulk delete, grouped by the table it targets
46    ///
47    /// Unlike the two tables above, this keeps all of them, because a bulk delete is defined by a
48    /// search key instead of an entity id: there is no way to know whether a later one subsumes an
49    /// earlier one without an entity instance in hand. Each list is ascending, because
50    /// [`Self::push`] only appends and [`Self::make_nop`] only removes; readers rely on that
51    /// ordering to skip past deletions which predate the operation they care about.
52    bulk_deletes: HashMap<&'static str, Vec<usize>>,
53}
54
55impl Operations {
56    /// Add an operation to the tail of this list, updating caches appropriately.
57    ///
58    /// An [`Operation::Nop`] is discarded instead of stored: it would do nothing on commit and
59    /// nothing for any reader, so the only Nops which ever appear in the list are the tombstones
60    /// [`Self::make_nop`] leaves behind.
61    pub(super) fn push(&mut self, operation: Operation) {
62        match &operation {
63            Operation::Nop => {
64                // why bother?
65                return;
66            }
67            Operation::Upsert { entity_id, .. } => {
68                self.last_upserts.insert(entity_id.clone(), self.operations.len());
69            }
70            Operation::Delete { entity_id, .. } => {
71                self.last_deletes.insert(entity_id.clone(), self.operations.len());
72            }
73            Operation::BulkDelete { table_name, .. } => self
74                .bulk_deletes
75                .entry(table_name)
76                .or_default()
77                .push(self.operations.len()),
78        }
79
80        // store the actual operation
81        self.operations.push(operation);
82    }
83
84    /// Convert the operation at the specified index to a NOP.
85    ///
86    /// Returns `None` if `idx` is out of bounds, or the displaced operation otherwise.
87    ///
88    /// Replacing in place rather than removing is what keeps every other cached index valid; see the
89    /// invariants on [`Operations`].
90    ///
91    /// If the displaced operation was the most recent of its kind for its entity id, that cache entry
92    /// has to move to the next most recent operation of the same kind, or disappear if there is none.
93    /// Finding that predecessor is the only place in this type which scans the operations list, and
94    /// even then only leftwards from `idx`.
95    pub(super) fn make_nop(&mut self, idx: usize) -> Option<Operation> {
96        let displaced = self
97            .operations
98            .get_mut(idx)
99            .map(|operation| std::mem::replace(operation, Operation::Nop));
100
101        // ensure we update the mutation tables for this entity
102        // if we just nop'd the last relevant mutation
103        if let Some(displaced) = displaced.as_ref() {
104            let scan_params = match displaced {
105                Operation::Nop => None,
106                Operation::BulkDelete { table_name, .. } => {
107                    self.bulk_deletes
108                        .get_mut(table_name)
109                        .expect("we displaced a bulk delete so its entry must exist")
110                        .retain(|delete_idx| *delete_idx != idx);
111                    None
112                }
113                Operation::Upsert { entity_id, .. } => (idx == self.last_upserts[entity_id]).then_some((
114                    &mut self.last_upserts,
115                    (|operation: &Operation| operation.is_upsert()) as fn(&Operation) -> bool,
116                    entity_id,
117                )),
118                Operation::Delete { entity_id, .. } => (idx == self.last_deletes[entity_id]).then_some((
119                    &mut self.last_deletes,
120                    (|operation: &Operation| operation.is_delete()) as _,
121                    entity_id,
122                )),
123            };
124
125            // scan_params are set on mutations where we have to scan to find the new relevant parameter, or remove
126            // that entity id from the table entirely if there are no more mutations of that kind.
127            if let Some((table, is_relevant, entity_id)) = scan_params {
128                // we have to scan here, but not from the end; only leftwards from the old position
129                let new_final_position = self.operations[..idx].iter().rposition(|operation| {
130                    is_relevant(operation)
131                        && operation
132                            .entity_id()
133                            .is_some_and(|op_entity_id| op_entity_id == entity_id)
134                });
135
136                match new_final_position {
137                    Some(position) => {
138                        let cached_position = table
139                            .get_mut(entity_id)
140                            .expect("last modification already exists because there exists a displaced operation");
141                        *cached_position = position;
142                    }
143                    None => {
144                        table.remove(entity_id);
145                    }
146                }
147            }
148        }
149
150        displaced
151    }
152
153    /// Get the index of the last upsert for the specified entity id
154    ///
155    /// `None` when this transaction has not upserted that entity at all. Note that an upsert being
156    /// the last one does not make it live: a later delete or bulk delete may have buried it.
157    pub(super) fn last_upsert_idx_for(&self, entity_id: &EntityId) -> Option<usize> {
158        self.last_upserts.get(entity_id).copied()
159    }
160
161    /// Get the index of the last delete for the specified entity id
162    ///
163    /// `None` when this transaction has not deleted that entity at all. Note that a delete being the
164    /// last one does not make the entity gone: a later upsert may have restored it, which is why
165    /// callers who care about the entity's current state compare this against
166    /// [`Self::last_upsert_idx_for`] rather than treating a hit here as final.
167    pub(super) fn last_delete_idx_for(&self, entity_id: &EntityId) -> Option<usize> {
168        self.last_deletes.get(entity_id).copied()
169    }
170
171    /// Get the index of the last mutation for the specified entity id
172    ///
173    /// A mutation here is an upsert or a delete. Bulk deletes are keyed by search key rather than by
174    /// entity id, so they are not mutations in this sense and are not considered.
175    ///
176    /// Taking the maximum works because both candidates index the same append-only list, so the
177    /// larger index is the later operation. It also does the right thing when only one of the two
178    /// kinds exists, because `None` sorts below every `Some`.
179    pub(super) fn last_mutation_idx_for(&self, entity_id: &EntityId) -> Option<usize> {
180        let upsert = self.last_upsert_idx_for(entity_id);
181        let delete = self.last_delete_idx_for(entity_id);
182        upsert.max(delete)
183    }
184
185    /// The most recent upsert of each entity of type `E`, as `(entity id, operation index)`
186    ///
187    /// This yields one item per entity id, *not* one per upsert: earlier upserts of the same entity
188    /// are not reachable through it. Nor is an item necessarily live, since a later delete or bulk
189    /// delete may have buried it, so callers wanting the transaction's current state have to check
190    /// for that themselves.
191    ///
192    /// Iteration order is unspecified; this walks a `HashMap`.
193    pub(super) fn upsert_indices_for_type<E>(&self) -> impl '_ + Iterator<Item = (&EntityId, usize)>
194    where
195        E: Entity,
196    {
197        self.last_upserts
198            .iter()
199            .filter_map(|(entity_id, idx)| entity_id.matches_type::<E>().then_some((entity_id, *idx)))
200    }
201
202    /// The index of the most recent delete of each entity of type `E`
203    ///
204    /// As with [`Self::upsert_indices_for_type`], this yields one item per entity id rather than one
205    /// per delete, and an item is not necessarily the entity's current state: a later upsert may
206    /// have restored it.
207    ///
208    /// Iteration order is unspecified; this walks a `HashMap`.
209    pub(super) fn delete_indices_for_type<E>(&self) -> impl '_ + Iterator<Item = usize>
210    where
211        E: Entity,
212    {
213        self.last_deletes
214            .iter()
215            .filter_map(|(entity_id, idx)| entity_id.matches_type::<E>().then_some(*idx))
216    }
217
218    /// Every bulk delete affecting type `E`, each paired with the operation index at which it happened.
219    ///
220    /// The indices are what let a caller apply only those deletions which came after some particular
221    /// operation, which is what an entity taken from the transaction cache needs; see
222    /// [`BulkDeleteFilters::applies_after`].
223    pub(super) fn bulk_delete_filters<E>(&self) -> BulkDeleteFilters<'_, E>
224    where
225        E: 'static + Entity + Send + Sync,
226    {
227        self.bulk_deletes
228            .get(E::TABLE_NAME)
229            .map(|indices| {
230                let filters = indices
231                    .iter()
232                    .copied()
233                    .map(|idx| {
234                        let filter = self.operations[idx]
235                            .as_bulk_delete_filter()
236                            .expect("bulk delete indices must point to bulk delete operation");
237                        Box::new(filter) as _
238                    })
239                    .collect();
240                BulkDeleteFilters::new(filters, indices)
241            })
242            .unwrap_or_default()
243    }
244
245    /// True if the operations list contains no non-Nop operations.
246    ///
247    /// This intentionally shadows Vec::is_empty
248    pub(super) fn is_empty(&self) -> bool {
249        // https://doc.rust-lang.org/stable/std/iter/trait.Iterator.html#method.all
250        // > An empty iterator returns `true`.
251        self.operations
252            .iter()
253            .all(|operation| matches!(operation, Operation::Nop))
254    }
