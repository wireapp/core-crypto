5/// Helper to perform an SQL query to get an entity by its primary key
6///
7/// This function prepares and caches a statement of the form `SELECT * FROM table_name WHERE id = ?`.
8/// You need to provide the primary key's column name and the actual primary key, and a row-mapping function.
9///
10/// The row-mapping function gets access to a `row` instance containing all fields known to the database. You can
11/// retrieve any of them by name by using `row.get("my_field")?`.
12///
13/// Your `from_row` implementation should ideally just need to map the database fields to an appropriate struct,
14/// but if it absolutely must handle errors, consider mapping them to [`rusqlite::Error::UserFunctionError`].
15pub(crate) fn get_helper<E, FromRow>(
16    conn: &Connection,
17    primary_key_column_name: &str,
18    primary_key: impl ToSql,
19    from_row: FromRow,
20) -> CryptoKeystoreResult<Option<E>>
21where
22    E: Entity,
23    FromRow: FnOnce(&Row<'_>) -> rusqlite::Result<E>,
24{
25    let mut statement = conn.prepare_cached(&format!(
26        "SELECT * FROM {table_name} WHERE {primary_key_column_name} = ?",
27        table_name = E::TABLE_NAME
28    ))?;
29    statement
30        .query_row([primary_key], from_row)
31        .optional()
32        .map_err(Into::into)
33}

35/// Helper to query an entity by its primary key.
36///
37/// This function prepares and caches a statement of the form `SELECT * FROM table_name WHERE key_part_1 = ? AND
38/// key_part_2 = ?`. The primary-key column names and corresponding parameter values must be provided in the same order.
39/// Both simple and composite primary keys are supported.
40///
41/// The row-mapping function receives a [`Row`] containing all fields returned by the database. Fields can be retrieved
42/// by name using `row.get("my_field")?`.
43///
44/// The `from_row` implementation should ideally only map database fields to the corresponding entity. If it must handle
45/// application-specific errors, consider mapping them to [`rusqlite::Error::UserFunctionError`].
46pub(crate) fn get_helper_composite_key<E, FromRow>(
47    conn: &Connection,
48    primary_key_column_names: &[&str],
49    primary_key: impl Params,
50    from_row: FromRow,
51) -> CryptoKeystoreResult<Option<E>>
52where
53    E: Entity,
54    FromRow: FnOnce(&Row<'_>) -> rusqlite::Result<E>,
55{
56    debug_assert!(
57        !primary_key_column_names.is_empty(),
58        "a primary key must contain at least one column"
59    );
60
61    let predicates = primary_key_column_names
62        .iter()
63        .map(|column| format!("{column} = ?"))
64        .collect::<Vec<_>>()
65        .join(" AND ");
66
67    let sql = format!(
68        "SELECT * FROM {table_name} WHERE {predicates}",
69        table_name = E::TABLE_NAME,
70    );
71
72    conn.prepare_cached(&sql)?
73        .query_row(primary_key, from_row)
74        .optional()
75        .map_err(Into::into)
76}

78/// Helper to perform an SQL query to count these entities in the database.
79///
80/// This function prepares and caches a statement of the form `SELECT count(*) FROM table_name`.
81pub(crate) fn count_helper<E: Entity>(conn: &Connection) -> CryptoKeystoreResult<u32> {
82    let mut statement = conn.prepare_cached(&format!(
83        "SELECT count(*) FROM {table_name}",
84        table_name = E::TABLE_NAME
85    ))?;
86    statement.query_one([], |row| row.get(0)).map_err(Into::into)
87}

89/// Helper to perform an SQL query to load all entities from the database.
90///
91/// This function prepares and caches a statement of the form `SELECT * FROM table_name`.
92///
93/// You need to provide a row-mapping function. This function gets access to a `row` instance containing all
94/// fields known to the database. You can retrieve any of them by using `row.get("my_field")?`.
95///
96/// Your `from_row` implementation should ideally just need to map the database fields to an appropriate struct,
97/// but if it absolutely must handle errors, consider mapping them to [`rusqlite::Error::UserFunctionError`].
98pub(crate) fn load_all_helper<E, FromRow>(conn: &Connection, from_row: FromRow) -> CryptoKeystoreResult<Vec<E>>
99where
100    E: Entity,
101    FromRow: FnMut(&Row<'_>) -> rusqlite::Result<E>,
102{
103    let mut statement = conn.prepare_cached(&format!("SELECT * FROM {table_name}", table_name = E::TABLE_NAME))?;
104    statement
105        .query_map([], from_row)?
106        .collect::<Result<_, _>>()
107        .map_err(Into::into)
108}

110/// Helper to perform an SQL query to delete an entity from the database.
111///
112/// This function prepares and caches a statement of the form `DELETE FROM table_name WHERE id = ?`.
113/// You need to provide the primary key's column name and the actual primary key.
114///
115/// Returns `true` if at least one entity was deleted, or `false` if the id was not found in the database.
116pub(crate) fn delete_helper<E: Entity>(
117    tx: &Transaction<'_>,
118    primary_key_column_name: &str,
119    primary_key: impl ToSql,
120) -> CryptoKeystoreResult<bool> {
121    let mut statement = tx.prepare_cached(&format!(
122        "DELETE FROM {table_name} WHERE {primary_key_column_name} = ?",
123        table_name = E::TABLE_NAME,
124    ))?;
125    let updated = statement.execute([primary_key])?;
126    Ok(updated > 0)
127}
