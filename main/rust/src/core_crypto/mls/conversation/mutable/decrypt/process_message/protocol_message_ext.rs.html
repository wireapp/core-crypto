10/// Some helper methods on `ProtocolMessage`
11pub(super) trait ProtocolMessageExt {
12    /// Extracts the confirmation tag from this message if it is a commit we created.
13    ///
14    /// If this is not a public message, or if the content type is not a commit, or if we were not the sender,
15    /// or the body data is not a commit, or the auth data somehow is missing the content tag, returns
16    /// `MlsMessageInvalidState`.
17    fn own_commit_confirmation_tag(&self, own_leaf_index: LeafNodeIndex) -> Result<&ConfirmationTag>;
18
19    /// A non-exhaustive early duplicate detection mechanism.
20    ///
21    /// In certain scenarios, we can determine whether a message is a duplicate before processing it.
22    /// This may return spurious `false`, but all `true` results are true duplicates.
23    async fn is_duplicate(&self, conversation: &ConversationMut) -> Result<bool>;
24}
25
26impl ProtocolMessageExt for ProtocolMessage {
27    fn own_commit_confirmation_tag(&self, own_leaf_index: LeafNodeIndex) -> Result<&ConfirmationTag> {
28        let ProtocolMessage::PublicMessage(message) = self else {
29            return Err(Error::MlsMessageInvalidState(
30                "only public protocol messages have a confirmation tag",
31            ));
32        };
33        if !matches!(message.content_type(), ContentType::Commit)
34            || !matches!(message.body(), FramedContentBodyIn::Commit(_))
35            || !matches!(message.sender(), Sender::Member(idx) if idx == &own_leaf_index)
36        {
37            return Err(Error::MlsMessageInvalidState("this was not a commit we created"));
38        }
39
40        message
41            .auth
42            .confirmation_tag
43            .as_ref()
44            .ok_or(Error::MlsMessageInvalidState(
45                "confirmation tag missing from message auth",
46            ))
47    }
48
49    async fn is_duplicate(&self, conversation: &ConversationMut) -> Result<bool> {
50        match self {
51            ProtocolMessage::PrivateMessage(_) => Ok(false),
52            ProtocolMessage::PublicMessage(message) => {
53                match (message.content_type(), message.sender()) {
54                    (ContentType::Commit, Sender::Member(_) | Sender::NewMemberCommit) => {
55                        // we use the confirmation tag to detect duplicate since it is issued from the GroupContext
56                        // which is supposed to be unique per epoch
57                        let Some(msg_ct) = message.confirmation_tag() else {
58                            // a commit MUST have a ConfirmationTag
59                            return Err(Error::MlsGroupInvalidState("a commit must have a ConfirmationTag"));
60                        };
61
62                        let crypto_provider = conversation.crypto_provider().await?;
63
64                        let group_ct = conversation
65                            .group()
66                            .await
67                            .compute_confirmation_tag(&crypto_provider)
68                            .map_err(OpenMlsError::wrap("computing confirmation tag"))?;
69
70                        Ok(msg_ct == &group_ct)
71                    }
72
73                    (ContentType::Proposal, Sender::Member(_) | Sender::NewMemberProposal) => {
74                        let FramedContentBodyIn::Proposal(proposal) = message.body() else {
75                            return Err(Error::MlsGroupInvalidState(
76                                "message body was not a proposal despite ContentType::Proposal",
77                            ));
78                        };
79
80                        let proposal = Proposal::from(proposal.clone()); // TODO: eventually remove this clone 😮‍💨. Tracking issue: WPB-9622
81                        let already_exists = conversation
82                            .group()
83                            .await
84                            .pending_proposals()
85                            .any(|pp| pp.proposal() == &proposal);
86                        Ok(already_exists)
87                    }
88
89                    _ => Ok(false),
90                }
91            }
92        }
93    }
94}
