1//! Sometimes, while still in epoch `n`, we receive messages for epoch `n + 1`.
2//! This is because our DS implementation doesn't guarantee that messages arrive in order.
3//! This module deals with buffering these messages until we receive the commit that advances the
4//! group epoch.

32#[derive(Clone, Copy, PartialEq, Eq)]
33pub(crate) enum MessageRestorePolicy {
34    /// Retrieve and decrypt pending messages, then clear them from the keystore.
35    DecryptAndClear,
36    /// Clear pending messages from the keystore without decrypting them.
37    ClearOnly,
38}

40impl ConversationMut {
41    pub(super) fn buffer_future_message(&self, message: impl AsRef<[u8]>) -> Result<()> {
42        let pending_msg = MlsPendingMessage {
43            conversation_id: self.id().into(),
44            message: message.as_ref().to_vec(),
45        };
46        let context_inner = self.tx_context.inner().map_err(RecursiveError::context(
47            "getting transaction inner to buffer future message",
48        ))?;
49        pending_msg
50            .save(context_inner.transaction())
51            .map_err(KeystoreError::wrap("saving pending mls message"))?;
52        Ok(())
53    }
54
55    fn clear_pending_messages(&self) -> Result<()> {
56        let tx = self
57            .tx_context
58            .inner()
59            .map_err(RecursiveError::context("getting tx context to clear pending messages"))?;
60        MlsPendingMessage::delete_all_matching(tx.transaction(), ConversationIdRef::new(self.id().as_ref()))
61            .map_err(KeystoreError::wrap("clearing pending messages"))?;
62        Ok(())
63    }
64
65    pub(super) async fn restore_and_clear_pending_messages(&mut self) -> Result<Option<Vec<BufferedDecryptedMessage>>> {
66        let pending_messages = self
67            .restore_pending_messages(MessageRestorePolicy::DecryptAndClear)
68            .await?;
69
70        let Some(pending_messages) = pending_messages else {
71            return Ok(None);
72        };
73
74        self.clear_pending_messages()?;
75
76        Ok(Some(pending_messages))
77    }

79    #[cfg_attr(target_os = "unknown", async_recursion::async_recursion(?Send))]
80    #[cfg_attr(not(target_os = "unknown"), async_recursion::async_recursion)]
81    pub(crate) async fn restore_pending_messages(
82        &mut self,
83        policy: MessageRestorePolicy,
84    ) -> Result<Option<Vec<BufferedDecryptedMessage>>> {
85        async move {
86            let conversation_id = self.id();
87            let database = self.database()?;
88            if policy == MessageRestorePolicy::ClearOnly {
89                self.clear_pending_messages()?;
90                return Ok(None);
91            }
92
93            let mut pending_messages = database
94                .search::<MlsPendingMessage, _>(ConversationIdRef::new(conversation_id.as_ref()))
95                .await
96                .map_err(KeystoreError::wrap("finding all mls pending messages"))?
97                .into_iter()
98                .map(|m| -> Result<_> {
99                    let wire_format = u16::tls_deserialize_exact(&m.message[2..4])
100                        .map_err(TlsCodecError::deserialize("u16 (wire format)"))?;
101                    if TntWireFormat::ALL.contains(&wire_format) {
102                    let message = TntMessage::tls_deserialize(&mut m.message.as_slice())
103                        .map_err(TlsCodecError::deserialize("TntMessage"))?;
104                        // We want to restore tnt messages with the same priority as application messages, i.e.,
105                        // before proposals or commits, which is why we're using the value equal to that content type
106                        // here.
107                        return Ok((ContentType::Application as u8, PendingMessage::Tnt(Box::new(message))));
108                    }
109
110                    let message = MlsMessageIn::tls_deserialize(&mut m.message.as_slice())
111                        .map_err(TlsCodecError::deserialize("mls message in"))?;
112                    let content_type = match message.body_as_ref() {
113                        MlsMessageInBody::PublicMessage(m) => m.content_type(),
114                        MlsMessageInBody::PrivateMessage(m) => m.content_type(),
115                        _ => return Err(Error::InappropriateMessageBodyType),
116                    };
117                    Ok((content_type as u8, PendingMessage::Mls(Box::new(message))))
118                })
119                .collect::<Result<Vec<_>>>()?;
120
121            // We want to restore application messages first, then Proposals & finally Commits
122            // luckily for us that's the exact same order as the [ContentType] enum
123            // Within the application messages, we'd ideally also like to sort by message generation.
124            // Unfortunately that's not possible given the current openmls APIs.
125            // We should look into this when we move to mls-rs.
126            pending_messages.sort_by_key(|(content_type, _)| *content_type);
127
128            info!(group_id = conversation_id.to_owned(); "Attempting to restore {} buffered messages", pending_messages.len());
129
130            let mut decrypted_messages = Vec::with_capacity(pending_messages.len());
131            for (_, pending_message) in pending_messages {
132                // we have to make a best-effort attempt at decrypting these messages, but ultimately the failure of a buffered
133                // message do decrypt should not short-circuit the decryption loop, and should not cause the commit to fail to decrypt
134                let maybe_decrypted = match pending_message {
135                    PendingMessage::Mls(mls_message) => self.decrypt_mls_message(*mls_message, RecursionPolicy::None).await,
136                    PendingMessage::Tnt(tnt_message) => self.decrypt_tnt_message(*tnt_message).await,
137                };
138                let decrypted = match maybe_decrypted {
139                    Ok(decrypted) => decrypted,
140                    Err(err) => {
141                        error!(error:% = err; "failed to decrypt buffered message");
142                        continue;
143                    }
144                };
145                decrypted_messages.push(decrypted.into());
146            }
147
148            let decrypted_messages = (!decrypted_messages.is_empty()).then_some(decrypted_messages);
149
150            Ok(decrypted_messages)
151        }
152        .await.inspect_err(|err| {
153            error!(error:% = err; "Error restoring pending messages");
154        })
155    }
156}

377    /// A buffered future message must not block us from processing our own removal.
378    ///
379    /// This is ordinary out-of-order delivery, the very thing the buffering machinery exists for:
380    /// the DS hands us an epoch `n + 1` application message before it hands us the epoch `n + 1`
381    /// commit which removed us. Once that commit arrives, the conversation must be wiped, and the
382    /// buffered message must go with it. If instead the buffered message is replayed against the
383    /// new epoch, decryption fails identically on every retry, in every subsequent transaction, so
384    /// the client can never process its own removal and the conversation is stuck forever.
385    #[apply(all_cred_cipher)]
386    async fn buffered_message_must_not_block_own_removal(case: TestContext) {
387        let [mut alice, bob, charlie] = case.sessions().await;
388        let conversation = case.create_conversation([&alice, &bob, &charlie]).await;
389        let conversation_id = conversation.id().clone();
390
391        // Bob removes Alice. Charlie hears about it; Alice does not, so she stays at epoch 1.
392        let commit_guard = conversation.acting_as(&bob).await.remove(&alice).await;
393        let removal_commit = commit_guard.message().to_bytes().unwrap();
394        let conversation = commit_guard
395            .notify_member(&charlie)
396            .await
397            .process_member_changes()
398            .await
399            .finish();
400
401        // Bob then speaks in the new epoch.
402        let app_msg = conversation
403            .guard_of(&bob)
404            .await
405            .encrypt_message(b"Hello from epoch 2!")
406            .await
407            .unwrap();
408
409        // Alice is still at epoch 1, so she buffers the message.
410        let decrypt = conversation.guard_of(&alice).await.decrypt_message(app_msg).await;
411        assert!(matches!(
412            decrypt.unwrap_err(),
413            Error::BufferedFutureMessage { message_epoch: 2 }
414        ));
415        assert_eq!(
416            alice.transaction.count_entities().await.pending_messages,
417            1,
418            "the message Alice could not decrypt must have been buffered"
419        );
420
421        // Commit Alice's transaction, so the buffered message is durable, as it would be
422        // between two real client calls.
423        drop(conversation);
424        alice.commit_transaction().await;
425
426        // Only now does Alice receive the commit which removed her.
427        let decrypted = alice
428            .transaction
429            .conversation(&conversation_id)
430            .await
431            .unwrap()
432            .decrypt_message(&removal_commit)
433            .await
434            .expect("Alice must be able to process her own removal")
435            .into_commit()
436            .unwrap();
437        assert!(
438            !decrypted.is_active,
439            "the commit removed Alice, so she is no longer an active member"
440        );
441
442        alice.commit_transaction().await;
443
444        let counts = alice.transaction.count_entities().await;
445        assert_eq!(
446            counts.group, 0,
447            "processing her own removal must have wiped the conversation"
448        );
449        assert_eq!(
450            counts.pending_messages, 0,
451            "the buffered message belongs to a conversation which no longer exists"
452        );
453    }

455    /// A buffered commit which removes us must still wipe the conversation.
456    ///
457    /// The eviction can arrive out of order: the DS hands us the commit which removes us before
458    /// the earlier commit we need in order to process it, so the removal is buffered and only
459    /// replayed once that earlier commit merges. The replay is what actually evicts us, so the
460    /// earlier commit's decryption result is the only place the caller can learn about it, and it
461    /// therefore has to report us as inactive. Otherwise nothing wipes the conversation: the
462    /// keystore keeps a group we've been evicted from, every subsequent operation on it fails
463    /// inside openmls with `UseAfterEviction`, and the client is never told it was removed.
464    #[apply(all_cred_cipher)]
465    async fn buffered_commit_removing_self_must_wipe_conversation(case: TestContext) {
466        let [mut alice, bob, charlie] = case.sessions().await;
467        let conversation = case.create_conversation([&alice, &bob, &charlie]).await;
468        let conversation_id = conversation.id().clone();
469
470        // Bob commits at epoch 1, advancing the group to epoch 2. Charlie hears about it;
471        // Alice does not, so she stays at epoch 1.
472        let commit_guard = conversation.acting_as(&bob).await.update().await;
473        let first_commit = commit_guard.message().to_bytes().unwrap();
474        let conversation = commit_guard.notify_member(&charlie).await.finish();
475
476        // Bob then removes Alice, at epoch 2.
477        let commit_guard = conversation.acting_as(&bob).await.remove(&alice).await;
478        let removal_commit = commit_guard.message().to_bytes().unwrap();
479        let conversation = commit_guard
480            .notify_member(&charlie)
481            .await
482            .process_member_changes()
483            .await
484            .finish();
485
486        // The DS delivers the removal to Alice first. She's still at epoch 1, so she buffers it.
487        let decrypt = conversation
488            .guard_of(&alice)
489            .await
490            .decrypt_message(&removal_commit)
491            .await;
492        assert!(matches!(
493            decrypt.unwrap_err(),
494            Error::BufferedFutureMessage { message_epoch: 2 }
495        ));
496        assert_eq!(
497            alice.transaction.count_entities().await.pending_messages,
498            1,
499            "the commit Alice could not yet apply must have been buffered"
500        );
501
502        // Commit Alice's transaction, so the buffered commit is durable, as it would be
503        // between two real client calls.
504        drop(conversation);
505        alice.commit_transaction().await;
506
507        // Only now does Alice receive the commit which unblocks the buffered removal.
508        let decrypted = alice
509            .transaction
510            .conversation(&conversation_id)
511            .await
512            .unwrap()
513            .decrypt_message(&first_commit)
514            .await
515            .expect("the earlier commit must apply")
516            .into_commit()
517            .unwrap();
518        assert!(
519            !decrypted.is_active,
520            "replaying the buffered commit removed Alice, so she is no longer an active member"
521        );
522
523        alice.commit_transaction().await;
524
525        let counts = alice.transaction.count_entities().await;
526        assert_eq!(
527            counts.group, 0,
528            "processing her own removal must have wiped the conversation"
529        );
530        assert_eq!(
531            counts.pending_messages, 0,
532            "the buffered commit belongs to a conversation which no longer exists"
533        );
534    }

536    /// Buffered application messages are restored as far as the out-of-order tolerance allows.
537    ///
538    /// The DS doesn't guarantee delivery order, so application messages for a new epoch can be
539    /// buffered in any order relative to each other. We replay them in the order they were
540    /// buffered, and we can't do better: a message's generation lives in its encrypted sender
541    /// data, so we can't sort the buffer by it without decrypting first.
542    ///
543    /// That costs us messages. Decrypting the newest generation first advances the ratchet past
544    /// it and immediately discards every past secret outside the tolerance window, so only
545    /// `OUT_OF_ORDER_TOLERANCE` of the buffered messages are still decryptable by the time we
546    /// reach them. We accept that loss, but it has to stay contained: the commit itself must
547    /// still merge and report success, and the buffer must still be cleared, so that an
548    /// undecryptable message can neither fail the commit nor poison every future restore.
549    #[apply(all_cred_cipher)]
550    async fn buffered_messages_are_restored_within_out_of_order_tolerance(case: TestContext) {
551        const MESSAGE_COUNT: u8 = 4;
552
553        let [alice, bob] = case.sessions().await;
554        let conversation = case.create_conversation([&alice, &bob]).await;
555
556        // Bob commits and merges immediately into his own state; Alice does not receive it yet.
557        let commit_guard = conversation.acting_as(&bob).await.update().await;
558        let commit = commit_guard.message();
559        let conversation = commit_guard.process_member_changes().await.finish();
560
561        // Bob sends more application messages in the new epoch than the tolerance window covers,
562        // at generations 0..MESSAGE_COUNT.
563        let mut app_messages = Vec::with_capacity(MESSAGE_COUNT.into());
564        for i in 0..MESSAGE_COUNT {
565            let msg = conversation.guard_of(&bob).await.encrypt_message([i]).await.unwrap();
566            app_messages.push(msg);
567        }
568
569        // The DS hands these to Alice newest first, oldest last. Alice is still on the old epoch,
570        // so each one gets buffered individually, in that same order.
571        for msg in app_messages.iter().rev() {
572            let decrypt = conversation.guard_of(&alice).await.decrypt_message(msg).await;
573            assert!(matches!(decrypt.unwrap_err(), Error::BufferedFutureMessage { .. }));
574        }
575        assert_eq!(
576            alice.transaction.count_entities().await.pending_messages,
577            u32::from(MESSAGE_COUNT)
578        );
579
580        // Alice now receives the commit. It must merge, and whichever buffered messages are still
581        // decryptable must come back with it.
582        let restored = conversation
583            .guard_of(&alice)
584            .await
585            .decrypt_message(commit.to_bytes().unwrap())
586            .await
587            .expect("the commit must merge even though some buffered messages can't be decrypted")
588            .into_commit()
589            .unwrap()
590            .buffered_messages
591            .expect("the buffered messages within the tolerance window should have been restored");
592
593        let restored_plaintexts = restored
594            .into_iter()
595            .map(|m| DecryptedMessage::from(m).into_application_message().unwrap().plaintext[0])
596            .collect::<Vec<_>>();
597        let expected_plaintexts = ((MESSAGE_COUNT - OUT_OF_ORDER_TOLERANCE as u8)..MESSAGE_COUNT)
598            .rev()
599            .collect::<Vec<_>>();
600        assert_eq!(
601            restored_plaintexts, expected_plaintexts,
602            "only the newest OUT_OF_ORDER_TOLERANCE messages are still decryptable in replay order"
603        );
604
605        assert_eq!(
606            conversation.guard_of(&alice).await.group().await.epoch().as_u64(),
607            conversation.guard_of(&bob).await.group().await.epoch().as_u64(),
608            "the commit must have merged into Alice's group"
609        );
610        assert_eq!(
611            alice.transaction.count_entities().await.pending_messages,
612            0,
613            "the buffer must be cleared even though some of its messages could not be decrypted"
614        );
615    }

617    /// Replicating [WPB-15810]
618    ///
619    /// [WPB-15810]: https://wearezeta.atlassian.net/browse/WPB-15810
620    #[apply(all_cred_cipher)]
621    async fn wpb_15810(mut case: TestContext) {
622        if case.is_pure_ciphertext() {
623            // The use case tested here requires inspecting your own commit.
624            // Openmls does not support this currently when protocol messages are encrypted.
625            return;
626        }
627
628        case.sessions_in_memory = true;
629        let [external_0, new_member, member_27, observer, member_114, member_115] = case.sessions().await;
630        Box::pin(async move {
631            // set up external_0 as the backend / delivery service
632            // scenario start: everyone except "new_member" is in the conversation
633            let conversation = case
634                .create_conversation_with_external_sender(
635                    &external_0,
636                    [&observer, &member_114, &member_115, &member_27],
637                )
638                .await;
639
640            // Everyone should agree on the overall state here, to wit: the group consists of everyone
641            // except "new_member", and "external_0", and no messages have been sent.
642            // At this point only the observer is going to receive messages, because that shouldn't impact group state.
643
644            // external 0 sends a proposal to remove 114
645            let proposal_guard = conversation.external_remove_proposal(&external_0, &member_114).await;
646            let proposal_remove_114_1 = proposal_guard.message();
647            let conversation = proposal_guard.finish();
648
649            // now bump the epoch in external_0: the new member has joined
650            let (commit_guard, mut pending_conversation) = conversation.external_join_unmerged(&new_member).await;
651
652            let new_member_join_commit = commit_guard.message();
653
654            pending_conversation.merge().await.unwrap();
655            let conversation = commit_guard.process_member_changes().await.finish();
656
657            // also create the same proposal with the epoch increased by 1
658            let proposal_guard = conversation
659                .acting_as(&new_member)
660                .await
661                .external_remove_proposal(&external_0, &member_114)
662                .await;
663            let proposal_remove_114_2 = proposal_guard.message();
664            let conversation = proposal_guard.finish();
665
666            // now our observer receives these messages out of order
667            println!("observer executing first proposal");
668            conversation
669                .guard()
670                .await
671                .decrypt_message(&proposal_remove_114_1.to_bytes().unwrap())
672                .await
673                .unwrap();
674            println!("observer executing second proposal");
675            let result = conversation
676                .guard()
677                .await
678                .decrypt_message(&proposal_remove_114_2.to_bytes().unwrap())
679                .await;
680            assert!(matches!(
681                result.unwrap_err(),
682                Error::BufferedFutureMessage { message_epoch: 2 }
683            ));
684            println!("executing commit adding new user");
685            conversation
686                .guard()
687                .await
688                .decrypt_message(&new_member_join_commit.to_bytes().unwrap())
689                .await
690                .unwrap();
691
692            // now the new member receives the messages in order
693            println!("new_member executing first proposal");
694            assert!(matches!(
695                conversation
696                    .guard_of(&new_member)
697                    .await
698                    .decrypt_message(&proposal_remove_114_1.to_bytes().unwrap())
699                    .await
700                    .unwrap_err(),
701                crate::mls::conversation::Error::StaleProposal,
702            ));
703            println!("new_member executing second proposal");
704            conversation
705                .guard_of(&new_member)
706                .await
707                .decrypt_message(&proposal_remove_114_2.to_bytes().unwrap())
708                .await
709                .unwrap();
710
711            // now let's switch to the perspective of member 27
712            // they have observed exactly one of the "remove 114" proposals,
713            // plus a "remove 115" proposal. We can assume that they observe the 2nd
714            // "remove 114" proposal because they advanced the epoch correctly when
715            // the new member was added.
716            let proposal_guard = conversation.external_remove_proposal(&external_0, &member_115).await;
717            let proposal_remove_115 = proposal_guard.message();
718            let conversation = proposal_guard.finish();
719
720            conversation
721                .guard_of(&member_27)
722                .await
723                .decrypt_message(&proposal_remove_114_1.to_bytes().unwrap())
724                .await
725                .unwrap();
726            let result = conversation
727                .guard_of(&member_27)
728                .await
729                .decrypt_message(&proposal_remove_114_2.to_bytes().unwrap())
730                .await;
731            assert!(matches!(
732                result.unwrap_err(),
733                Error::BufferedFutureMessage { message_epoch: 2 }
734            ));
735            conversation
736                .guard_of(&member_27)
737                .await
738                .decrypt_message(&new_member_join_commit.to_bytes().unwrap())
739                .await
740                .unwrap();
741            conversation
742                .guard_of(&member_27)
743                .await
744                .decrypt_message(&proposal_remove_115.to_bytes().unwrap())
745                .await
746                .unwrap();
747
748            // In this case, note that observer receives the proposal before the commit.
749            // This is the straightforward ordering and easy to deal with.
750            conversation
751                .guard()
752                .await
753                .decrypt_message(&proposal_remove_115.to_bytes().unwrap())
754                .await
755                .unwrap();
756
757            let commit_guard = conversation
758                .acting_as(&member_27)
759                .await
760                .commit_pending_proposals()
761                .await
762                .notify_member(&observer)
763                .await;
764
765            // In this case, new_member receives the commit before the proposal. This means that
766            // the commit has to be buffered until the proposal it references is received.
767            let (commit_guard, result) = commit_guard.notify_member_fallible(&new_member).await;
768            assert!(matches!(result.unwrap_err(), Error::BufferedCommit));
769            let conversation = commit_guard.process_member_changes().await.finish();
770            conversation
771                .guard_of(&new_member)
772                .await
773                .decrypt_message(&proposal_remove_115.to_bytes().unwrap())
774                .await
775                .unwrap();
776
777            assert!(
778                conversation
779                    .is_functional_and_contains([&observer, &new_member, &member_27])
780                    .await
781            )
782        })
783        .await
784    }
