1use std::sync::Arc;
2
3use crate::{ClientId, DeviceId, Uuid};
4
5/// This directly represents a `ClientId` of the `<userid>:<device-id>@<domain>` format.
6/// Instantiate via [ClientId::deserialize].
7///
8/// Its `Display` reproduces the bytes of the [`ClientId`] it came from, so the device id is
9/// rendered as unpadded hex. Note that this differs from [`DeviceId::to_hex_string`], which is
10/// fixed-width: a client id is not a hex-encoded binary value, and only the unpadded form matches
11/// what an E2EI certificate carries.
12#[derive(Debug, uniffi::Record, derive_more::Display)]
13#[display(
14    "{user_id}{}{device_id_hex}{}{domain}",
15    core_crypto::ClientId::DELIMITER,
16    core_crypto::ClientId::DOMAIN_SEPERATOR,
17    device_id_hex = core_crypto::ClientId::encode_device_id(device_id.to_u64()),
18)]
19// only supported for records in uniffi >= 0.31. Remove condition after globally migrating to that version.
20#[cfg_attr(any(feature = "wasm", feature = "napi"), uniffi::export(Display))]
21pub struct DeserializedClientId {
22    /// The client id this was deserialized from
23    pub client_id: Arc<ClientId>,
24    /// The user id component
25    pub user_id: Arc<Uuid>,
26    /// The device id component
27    pub device_id: Arc<DeviceId>,
28    /// The domain
29    pub domain: String,
30}
31
32impl DeserializedClientId {
33    pub(crate) fn new(client_id: ClientId) -> Self {
34        let serialized = client_id.0.deserialize();
35        Self {
36            client_id: client_id.into(),
37            user_id: Arc::new(serialized.user_id.into()),
38            device_id: Arc::new(serialized.device_id.into()),
39            domain: serialized.domain,
40        }
41    }
42}

44#[cfg(test)]
45mod tests {
46    use std::sync::Arc;
47
48    use crate::{ClientId, DeviceId, Uuid};
49
50    /// `Display` has to reproduce the client id it was deserialized from, including when the
51    /// device id's leading nibbles are zero. It previously rendered the device id fixed-width,
52    /// which does not match the client id, nor the `wireapp://` SAN of an E2EI certificate.
53    #[test]
54    fn display_round_trips_the_client_id() {
55        let user_id = Arc::new(Uuid::new("b01ec765-db9d-4502-9859-a9fa8c437c6f").unwrap());
56
57        for device_id in [
58            0x0u64,
59            0xf,
60            0x00ff,
61            0x03e7_98f8_29a2_1f2a,
62            0x8e64_2443_0d3b_28be,
63            u64::MAX,
64        ] {
65            let client_id = ClientId::new(
66                user_id.clone(),
67                Arc::new(DeviceId::new(device_id)),
68                "wire.com".to_string(),
69            );
70            let expected = String::from_utf8(client_id.copy_bytes()).unwrap();
71
72            assert_eq!(
73                client_id.deserialize().to_string(),
74                expected,
75                "rendering a deserialized client id must reproduce the client id"
76            );
77        }
78    }
