15impl Conversation {
16    /// restore the conversation from a persisted group
17    fn from_persisted(session: Weak<Session>, persisted: &PersistedMlsGroup) -> Result<Self> {
18        let group: MlsGroup =
19            core_crypto_keystore::deser(&persisted.state).map_err(KeystoreError::wrap("deserializing group state"))?;
20        let id = ConversationId::from(group.group_id().as_slice());
21        let configuration = ConversationConfiguration {
22            cipher_suite: group.ciphersuite().into(),
23            ..Default::default()
24        };
25
26        // The tnt message counter is stored in the same row as the rest of the group.
27        let group = MlsGroupState::new(group, persisted.tnt_tx_counter.into()).into();
28
29        Ok(Self {
30            id,
31            group,
32            configuration,
33            session,
34        })
35    }
36
37    /// Load a conversation from the database
38    ///
39    /// A row marked `is_pending` is a join-by-external-commit awaiting confirmation, not a usable
40    /// conversation yet, so it's treated the same as not found here — callers fall back to
41    /// [`crate::transaction_context::TransactionContext::pending_conversation`] for that case.
42    pub(crate) async fn load(session: &Arc<Session>, id: impl AsRef<[u8]>) -> Result<Option<Self>> {
43        let group = session
44            .database()
45            .get_borrowed::<PersistedMlsGroup>(ConversationIdRef::new(id.as_ref()))
46            .await
47            .map_err(KeystoreError::wrap("finding a persisted mls group"))?;
48        let Some(group) = group.filter(|group| !group.is_pending) else {
49            return Ok(None);
50        };
51        let conversation = Self::from_persisted(Arc::downgrade(session), &group)?;
52        Ok(Some(conversation))
53    }
54
55    /// Replace the group state with the one in the database, in place, so every handle sees it.
56    ///
57    /// Returns `false`, changing nothing, if the conversation is no longer in the database.
58    pub(crate) async fn reload(&self) -> Result<bool> {
59        let session = self.session()?;
60        let Some(persisted) = Self::load(&session, self.id.as_ref()).await? else {
61            return Ok(false);
62        };
63        *self.group.write().await = persisted.group.into_inner();
64        Ok(true)
65    }
66
67    /// Effectively `Database::load_all::<PersistedMlsGroup>` but with better types
68    ///
69    /// Skips rows marked `is_pending`, for the same reason [`Self::load`] does.
70    pub(crate) async fn load_all(session: &Arc<Session>) -> Result<HashMap<ConversationId, Self>> {
71        let groups = session
72            .database()
73            .load_all::<PersistedMlsGroup>()
74            .await
75            .map_err(KeystoreError::wrap("finding all persisted mls groups"))?;
76        groups
77            .into_iter()
78            .filter(|group| !group.is_pending)
79            .map(|group| {
80                // we can't just destructure the fields straight out of the group, because we derive `Zeroize`, which
81                // zeroizes on drop, which means we are forced to clone all the group's fields, because
82                // otherwise the drop impl couldn't run.
83                let conversation = Self::from_persisted(Arc::downgrade(session), &group)?;
84                Ok((conversation.id.clone(), conversation))
85            })
86            .collect()
87    }
88}
