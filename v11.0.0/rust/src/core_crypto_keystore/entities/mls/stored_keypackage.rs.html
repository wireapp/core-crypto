17/// Entity representing a persisted `KeyPackage`
18///
19/// This entity cannot be updated in the DB: the primary key is the OpenMLS hash reference of the
20/// key package itself, so two rows sharing a key necessarily share their contents. A save which
21/// disagrees is a bug, and reports
22/// [`CryptoKeystoreError::AlreadyExists`][crate::CryptoKeystoreError::AlreadyExists] instead of
23/// silently replacing the stored key package.
24#[derive(core_crypto_macros::Debug, Clone, PartialEq, Eq, ZeroizeOnDrop)]
25pub struct StoredKeyPackage {
26    pub key_package_ref: Vec<u8>,
27    #[sensitive]
28    pub key_package: Vec<u8>,
29    #[zeroize(skip)]
30    pub credential: StoredCredentialPk,
31    #[zeroize(skip)]
32    pub hpke_init_public_key_hash: Sha256Hash,
33    #[zeroize(skip)]
34    pub leaf_encryption_key_hash: Sha256Hash,
35}
36
37impl StoredKeyPackage {
38    // this is fundamentally why we need a manual impl of the entity traits:
39    // we don't want to load the hpke init private key into memory at any point,
40    // even though the table owns it. That belongs in the `StoredHpkePrivateKey` entity.
41    const QUERY_ALL: &str = formatcp!(
42        "SELECT
43            key_package_ref,
44            key_package,
45            credential_id,
46            credential_type,
47            hpke_init_public_key_hash,
48            leaf_encryption_key_hash
49        FROM {table_name}",
50        table_name = StoredKeyPackage::TABLE_NAME,
51    );
52
53    const QUERY_ONE: &str = formatcp!(
54        "{query_all} WHERE key_package_ref = ?",
55        query_all = StoredKeyPackage::QUERY_ALL
56    );
57
58    const QUERY_BY_CREDENTIAL: &str = formatcp!(
59        "{query_all} WHERE credential_id = :credential_id AND credential_type = :credential_type",
60        query_all = StoredKeyPackage::QUERY_ALL,
61    );
62
63    const DELETE_BY_CREDENTIAL: &str = formatcp!(
64        "DELETE FROM {table_name} WHERE credential_id = :credential_id AND credential_type = :credential_type",
65        table_name = StoredKeyPackage::TABLE_NAME,
66    );

68    /// Construct an instance from a pre-computed id, serialization, and a dynamic openmls entity
69    ///
70    /// Panics if `V` is not a `KeyPackage`.
71    pub(crate) fn from_openmls<V: MlsEntity>(id: &[u8], serialization: Vec<u8>, value: &V) -> Self {
72        // Ideally we'd write this whole section like
73        //
74        // ```rust
75        // let Some(key_package) = value.downcast::<KeyPackage>() else {
76        //     panic!();
77        // };
78        // ```
79        //
80        // but it turns out the `MlsEntity::downcast` impl is broken and just recurses infinitely.
81
82        // we use typeid and not the `MlsEntity::ID` const, because it's not a sealed trait and anything
83        // can claim any id there.
84        // we use typeid and not `std::any::TypeId::of` (or for that matter `std::any::Any::downcast`)
85        // because the std version requires `T: 'static`, which we can't get here without changing openmls
86        assert_eq!(
87            typeid::of::<V>(),
88            typeid::of::<KeyPackage>(),
89            "implementation error: only call this on a value known to be a key package"
90        );
91
92        // SAFETY: we just checked the type identity, and `KeyPackage` has no lifetime parameters,
93        // so we know this must be valid.
94        let key_package = unsafe { &*std::ptr::from_ref(value).cast::<KeyPackage>() };
95
96        Self {
97            key_package_ref: id.to_owned(),
98            key_package: serialization,
99            credential: StoredCredentialPk {
100                public_key_hash: Sha256Hash::hash_from(key_package.leaf_node().signature_key().as_slice()),
101                credential_type: u16::from(key_package.leaf_node().credential().credential_type()),
102            },
103            hpke_init_public_key_hash: Sha256Hash::hash_from(key_package.hpke_init_key().as_slice()),
104            leaf_encryption_key_hash: Sha256Hash::hash_from(key_package.leaf_node().encryption_key().as_slice()),
105        }
106    }
107
108    fn from_row(row: &Row<'_>) -> rusqlite::Result<Self> {
109        Ok(Self {
110            key_package_ref: row.get("key_package_ref")?,
111            key_package: row.get("key_package")?,
112            credential: StoredCredentialPk {
113                public_key_hash: row.get("credential_id")?,
114                credential_type: row.get("credential_type")?,
115            },
116            hpke_init_public_key_hash: row.get("hpke_init_public_key_hash")?,
117            leaf_encryption_key_hash: row.get("leaf_encryption_key_hash")?,
118        })
119    }
120}

136impl Entity for StoredKeyPackage {
137    const TABLE_NAME: &'static str = "mls_key_packages";
138    fn get(conn: &Connection, key: &Self::PrimaryKey) -> CryptoKeystoreResult<Option<Self>> {
139        <Self as EntityGetBorrowed>::get_borrowed(conn, key)
140    }
141    fn count(conn: &Connection) -> CryptoKeystoreResult<u32> {
142        helpers::count_helper::<Self>(conn)
143    }
144    fn load_all(conn: &Connection) -> CryptoKeystoreResult<Vec<Self>> {
145        conn.prepare_cached(Self::QUERY_ALL)?
146            .query_map([], Self::from_row)?
147            .collect::<Result<_, _>>()
148            .map_err(Into::into)
149    }
150}

152impl EntityGetBorrowed for StoredKeyPackage {
153    fn get_borrowed(conn: &Connection, key: Self::BorrowedPrimaryKey<'_>) -> CryptoKeystoreResult<Option<Self>> {
154        conn.prepare_cached(Self::QUERY_ONE)?
155            .query_row(params![key], Self::from_row)
156            .optional()
157            .map_err(Into::into)
158    }
159}

161impl EntityDatabaseMutation for StoredKeyPackage {
162    fn save<'a, Tx>(&self, tx: &'a Tx) -> CryptoKeystoreResult<()>
163    where
164        &'a Tx: Into<Transactionlike<'a>>,
165    {
166        let conn = tx.into().conn()?;
167        let mut stmt = conn.prepare_cached(
168            "INSERT INTO mls_key_packages (
169                key_package_ref,
170                key_package,
171                credential_id,
172                credential_type,
173                hpke_init_public_key_hash,
174                leaf_encryption_key_hash
175            )
176            VALUES (
177                :key_package_ref,
178                :key_package,
179                :credential_id,
180                :credential_type,
181                :hpke_init_public_key_hash,
182                :leaf_encryption_key_hash
183            )",
184        )?;
185        stmt.execute(named_params! {
186            ":key_package_ref": self.key_package_ref,
187            ":key_package": self.key_package,
188            ":credential_id": self.credential.public_key_hash,
189            ":credential_type": self.credential.credential_type,
190            ":hpke_init_public_key_hash": self.hpke_init_public_key_hash,
191            ":leaf_encryption_key_hash": self.leaf_encryption_key_hash,
192        })
193        .map_err(crate::CryptoKeystoreError::map_already_exists(
194            <Self as Entity>::TABLE_NAME,
195        ))?;
196        Ok(())
197    }
198    fn delete<'a, Tx>(tx: &'a Tx, id: &Self::PrimaryKey) -> CryptoKeystoreResult<bool>
199    where
200        &'a Tx: Into<Transactionlike<'a>>,
201    {
202        <Self as EntityDeleteBorrowed>::delete_borrowed(tx, id)
203    }
204}

218impl SearchableEntity<StoredCredentialPk> for StoredKeyPackage {
219    fn find_all_matching(conn: &Connection, search_key: &StoredCredentialPk) -> CryptoKeystoreResult<Vec<Self>> {
220        conn.prepare_cached(Self::QUERY_BY_CREDENTIAL)?
221            .query_map(
222                named_params! {
223                    ":credential_id": search_key.public_key_hash,
224                    ":credential_type": search_key.credential_type,
225                },
226                Self::from_row,
227            )?
228            .collect::<Result<_, _>>()
229            .map_err(Into::into)
230    }
231
232    fn matches(&self, search_key: &StoredCredentialPk) -> bool {
233        self.credential == *search_key
234    }
235}

237impl DeletableBySearchKey<StoredCredentialPk> for StoredKeyPackage {
238    fn delete_all_matching<'a, Tx>(tx: &'a Tx, search_key: &StoredCredentialPk) -> CryptoKeystoreResult<()>
