18/// Outcome of processing a message
19#[derive(derive_more::From)]
20pub(super) enum ProcessedMessage {
21    Content(MlsProcessedMessage),
22    OwnCommit(ConfirmationTag),
23    Deferred(DeferralReason),
24}
25
26/// Parse and process a message. This is the step which actually performs decryption.
27pub(super) async fn process_message(
28    conversation: &mut ConversationMut,
29    message: MlsMessageIn,
30) -> Result<ProcessedMessage> {
31    // Parse the incoming message
32    let protocol_message: ProtocolMessage = match message.extract() {
33        MlsMessageInBody::PublicMessage(public_message_in) => public_message_in.into(),
34        MlsMessageInBody::PrivateMessage(private_message_in) => private_message_in.into(),
35        _ => {
36            return Err(
37                OpenMlsError::wrap("parsing inbound message")(ProcessMessageError::IncompatibleWireFormat).into(),
38            );
39        }
40    };
41
42    let is_duplicate = protocol_message.is_duplicate(conversation).await?;
43    let msg_epoch = protocol_message.epoch().as_u64();
44    let content_type = protocol_message.content_type();
45    // we need to compute the confirmation tag now, but it's only relevant in a particular case of processing the
46    // message, so don't propagate the error yet.
47    let confirmation_tag_result = protocol_message
48        .own_commit_confirmation_tag(conversation.group().await.own_leaf_index())
49        .map(ToOwned::to_owned);
50
51    let backend = conversation.crypto_provider().await?;
52    conversation
53        .mutate_group_with_operation_check(
54            async move |_, group, _| {
55                let group_epoch = group.epoch().as_u64();
56
57                // Here, we inspect the results OpenMLS retured to us and map those to our own, hopefully-more-helpful
58                // response types.
59                //
60                // It's worth noting that `group.process_message` unconditionally sets the group's
61                // `InnerState::Changed`, and `mutate_group` uses only two signals to determine
62                match group.process_message(&backend, protocol_message).await {
63                    // now that openmls has handled the message, we can return the duplicate message error
64                    Ok(_) if is_duplicate => Err(Error::DuplicateMessage),
65                    // processing message went well
66                    Ok(processed) => Ok(processed.into()),
67
68                    // Attempt to unpack whether a message was too old, too new, or duplicate.
69                    // OpenMLS makes this surprisingly challenging!
70                    Err(ProcessMessageError::ValidationError(ValidationError::UnableToDecrypt(
71                        MessageDecryptionError::GenerationOutOfBound,
72                    ))) => {
73                        // This is an annoyance, but not wrong: OpenMLS knows whether the message was too old, too new,
74                        // or an actual duplicate. It emits a log line and flattens all three variants into this single
75                        // `GenerationOutOfBound` variant, so we can't tell what situation we're in at this level.
76                        // `DuplicateMessage` is more likely than the other two, so that's what we unpack here.
77                        Err(Error::DuplicateMessage)
78                    }
79                    Err(ProcessMessageError::ValidationError(ValidationError::WrongEpoch)) if is_duplicate => {
80                        Err(Error::DuplicateMessage)
81                    }
82                    Err(ProcessMessageError::ValidationError(ValidationError::WrongEpoch))
83                        if msg_epoch > group_epoch && msg_epoch <= group_epoch.saturating_add(MAX_FUTURE_EPOCHS) =>
84                    {
85                        // limit to next epoch otherwise if we were buffering a commit for epoch + 2
86                        // we would fail when trying to decrypt it in mls.
87
88                        // We need to buffer the message until the group has advanced to the right
89                        // epoch. We can't do that here--we don't have the appropriate data in scope
90                        // --but we can at least produce the proper result and return that, so our
91                        // caller can handle it. Our caller needs to know about the epoch number, so
92                        // we pass it back inside the deferral.
93                        Ok(ProcessedMessage::Deferred(DeferralReason::FutureEpoch {
94                            epoch: msg_epoch,
95                        }))
96                    }
97                    Err(ProcessMessageError::ValidationError(ValidationError::WrongEpoch))
98                        if msg_epoch < group_epoch =>
99                    {
100                        match content_type {
101                            ContentType::Application => Err(Error::MessageEpochTooOld),
102                            ContentType::Commit => Err(Error::StaleCommit),
103                            ContentType::Proposal => Err(Error::StaleProposal),
104                        }
105                    }
106                    Err(ProcessMessageError::ValidationError(ValidationError::WrongEpoch)) => {
107                        Err(Error::UnbufferedFarFutureMessage)
108                    }
109                    Err(ProcessMessageError::ValidationError(ValidationError::UnableToDecrypt(
110                        MessageDecryptionError::SecretTreeError(SecretTreeError::TooDistantInThePast),
111                    ))) => Err(Error::MessageEpochTooOld),
112
113                    // other errors we care to unpack
114                    Err(ProcessMessageError::InvalidCommit(StageCommitError::MissingProposal)) => {
115                        Ok(ProcessedMessage::Deferred(DeferralReason::MissingProposal))
116                    }
117                    Err(ProcessMessageError::InvalidCommit(StageCommitError::OwnCommit)) => {
118                        confirmation_tag_result.map(Into::into)
119                    }
120                    Err(ProcessMessageError::ValidationError(ValidationError::UnableToDecrypt(
121                        MessageDecryptionError::AeadError,
122                    ))) => Err(Error::DecryptionError),
123
124                    // everything else gets a generic wrapping
125                    Err(err) => Err(OpenMlsError::wrap("processing message")(err).into()),
126                }
127            },
128            // only outcomes producing a message with content actually mutate the group
129            |outcome| !matches!(outcome, ProcessedMessage::Content(_)),
130        )
131        .await
132}
