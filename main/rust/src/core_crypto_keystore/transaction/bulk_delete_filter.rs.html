1/// A filter function which can be used to omit entities returned from the database from a transaction query.
2///
3/// Each bulk delete excludes certain entities, but due to the types and traits in play,
4/// we can't tell whether it applies to a given entity without having an instance
5/// in hand.
6pub(crate) type BulkDeleteFilter<E> = Box<dyn Fn(&E) -> bool + Send + Sync>;
7
8/// The bulk deletions affecting one entity type, each paired with the operation index at which it happened.
9///
10/// A bulk deletion buries an entity only if it came *after* the operation which produced that
11/// entity: clearing a conversation and then buffering a new message for it must leave the new
12/// message alone. Knowing which filters match an entity is therefore not enough on its own; we also
13/// have to know where each one sat in the operations list. Hence the indices.
14///
15/// The two fields are parallel: `filters[i]` belongs to the bulk delete at operation index
16/// `filter_indices[i]`, and `filter_indices` is ascending. [`Self::applies_after`] relies on both
17/// properties. Nothing in this type enforces them, so [`Operations::bulk_delete_filters`] — which
18/// derives the filters from the indices it is handed — should remain the only place which constructs
19/// one.
20///
21/// [`Operations::bulk_delete_filters`]: crate::transaction::operations::Operations::bulk_delete_filters
22#[derive(derive_more::Constructor)]
23pub(super) struct BulkDeleteFilters<'a, E> {
24    filters: Vec<BulkDeleteFilter<E>>,
25    filter_indices: &'a [usize],
26}
27
28impl<'a, E> Default for BulkDeleteFilters<'a, E> {
29    fn default() -> Self {
30        Self {
31            filters: Default::default(),
32            filter_indices: Default::default(),
33        }
34    }
35}
36
37impl<'a, E> BulkDeleteFilters<'a, E> {
38    /// `true` when a filter was added after the operation index `idx` which indicates to omit the provided entity
39    ///
40    /// Pass the index of the operation which produced `entity`. Bulk deletions before that point
41    /// removed whatever was in the table at the time; they say nothing about an entity written
42    /// afterwards, so they must not be consulted.
43    ///
44    /// Strictly the comparison is "at or after `idx`", but no two operations share an index, so a
45    /// filter can never sit at the same index as the operation which produced the entity.
46    pub(super) fn applies_after(&self, entity: &E, idx: usize) -> bool {
47        self.filters
48            .iter()
49            .zip(self.filter_indices)
50            .skip_while(|(_filter, filter_idx)| **filter_idx < idx)
51            .any(|(filter, _idx)| filter(entity))
52    }
53
54    /// Extract the filters, discarding the filter indices
55    ///
56    /// Dropping the indices is only sound for entities which predate every operation in this
57    /// transaction, which in practice means records loaded from the database. Every bulk deletion in
58    /// the list necessarily comes after such a record, so there is nothing left for an index to
59    /// decide. An entity which came out of the transaction cache has a position of its own and must
60    /// go through [`Self::applies_after`] instead.
61    pub(super) fn into_inner(self) -> Vec<BulkDeleteFilter<E>> {
62        self.filters
63    }
64}
