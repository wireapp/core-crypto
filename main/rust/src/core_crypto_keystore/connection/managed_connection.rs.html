1//! SQLite ownership and entry on Wasm, where synchronous I/O can suspend through JSPI.
2//!
3//! The entry guard covers whole SQLite operations and must precede synchronous
4//! connection locks. It is not held for the lifetime of a keystore transaction.
5//! Native builds retain rusqlite's connection and their existing locking behavior.
6
7#[cfg(not(target_os = "unknown"))]
8pub(crate) type ManagedConnection = rusqlite::Connection;

10#[cfg(not(target_os = "unknown"))]
11pub(crate) struct SqliteGuard;

13#[cfg(not(target_os = "unknown"))]
14impl SqliteGuard {
15    pub(crate) fn lock() -> Self {
16        Self
17    }
18}

26/// Drop the connection lock before releasing the SQLite entry guard.
27pub(crate) struct Guarded<T> {
28    pub(crate) inner: T,
29    pub(crate) _sqlite: SqliteGuard,
30}
31
32impl<T: std::ops::Deref> std::ops::Deref for Guarded<T> {
33    type Target = T::Target;
34
35    fn deref(&self) -> &Self::Target {
36        &self.inner
37    }
38}

40#[cfg(target_os = "unknown")]
41mod wasm {
42    use std::ops::{Deref, DerefMut};
43
44    use super::{SqliteGuard, defer};
45
46    #[derive(Debug)]
47    // The inner is always `Some` while the value is usable; taken only by `close` or `Drop`.
48    pub(crate) struct ManagedConnection(Option<rusqlite::Connection>);
49
50    impl From<rusqlite::Connection> for ManagedConnection {
51        fn from(conn: rusqlite::Connection) -> Self {
52            Self(Some(conn))
53        }
54    }
55
56    impl Deref for ManagedConnection {
57        type Target = rusqlite::Connection;
58
59        fn deref(&self) -> &Self::Target {
60            self.0.as_ref().expect("connection is open")
61        }
62    }
63
64    impl DerefMut for ManagedConnection {
65        fn deref_mut(&mut self) -> &mut Self::Target {
66            self.0.as_mut().expect("connection is open")
67        }
68    }
69
70    impl ManagedConnection {
71        pub(crate) fn close(mut self) -> Result<(), (Self, rusqlite::Error)> {
72            let _sqlite = SqliteGuard::lock();
73            self.0
74                .take()
75                .expect("connection is open")
76                .close()
77                .map_err(|(conn, error)| (conn.into(), error))
78        }
79    }
80
81    impl Drop for ManagedConnection {
82        fn drop(&mut self) {
83            if let Some(conn) = self.0.take() {
84                defer(move || {
85                    if let Err((conn, error)) = conn.close() {
86                        log::warn!(err:err = error; "failed to close SQLite connection");
87                        drop(conn);
88                    }
89                });
90            }
91        }
92    }
93}

95#[cfg(all(test, target_os = "unknown"))]
96mod tests {
97    use std::sync::Arc;

99    use wasm_bindgen::JsValue;

101    use crate::{Database, DatabaseKey};

103    #[core_crypto_macros::jspi_wasm_bindgen_test]
104    async fn drop_outside_jspi_rolls_back_and_closes_before_reopen() {
105        let name = format!("managed-{}.db", uuid::Uuid::new_v4());
106        let key = DatabaseKey::generate();
107        let db = Database::open(&name, &key).await.unwrap();
108        db.conn()
109            .await
110            .execute_batch("CREATE TABLE cleanup_probe(value INTEGER); INSERT INTO cleanup_probe VALUES (1);")
111            .unwrap();
112        let tx = db.new_transaction().await.unwrap();
113        tx.conn()
114            .unwrap()
115            .execute("UPDATE cleanup_probe SET value=2", [])
116            .unwrap();
117
118        // Deliberately outside a promising export, as with a JavaScript finalizer:
119        // `spawn_local` tasks are polled from a plain microtask callback.
120        let mut owners = Some((db, tx));
121        let dropped = js_sys::Promise::new(&mut |resolve, _reject| {
122            let (db, tx) = owners.take().expect("promise executor runs once");
123            wasm_bindgen_futures::spawn_local(async move {
124                drop(tx);
125                drop(db);
126                resolve.call0(&JsValue::UNDEFINED).unwrap();
127            });
128        });
129        wasm_bindgen_futures::JsFuture::from(dropped).await.unwrap();
130
131        let db = Database::open(&name, &key).await.unwrap();
132        assert_eq!(
133            db.conn()
134                .await
135                .query_row("SELECT value FROM cleanup_probe", [], |row| row.get::<_, i64>(0))
136                .unwrap(),
137            1
138        );
139        assert_eq!(
140            db.conn()
141                .await
142                .pragma_query_value(None, "journal_mode", |row| row.get::<_, String>(0))
143                .unwrap(),
144            "wal"
145        );
146        Arc::into_inner(db).unwrap().wipe().await.unwrap();
147    }

149    #[core_crypto_macros::jspi_wasm_bindgen_test]
150    async fn rekey_keeps_wal_and_reopens_with_the_new_key() {
151        let name = format!("rekey-{}.db", uuid::Uuid::new_v4());
152        let old_key = DatabaseKey::generate();
153        let new_key = DatabaseKey::generate();
154        let db = Database::open(&name, &old_key).await.unwrap();
155        db.conn()
156            .await
157            .execute_batch("CREATE TABLE rekey_probe(value INTEGER); INSERT INTO rekey_probe VALUES (7)")
158            .unwrap();
159        db.update_key(&new_key).await.unwrap();
160        assert_eq!(
161            db.conn()
162                .await
163                .pragma_query_value(None, "journal_mode", |row| row.get::<_, String>(0))
164                .unwrap(),
165            "wal"
166        );
167        Arc::into_inner(db).unwrap().close().await.unwrap();
168        assert!(Database::open(&name, &old_key).await.is_err());
169        let db = Database::open(&name, &new_key).await.unwrap();
170        assert_eq!(
171            db.conn()
172                .await
173                .query_row("SELECT value FROM rekey_probe", [], |row| row.get::<_, i64>(0))
174                .unwrap(),
175            7
176        );
177        Arc::into_inner(db).unwrap().wipe().await.unwrap();
178    }

181    async fn dropped_transaction_releases_connection_for_the_next_transaction() {
182        let name = format!("rollback-{}.db", uuid::Uuid::new_v4());
183        let db = Database::open(&name, &DatabaseKey::generate()).await.unwrap();
184        db.conn()
185            .await
186            .execute_batch("CREATE TABLE rollback_probe(value INTEGER)")
187            .unwrap();
188        let tx = db.new_transaction().await.unwrap();
189        tx.conn()
190            .unwrap()
191            .execute("INSERT INTO rollback_probe VALUES (1)", [])
192            .unwrap();
193        drop(tx);
194        let tx = db.new_transaction().await.unwrap();
195        assert_eq!(
196            tx.conn()
197                .unwrap()
198                .query_row("SELECT count(*) FROM rollback_probe", [], |row| row.get::<_, i64>(0))
199                .unwrap(),
200            0
201        );
202        tx.commit().await.unwrap();
203        Arc::into_inner(db).unwrap().wipe().await.unwrap();
204    }
