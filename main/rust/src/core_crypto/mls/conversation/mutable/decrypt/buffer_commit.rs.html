12impl ConversationMut {
13    /// Cache the bytes of a buffered commit in the backend.
14    ///
15    /// By storing the raw commit bytes and doing deserialization/decryption from scratch, we preserve all
16    /// security guarantees. When we do restore, it's as though the commit had simply been received later.
17    pub(super) fn buffer_commit(&self, commit: impl AsRef<[u8]>) -> Result<()> {
18        info!(group_id = self.id().to_owned(); "buffering commit");
19
20        let buffered_commit = StoredBufferedCommit::new(self.id().to_bytes(), commit.as_ref().to_owned());
21
22        let context_inner = self.tx_context.inner().map_err(RecursiveError::context(
23            "getting context inner for transaction for buffering commit",
24        ))?;
25        buffered_commit
26            .save(context_inner.transaction())
27            .map_err(KeystoreError::wrap("buffering commit"))?;
28
29        Ok(())
30    }
31
32    /// Retrieve the bytes of a pending commit.
33    pub(super) async fn retrieve_buffered_commit(&self) -> Result<Option<Vec<u8>>> {
34        let database = self.database()?;
35        info!(group_id = self.id().to_owned(); "attempting to retrieve buffered commit");
36        database
37            .get_borrowed::<StoredBufferedCommit>(self.id().as_ref())
38            .await
39            .map(|option| option.map(|commit| commit.commit_data().to_owned()))
40            .map_err(KeystoreError::wrap("attempting to retrieve buffered commit"))
41            .map_err(Into::into)
42    }
43
44    /// Try to apply a buffered commit.
45    ///
46    /// This is largely a convenience function which handles deserializing the message, and
47    /// gives a convenient point around which we can add context to errors. However, it's also
48    /// a place where we can introduce a pin, given that we're otherwise doing a recursive
49    /// async call, which would result in an infinitely-sized future.
50    pub(super) async fn try_process_buffered_commit(
51        &mut self,
52        commit: impl AsRef<[u8]>,
53        recursion_policy: RecursionPolicy,
54    ) -> Result<DecryptedMessage> {
55        info!(group_id = self.id().to_owned(); "attempting to process buffered commit");
56
57        let message = MlsMessageIn::tls_deserialize(&mut commit.as_ref())
58            .map_err(TlsCodecError::deserialize("mls message in"))?;
59
60        Box::pin(self.decrypt_mls_message(message, recursion_policy)).await
61    }
62
63    /// Remove the buffered commit for this conversation; it has been applied.
64    pub(super) fn clear_buffered_commit(&self) -> Result<bool> {
65        info!(group_id = self.id().to_owned(); "attempting to delete buffered commit");
66        let context_inner = self.tx_context.inner().map_err(RecursiveError::context(
67            "getting context inner for transaction for clearing buffered commit",
68        ))?;
69        StoredBufferedCommit::delete_borrowed(context_inner.transaction(), self.id().as_ref())
70            .map_err(KeystoreError::wrap("attempting to clear buffered commit"))
71            .map_err(Into::into)
72    }
73}

81    /// Set up the common scenario for these tests: Alice has buffered Bob's commit, which removes
82    /// Charlie, because it refers to a proposal she never received.
83    ///
84    /// Bob's commit is merged on his side, so he is at epoch 2 while Alice and Charlie are still at epoch 1.
85    /// Returns the conversation and the serialized proposal Alice is missing.
86    async fn alice_buffers_commit_missing_proposal<'a>(
87        case: &'a TestContext,
88        [alice, bob, charlie]: [&'a SessionContext; 3],
89    ) -> (TestConversation<'a>, Vec<u8>) {
90        let conversation = case.create_conversation([alice, bob, charlie]).await;
91
92        let proposal_guard = conversation.acting_as(bob).await.remove_proposal(charlie).await;
93        let missing_proposal = proposal_guard.message().to_bytes().unwrap();
94        let conversation = proposal_guard.finish();
95
96        let commit_guard = conversation.acting_as(bob).await.commit_pending_proposals().await;
97        let (commit_guard, result) = commit_guard.notify_member_fallible(alice).await;
98        assert!(matches!(result.unwrap_err(), Error::BufferedCommit));
99        let conversation = commit_guard.finish();
100        assert_eq!(
101            alice.transaction.count_entities().await.buffered_commits,
102            1,
103            "the commit Alice could not apply must have been buffered"
104        );
105
106        (conversation, missing_proposal)
107    }

109    /// A proposal which doesn't complete the buffered commit must leave it buffered.
110    ///
111    /// Every incoming proposal retries the buffered commit. When the retry reports that proposals are
112    /// still missing, the commit must stay where it is, so that it can apply once the proposal it is
113    /// actually waiting for arrives. Discarding it instead would silently lose that epoch.
114    #[apply(all_cred_cipher)]
115    async fn unrelated_proposal_must_not_discard_buffered_commit(case: TestContext) {
116        let [alice, bob, charlie] = case.sessions().await;
117        let (conversation, missing_proposal) =
118            alice_buffers_commit_missing_proposal(&case, [&alice, &bob, &charlie]).await;
119
120        // Charlie, still at epoch 1, sends an unrelated proposal. Alice accepts it, but it doesn't
121        // complete Bob's commit.
122        let proposal_guard = conversation.acting_as(&charlie).await.remove_proposal(&bob).await;
123        let (proposal_guard, result) = proposal_guard.notify_member_fallible(&alice).await;
124        assert!(
125            matches!(result, Ok(DecryptedMessage::Proposal(_))),
126            "the unrelated proposal must decrypt as a proposal: {result:?}"
127        );
128        let conversation = proposal_guard.finish();
129        assert_eq!(
130            alice.transaction.count_entities().await.buffered_commits,
131            1,
132            "the commit is still missing a proposal, so it must still be buffered"
133        );
134
135        // The proposal Bob's commit is waiting for arrives, and the commit applies.
136        let decrypted = conversation
137            .guard_of(&alice)
138            .await
139            .decrypt_message(&missing_proposal)
140            .await
141            .unwrap();
142        assert!(
143            decrypted.as_commit().is_some(),
144            "the missing proposal must apply the buffered commit"
145        );
146    }

155            alice_buffers_commit_missing_proposal(&case, [&alice, &bob, &charlie]).await;

174    /// Applying a buffered commit must not retry that same commit while it is being applied.
175    ///
176    /// Once the buffered commit merges, the messages buffered for the new epoch are replayed. If one of
177    /// them is a proposal, it retries whatever commit is buffered. That must not be the commit which is
178    /// being applied right now: the retry would fail as a duplicate, and the failure handling would
179    /// then reload the conversation from under the operation still in progress. The replayed proposal
180    /// must also be reported to the caller, since it has been stored.
181    #[apply(all_cred_cipher)]
182    async fn applying_buffered_commit_must_not_retry_itself(case: TestContext) {
183        let [alice, bob, charlie] = case.sessions().await;
184        let (conversation, missing_proposal) =
185            alice_buffers_commit_missing_proposal(&case, [&alice, &bob, &charlie]).await;
186        let conversation_id = conversation.id().clone();
187
188        // Bob, at epoch 2, proposes something. Alice is still at epoch 1, so she buffers it.
189        let proposal_guard = conversation.acting_as(&bob).await.invite_proposal(&charlie).await;
190        let (proposal_guard, result) = proposal_guard.notify_member_fallible(&alice).await;
191        assert!(matches!(
192            result.unwrap_err(),
193            Error::BufferedFutureMessage { message_epoch: 2 }
194        ));
195        let conversation = proposal_guard.finish();
196
197        let decrypted = conversation
198            .guard_of(&alice)
199            .await
200            .decrypt_message(&missing_proposal)
201            .await
202            .unwrap();
203        let commit = decrypted
204            .as_commit()
205            .expect("the missing proposal must apply the buffered commit");
206        assert!(
207            matches!(
208                commit.buffered_messages.as_deref(),
209                Some([BufferedDecryptedMessage::Proposal(_)])
210            ),
211            "the replayed proposal must be reported as a proposal: {:?}",
212            commit.buffered_messages
213        );
214
215        // A failed retry evicts the conversation from the cache, so this is how we observe that
216        // the commit being applied was not retried.
217        let still_cached = alice
218            .session()
219            .await
220            .conversation_cache
221            .lock()
222            .await
223            .remove(&conversation_id)
224            .is_some();
225        assert!(still_cached, "the commit being applied must not have been retried");
226        assert_eq!(
227            alice.transaction.count_entities().await.buffered_commits,
228            0,
229            "the buffered commit has applied, so it must no longer be buffered"
230        );
231    }

233    /// A buffered commit which fails after merging must leave no trace, in memory or in the database.
234    ///
235    /// Merging writes the new epoch both to the in-memory group and to the database, and the steps after
236    /// the merge can still fail. When that failure is swallowed so that the triggering proposal can
237    /// succeed, both copies of the group have to be back at the old epoch; if only the database is
238    /// rolled back, the in-memory group runs ahead of what is persisted. The proposal itself must
239    /// still be stored and reported.
240    ///
241    /// We force the failure with an undecodable message in the pending-message buffer, which the
242    /// buffered commit tries to replay only after it has merged.
243    #[apply(all_cred_cipher)]
244    async fn buffered_commit_failing_after_merge_must_roll_back_memory_and_database(case: TestContext) {
245        let [alice, bob, charlie] = case.sessions().await;
246        let (conversation, missing_proposal) =
247            alice_buffers_commit_missing_proposal(&case, [&alice, &bob, &charlie]).await;
248        let conversation_id = conversation.id().clone();
249
250        // A protocol version and a public-message wire format, then a group id claiming five bytes
251        // which aren't there: decoding the message fails after the wire format check.
252        let undecodable = MlsPendingMessage {
253            conversation_id: conversation_id.as_ref().into(),
254            message: vec![0, 1, 0, 1, 5],
255        };
256        undecodable
257            .save(alice.transaction.inner().unwrap().transaction())
258            .unwrap();
259
260        let mut alice_conversation = conversation.guard_of(&alice).await;
261        let decrypted = alice_conversation.decrypt_message(&missing_proposal).await;
262        assert!(
263            matches!(decrypted, Ok(DecryptedMessage::Proposal(_))),
264            "the commit failed, so the proposal's own result must be reported: {decrypted:?}"
265        );
266        assert_eq!(
267            alice_conversation.epoch().await,
268            1,
269            "the in-memory group must have been rolled back to the epoch before the commit"
270        );
271        drop(alice_conversation);
272
273        // Load the group afresh from the database.
274        alice
275            .session()
276            .await
277            .conversation_cache
278            .lock()
279            .await
280            .remove(&conversation_id);
281        let reloaded = alice.transaction.conversation(&conversation_id).await.unwrap();
282        assert_eq!(
283            reloaded.epoch().await,
284            1,
285            "the stored group must have been rolled back to the epoch before the commit"
286        );
287        assert_eq!(
288            reloaded.group().await.pending_proposals().count(),
289            1,
290            "the proposal must still be stored"
291        );
292        drop(reloaded);
293
294        assert_eq!(
295            alice.transaction.count_entities().await.buffered_commits,
296            0,
297            "the failed commit must have been discarded"
298        );
299    }

301    /// A buffered commit which can never apply must not poison every later proposal.
302    ///
303    /// Each incoming proposal retries the buffered commit, but the buffered commit is only cleared
304    /// once it applies. If the group moves past the buffered commit's epoch by some other route
305    /// before its missing proposal arrives, the retry can only ever fail. That failure must not
306    /// replace the result of decrypting the proposal: the proposal is perfectly valid, and the
307    /// stale commit is now useless and should be discarded. Otherwise every subsequent proposal
308    /// fails with the same error for the lifetime of the conversation.
309    #[apply(all_cred_cipher)]
310    async fn stale_buffered_commit_must_not_fail_later_proposals(case: TestContext) {
311        let [alice, bob, charlie] = case.sessions().await;
312        let conversation = case.create_conversation([&alice, &bob, &charlie]).await;
313
314        // Bob proposes removing Charlie at epoch 1, but nobody else is told about the proposal.
315        let conversation = conversation
316            .acting_as(&bob)
317            .await
318            .remove_proposal(&charlie)
319            .await
320            .finish();
321
322        // Bob then commits it. The commit refers to the proposal by reference, so Alice, who never
323        // received that proposal, cannot apply the commit, and buffers it to retry once she does.
324        let commit_guard = conversation
325            .acting_as(&bob)
326            .await
327            .commit_pending_proposals_unmerged()
328            .await;
329        let (commit_guard, result) = commit_guard.notify_member_fallible(&alice).await;
330        assert!(matches!(result.unwrap_err(), Error::BufferedCommit));
331        let conversation = commit_guard.finish();
332        assert_eq!(
333            alice.transaction.count_entities().await.buffered_commits,
334            1,
335            "the commit Alice could not apply must have been buffered"
336        );
337
338        // Before the proposal arrives, Alice updates her key material, and her commit wins:
339        // everyone moves to epoch 2, and Bob's commit is discarded. Bob's buffered commit belongs
340        // to epoch 1, so it can never apply now.
341        let conversation = conversation
342            .acting_as(&alice)
343            .await
344            .update()
345            .await
346            .notify_member(&bob)
347            .await
348            .notify_member(&charlie)
349            .await
350            .finish();
351
352        // Two perfectly valid epoch 2 proposals arrive, from two different members.
353        let proposal_guard = conversation.acting_as(&bob).await.remove_proposal(&charlie).await;
354        let (proposal_guard, bob_proposal_result) = proposal_guard.notify_member_fallible(&alice).await;
355        let conversation = proposal_guard.finish();
356
357        let proposal_guard = conversation.acting_as(&charlie).await.remove_proposal(&bob).await;
358        let (_proposal_guard, charlie_proposal_result) = proposal_guard.notify_member_fallible(&alice).await;
359
360        assert!(
361            bob_proposal_result.is_ok() && charlie_proposal_result.is_ok(),
362            "valid proposals must decrypt regardless of the stale buffered commit\n\
363                 bob's proposal: {bob_proposal_result:?}\n\
364                 charlie's proposal: {charlie_proposal_result:?}"
365        );
366        assert_eq!(
367            alice.transaction.count_entities().await.buffered_commits,
368            0,
369            "the stale buffered commit can never apply, so it must have been discarded"
370        );
371    }
