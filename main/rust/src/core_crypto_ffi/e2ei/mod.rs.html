12/// The end-to-end identity verification state of a conversation.
13///
14/// Note: this does not check pending state (pending commit, pending proposals), so it does not
15/// consider members about to be added or removed.
16#[derive(Debug, Copy, Clone, uniffi::Enum)]
17#[repr(u8)]
18pub enum E2eiConversationState {
19    /// All clients have a valid E2EI certificate.
20    Verified = 1,
21    /// Some clients are either still using Basic credentials or their certificate has expired.
22    NotVerified,
23    /// All clients are still using Basic credentials.
24    ///
25    /// Note: if all clients have expired certificates, `NotVerified` is returned instead.
26    NotEnabled,
27}
28
29impl From<core_crypto::E2eiConversationState> for E2eiConversationState {
30    fn from(value: core_crypto::E2eiConversationState) -> Self {
31        match value {
32            core_crypto::E2eiConversationState::Verified => Self::Verified,
33            core_crypto::E2eiConversationState::NotVerified => Self::NotVerified,
34            core_crypto::E2eiConversationState::NotEnabled => Self::NotEnabled,
35        }
36    }
37}

39impl TryFrom<FfiCiphersuite> for JwsAlgorithm {
40    type Error = CoreCryptoError;
41
42    fn try_from(value: FfiCiphersuite) -> Result<Self, Self::Error> {
43        // This match is deliberately exhaustive, so that adding a ciphersuite requires deciding
44        // whether it supports certificate acquisition.
45        match value {
46            FfiCiphersuite::MLS_128_DHKEMX25519_AES128GCM_SHA256_Ed25519
47            | FfiCiphersuite::MLS_128_DHKEMX25519_CHACHA20POLY1305_SHA256_Ed25519
48            | FfiCiphersuite::MLS_128_MLKEM768X25519_AES128GCM_SHA256_Ed25519
49            | FfiCiphersuite::MLS_128_MLKEM768X25519_AES256GCM_SHA384_Ed25519
50            | FfiCiphersuite::MLS_128_MLKEM768_AES256GCM_SHA384_Ed25519 => Ok(Self::Ed25519),
51            FfiCiphersuite::MLS_128_DHKEMP256_AES128GCM_SHA256_P256
52            | FfiCiphersuite::MLS_128_MLKEM768P256_AES128GCM_SHA256_P256
53            | FfiCiphersuite::MLS_128_MLKEM768P256_AES256GCM_SHA384_P256
54            | FfiCiphersuite::MLS_128_MLKEM768_AES256GCM_SHA384_P256 => Ok(Self::P256),
55            FfiCiphersuite::MLS_256_DHKEMP384_AES256GCM_SHA384_P384
56            | FfiCiphersuite::MLS_192_MLKEM1024P384_AES256GCM_SHA384_P384
57            | FfiCiphersuite::MLS_192_MLKEM1024_AES256GCM_SHA384_P384 => Ok(Self::P384),
58            FfiCiphersuite::MLS_256_DHKEMP521_AES256GCM_SHA512_P521 => Ok(Self::P521),
59            // There is no JWS algorithm for Ed448.
60            FfiCiphersuite::MLS_256_DHKEMX448_AES256GCM_SHA512_Ed448
61            | FfiCiphersuite::MLS_256_DHKEMX448_CHACHA20POLY1305_SHA512_Ed448
62            // We don't yet support ML-DSA signatures with x509 credentials.
63            | FfiCiphersuite::MLS_128_MLKEM768X25519_CHACHA20POLY1305_SHA384_MLDSA44
64            | FfiCiphersuite::MLS_192_MLKEM768_AES256GCM_SHA384_MLDSA65
65            | FfiCiphersuite::MLS_256_MLKEM1024_AES256GCM_SHA384_MLDSA87 => Err(CoreCryptoError::ad_hoc(
66                "cipher_suite is not supported for certificate acquisition",
67            )),
68        }
69    }
70}

72/// Restore the ciphersuite of a deserialized acquisition.
73fn cipher_suite_from_snapshot(cipher_suite: u16, sign_alg: JwsAlgorithm) -> CoreCryptoResult<FfiCiphersuite> {
74    let cipher_suite = FfiCiphersuite::try_from(cipher_suite).map_err(CoreCryptoError::generic())?;
75    if JwsAlgorithm::try_from(cipher_suite)? != sign_alg {
76        return Err(CoreCryptoError::ad_hoc(
77            "acquisition cipher suite doesn't match its signing algorithm",
78        ));
79    }
80    Ok(cipher_suite)
81}

83/// Configuration for an X509 credential acquisition flow.
84#[derive(Debug, Clone, uniffi::Record)]
85pub struct X509CredentialAcquisitionConfiguration {
86    /// ACME directory URL.
87    pub acme_directory_url: String,
88    /// Ciphersuite of the acquired credential.
89    pub cipher_suite: FfiCiphersuite,
90    /// User-visible display name.
91    pub display_name: String,
92    /// Wire client id for the device acquiring the credential.
93    pub client_id: Arc<ClientId>,
94    /// Wire handle without the domain suffix.
95    pub handle: String,
96    /// Wire domain.
97    pub domain: String,
98    /// Optional Wire team id.
99    pub team: Option<String>,
100    /// Certificate validity period in seconds.
101    pub validity_period_secs: u64,
102}
103
104impl X509CredentialAcquisitionConfiguration {
105    fn try_into_core(self) -> CoreCryptoResult<wire_e2e_identity::acquisition::X509CredentialConfiguration> {
106        let sign_alg: JwsAlgorithm = self.cipher_suite.try_into()?;
107
108        Ok(wire_e2e_identity::acquisition::X509CredentialConfiguration {
109            acme_directory_url: self.acme_directory_url,
110            sign_alg,
111            hash_alg: HashAlgorithm::SHA256,
112            display_name: self.display_name,
113            client_id: self.client_id.as_e2ei_client_id()?,
114            handle: self.handle,
115            domain: self.domain,
116            team: self.team,
117            validity_period: std::time::Duration::from_secs(self.validity_period_secs),
118            cipher_suite: self.cipher_suite as u16,
119        })
120    }
121}

123/// X509 credential acquisition flow.
124///
125/// This allows acquiring a X509 credential for a CoreCrypto client.
126#[derive(uniffi::Object)]
127pub struct X509CredentialAcquisition {
128    state: Mutex<AcquisitionState>,
129    cipher_suite: FfiCiphersuite,
130}
131
132enum AcquisitionState {
133    Initialized(Box<wire_e2e_identity::X509CredentialAcquisition>),
134    DpopChallengeCompleted(Box<wire_e2e_identity::X509CredentialAcquisition<states::DpopChallengeCompleted>>),
135    InProgress,
136    Finalized,
137}
138
139fn signing_key_bytes(sign_alg: wire_e2e_identity::JwsAlgorithm, signing_key_pem: &str) -> CoreCryptoResult<Vec<u8>> {
140    match sign_alg {
141        JwsAlgorithm::Ed25519 => Ed25519KeyPair::from_pem(signing_key_pem).map(|key| key.to_bytes()),
142        JwsAlgorithm::P256 => ES256KeyPair::from_pem(signing_key_pem).map(|key| key.to_bytes()),
143        JwsAlgorithm::P384 => ES384KeyPair::from_pem(signing_key_pem).map(|key| key.to_bytes()),
144        JwsAlgorithm::P521 => ES512KeyPair::from_pem(signing_key_pem).map(|key| key.to_bytes()),
145    }
146    .map_err(CoreCryptoError::generic())
147}
148
149fn credential_from_acquisition_result(
150    cipher_suite: FfiCiphersuite,
151    signing_key_pem: &str,
152    certificate_chain: Vec<x509_cert::Certificate>,
153) -> CoreCryptoResult<Credential> {
154    let sign_alg = cipher_suite.try_into()?;
155    let signing_key = signing_key_bytes(sign_alg, signing_key_pem)?;
156    let certificate_chain = certificate_chain
157        .into_iter()
158        .map(|cert| {
159            cert.to_der().map_err(|err| CoreCryptoError::E2ei {
160                e2ei_error: err.to_string(),
161            })
162        })
163        .collect::<CoreCryptoResult<Vec<_>>>()?;
164
165    let certificate_bundle = core_crypto::CertificateBundle::from_raw(
166        certificate_chain,
167        signing_key,
168        core_crypto::CipherSuite::from(cipher_suite).signature_algorithm(),
169    );
170
171    core_crypto::Credential::x509(cipher_suite.into(), certificate_bundle)
172        .map(Credential)
173        .map_err(Into::into)
174}

176#[uniffi::export]
177impl X509CredentialAcquisition {
178    /// Create a new credential acquisition
179    #[uniffi::constructor]
180    pub fn new(
181        pki_environment: Arc<PkiEnvironment>,
182        config: X509CredentialAcquisitionConfiguration,
183    ) -> CoreCryptoResult<Self> {
184        let cipher_suite = config.cipher_suite;
185        let inner = wire_e2e_identity::X509CredentialAcquisition::try_new(
186            pki_environment.clone_inner(),
187            config.try_into_core()?,
188        )?;
189
190        Ok(Self {
191            state: Mutex::new(AcquisitionState::Initialized(inner.into())),
192            cipher_suite,
193        })
194    }
195
196    /// Deserialize a credential acquisition flow.
197    #[uniffi::constructor(name = "fromBytes")]
198    pub fn from_bytes(pki_environment: Arc<PkiEnvironment>, bytes: &[u8]) -> CoreCryptoResult<Self> {
199        let snapshot = wire_e2e_identity::X509CredentialAcquisition::<states::DpopChallengeCompleted>::deserialize(
200            pki_environment.clone_inner(),
201            bytes,
202        )
203        .map_err(CoreCryptoError::generic())?;
204
205        let cipher_suite = cipher_suite_from_snapshot(snapshot.cipher_suite(), snapshot.sign_alg())?;
206
207        Ok(Self {
208            state: Mutex::new(AcquisitionState::DpopChallengeCompleted(snapshot.into())),
209            cipher_suite,
210        })
211    }
212
213    /// Complete the DPoP and OIDC challenges and return the acquired X509 credential.
214    pub async fn finalize(&self) -> CoreCryptoResult<Credential> {
215        let state = {
216            let mut state = self.state.lock().await;
217            std::mem::replace(&mut *state, AcquisitionState::InProgress)
218        };
219
220        let result = match state {
221            AcquisitionState::Initialized(inner) => inner
222                .complete_dpop_challenge()
223                .await
224                .map_err(|err| CoreCryptoError::E2ei {
225                    e2ei_error: err.to_string(),
226                })?
227                .complete_oidc_challenge()
228                .await
229                .map_err(|err| CoreCryptoError::E2ei {
230                    e2ei_error: err.to_string(),
231                }),
232            AcquisitionState::DpopChallengeCompleted(inner) => {
233                inner
234                    .complete_oidc_challenge()
235                    .await
236                    .map_err(|err| CoreCryptoError::E2ei {
237                        e2ei_error: err.to_string(),
238                    })
239            }
240            AcquisitionState::InProgress => {
241                return Err(CoreCryptoError::ad_hoc(
242                    "x509 credential acquisition is already in progress",
243                ));
244            }
245            AcquisitionState::Finalized => {
246                return Err(CoreCryptoError::ad_hoc(
247                    "x509 credential acquisition has already been finalized",
248                ));
249            }
250        };
251
252        *self.state.lock().await = AcquisitionState::Finalized;
253        let (signing_key_pem, certificate_chain) = result?;
254        credential_from_acquisition_result(self.cipher_suite, signing_key_pem.as_str(), certificate_chain)
255    }
256}

258#[cfg(test)]
259mod tests {
260    use core_crypto::SignatureScheme;
261
262    use super::*;
263
264    fn all_cipher_suites() -> impl Iterator<Item = FfiCiphersuite> {
265        (0..=u16::MAX).filter_map(|cs| FfiCiphersuite::try_from(cs).ok())
266    }
267
268    #[test]
269    fn acquisition_signing_algorithm_matches_cipher_suite() {
270        for cipher_suite in all_cipher_suites() {
271            let Ok(sign_alg) = JwsAlgorithm::try_from(cipher_suite) else {
272                continue;
273            };
274            let expected = match core_crypto::CipherSuite::from(cipher_suite).signature_algorithm() {
275                SignatureScheme::ED25519 => JwsAlgorithm::Ed25519,
276                SignatureScheme::ECDSA_SECP256R1_SHA256 => JwsAlgorithm::P256,
277                SignatureScheme::ECDSA_SECP384R1_SHA384 => JwsAlgorithm::P384,
278                SignatureScheme::ECDSA_SECP521R1_SHA512 => JwsAlgorithm::P521,
279                scheme => panic!("{cipher_suite:?} with {scheme:?} must not support acquisition"),
280            };
281            assert_eq!(sign_alg, expected, "{cipher_suite:?}");
282        }
283    }
284
285    #[test]
286    fn cipher_suite_is_restored_from_snapshot() {
287        for cipher_suite in all_cipher_suites() {
288            let Ok(sign_alg) = JwsAlgorithm::try_from(cipher_suite) else {
289                continue;
290            };
291            let restored = cipher_suite_from_snapshot(cipher_suite as u16, sign_alg).unwrap();
292            assert_eq!(restored, cipher_suite);
293        }
294    }
