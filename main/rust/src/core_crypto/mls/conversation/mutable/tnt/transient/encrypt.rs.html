16impl ConversationMut {
17    /// Transient messages are messages distributed to clients with an active WebSocket connection. They don't mutate
18    /// regular group state (i.e., as defined by RFC 9420). That is because offline clients will never receive those
19    /// messages, and will not have the appropriate cryptographic state to decrypt the messages.
20    ///
21    /// Any feature using a targeted, transient or transient targeted message MUST specify why the lower security
22    /// guarantees (compared to MLS application messages) are acceptable and/or how they are mitigated.
23    pub async fn encrypt_transient(&mut self, message: Vec<u8>) -> Result<Vec<u8>> {
24        let crypto_provider = self
25            .tx_context
26            .crypto_provider()
27            .await
28            .map_err(RecursiveError::transaction("obtaining crypto provider"))?;
29        let database = self.database().await?;
30
31        let transient_message = self
32            .mutate_group(async |_, group_state, _| {
33                Self::create_transient_message(&database, group_state, &crypto_provider, &message).await
34            })
35            .await?;
36
37        let tbs = TntMessageTBS::new_transient(transient_message);
38        self.sign_tnt_message(tbs).await
39    }
40
41    async fn create_transient_message(
42        database: &Database,
43        group_state: &mut MlsGroupState,
44        crypto_provider: &CryptoProvider,
45        message: &[u8],
46    ) -> Result<TransientMessage> {
47        let counter = group_state.obtain_transient_message_tx_counter(database).await?;
48        let mls_group = group_state.mls_group();
49        let aad = TransientMessageAad::new(mls_group.own_leaf_index(), counter, mls_group.export_group_context());
50
51        let secrets = transient_message_secrets(crypto_provider, &aad, mls_group)?;
52        let message = tls_serialize_padded(message).map_err(TlsCodecError::serialize("transient message content"))?;
53        let cipher_suite = mls_group.ciphersuite();
54        let aad = aad
55            .tls_serialize_detached()
56            .map_err(TlsCodecError::serialize("TransientMessageAad"))?;
57        let payload = crypto_provider
58            .aead_encrypt(
59                cipher_suite.aead_algorithm(),
60                &secrets.secret_key,
61                &message,
62                secrets.aead_nonce.as_slice(),
63                &aad,
64            )
65            .map_err(OpenMlsError::wrap("encrypting transient message"))?;
66
67        let sender = mls_group.own_leaf_index();
68        let group_context = mls_group.export_group_context().clone();
69
70        let transient_message = TransientMessage::new(sender, counter, group_context, payload);
71        Ok(transient_message)
72    }
73}
