9/// Unique entity implementation/migration helper.
10///
11/// Few unique entities actually care about what key gets used; it's arbitrary, imposed
12/// only by the requirement that an [`Entity`] has a [`PrimaryKey`]. Really they only need
13/// to know what table name to use, and how to serialize/deserialize. For reasons of
14/// historical compatibility we express the serialization requirements in terms of inline
15/// methods here instead of a trait implementation.
16///
17/// If you implement this trait, you get the following traits auto-implemented:
18///
19/// - `PrimaryKey`
20/// - `UniqueEntity`
21/// - `Entity`
22/// - `EntityDatabaseMutation`
23pub trait UniqueEntityImplementationHelper {
24    /// Table name for this entity.
25    const TABLE_NAME: &str;
26    fn new(content: Vec<u8>) -> Self;
27    fn content(&self) -> &[u8];
28}
29
30impl<T> UniqueEntity for T
31where
32    T: Entity + UniqueEntityImplementationHelper + PrimaryKey<PrimaryKey = u32>,
33{
34    const KEY: u32 = 0;
35}
36
37impl<T> PrimaryKey for T
38where
39    T: UniqueEntityImplementationHelper,
40{
41    type PrimaryKey = u32;
42
43    fn primary_key(&self) -> Self::PrimaryKey {
44        Self::KEY
45    }
46}

48/// A unique entity can appear either 0 or 1 times in the database.
49pub trait UniqueEntity: PrimaryKey {
50    /// The id used as the key when storing this entity in a KV store.
51    const KEY: Self::PrimaryKey;
52}
53
54/// Unique entities get some convenience methods implemented automatically.
55pub trait UniqueEntityExt: UniqueEntity + EntityDatabaseMutation {
56    /// Get this unique entity from the database.
57    fn get_unique(conn: &Connection) -> CryptoKeystoreResult<Option<Self>>;
58
59    /// Set this unique entity into the database, replacing it if it already exists.
60    ///
61    /// Returns `true` if the entity previously existed and was replaced, or
62    /// `false` if it was not removed and this was a pure insertion.
63    fn set_and_replace(&self, tx: &Transaction) -> CryptoKeystoreResult<bool>;
64
65    /// Set this unique entity into the database if it does not already exist.
66    ///
67    /// Returns `true` if the entity was saved, or `false` if it aborted due to an already-existing entity.
68    fn set_if_absent(&self, tx: &Transaction) -> CryptoKeystoreResult<bool>;
69
70    /// Returns whether or not the database contains an instance of this unique entity.
71    fn exists(conn: &Connection) -> CryptoKeystoreResult<bool>;
72}
73
74impl<E> UniqueEntityExt for E
75where
76    E: UniqueEntity + EntityDatabaseMutation + Sync,
77{
78    /// Get this unique entity from the database.
79    fn get_unique(conn: &Connection) -> CryptoKeystoreResult<Option<Self>> {
80        Self::get(conn, &Self::KEY)
81    }
82
83    /// Set this unique entity into the database, replacing it if it already exists.
84    ///
85    /// Returns `true` if the entity previously existed and was replaced, or
86    /// `false` if it was not removed and this was a pure insertion.
87    fn set_and_replace(&self, tx: &Transaction) -> CryptoKeystoreResult<bool> {
88        let deleted = Self::delete(tx, &Self::KEY)?;
89        self.save(tx)?;
90        Ok(deleted)
91    }
92
93    /// Set this unique entity into the database if it does not already exist.
94    ///
95    /// Returns `true` if the entity was saved, or `false` if it aborted due to an already-existing entity.
96    fn set_if_absent(&self, tx: &Transaction) -> CryptoKeystoreResult<bool> {
97        if Self::exists(tx)? {
98            return Ok(false);
99        }
100        self.save(tx)?;
101        Ok(true)
102    }
103
104    /// Returns whether or not the database contains an instance of this unique entity.
105    fn exists(conn: &Connection) -> CryptoKeystoreResult<bool> {
106        Self::count(conn).map(|count| count > 0)
107    }
108}

110impl<T> Entity for T
111where
112    T: PrimaryKey + UniqueEntityImplementationHelper,
113    <T as PrimaryKey>::PrimaryKey: ToSql,
114{
115    const TABLE_NAME: &'static str = <Self as UniqueEntityImplementationHelper>::TABLE_NAME;
116
117    fn get(conn: &Connection, key: &Self::PrimaryKey) -> CryptoKeystoreResult<Option<Self>> {
118        let mut statement = conn.prepare_cached(&format!(
119            "SELECT content FROM {table_name} WHERE id = ?",
120            table_name = <Self as Entity>::TABLE_NAME
121        ))?;
122        statement
123            .query_row([key], |row| Ok(Self::new(row.get("content")?)))
124            .optional()
125            .map_err(Into::into)
126    }
127
128    /// Count the number of entities of this type in the database.
129    fn count(conn: &Connection) -> CryptoKeystoreResult<u32> {
130        count_helper::<Self>(conn)
131    }
132
133    /// Retrieve all entities of this type from the database.
134    fn load_all(conn: &Connection) -> CryptoKeystoreResult<Vec<Self>> {
135        load_all_helper::<Self, _>(conn, |row| Ok(Self::new(row.get("content")?)))
136    }
137}

139impl<T> EntityDatabaseMutation for T
140where
141    T: Entity + UniqueEntityImplementationHelper + UniqueEntity,
142    <T as PrimaryKey>::PrimaryKey: ToSql,
143{
144    type AutoGeneratedFields = ();
145
146    fn save(&self, tx: &Transaction) -> CryptoKeystoreResult<()> {
147        let mut stmt = tx.prepare_cached(&format!(
148            "INSERT OR REPLACE INTO {table_name} (id, content) VALUES (?, ?)",
149            table_name = <Self as Entity>::TABLE_NAME,
150        ))?;
151        stmt.execute(params![Self::KEY, self.content()])?;
152        Ok(())
153    }
154
155    fn delete(tx: &Transaction, id: &Self::PrimaryKey) -> CryptoKeystoreResult<bool> {
156        delete_helper::<Self>(tx, "id", id)
157    }
