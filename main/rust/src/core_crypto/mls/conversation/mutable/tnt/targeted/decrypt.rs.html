34    pub(in crate::mls::conversation::mutable::tnt) async fn decrypt_targeted(
35        &self,
36        message: TargetedMessage,
37        policy: TargetedMessagePolicy,
38        sender: &Member,
39    ) -> Result<DecryptedMessage> {
40        let mls_group = self.group().await;
41        if message.recipient != mls_group.own_leaf_index() {
42            return Err(Error::MlsMessageInvalidState(
43                "TargetedMessage wasn't targeted to this client",
44            ));
45        }
46
47        let message_epoch = message.epoch.as_u64();
48        let group_epoch = mls_group.epoch().as_u64();
49        let max_future_epoch = group_epoch.saturating_add(MAX_FUTURE_EPOCHS);
50        if message_epoch > group_epoch && message_epoch <= max_future_epoch {
51            return Err(Error::BufferedFutureMessage { message_epoch });
52        } else if message_epoch > max_future_epoch {
53            return Err(Error::UnbufferedFarFutureMessage);
54        }
55
56        let database = self.database().await?;
57        let counter_pk = TargetedMessageRxCounterPkRef::new(self.id.as_ref(), message.sender().u32(), group_epoch);
58        let existing_counter = database
59            .get_borrowed::<TargetedMessageRxCounter>(counter_pk)
60            .await
61            .map_err(KeystoreError::wrap("getting TargetedMessageRxCounter"))?
62            .map(|counter| counter.count)
63            .unwrap_or_default();
64        if u32::from(message.counter) <= existing_counter {
65            return Err(Error::DuplicateMessage);
66        }
67
68        let crypto_provider = self.crypto_provider().await?;
69        let cipher_suite = &self.cipher_suite();
70        let (context_data, decryption_key) = self
71            .load_hpke_decryption_data(&mls_group, &crypto_provider, cipher_suite, &message, policy)
72            .await?;
73        let aad = message
74            .counter
75            .tls_serialize_detached()
76            .map_err(TlsCodecError::serialize("TntMessageCounter"))?;
77        let padded_plaintext = crypto_provider
78            .hpke_open_psk(
79                cipher_suite.hpke_config(),
80                &message.payload,
81                &decryption_key,
82                &context_data.info,
83                &aad,
84                &context_data.psk,
85                &context_data.psk_id,
86            )
87            .map_err(OpenMlsError::wrap("decrypting targeted message"))?;
88
89        // Remove padding from plaintext: VLBytes contains a length prefix which tells the deserializer the plaintext
90        // length.
91        let plaintext = VLBytes::tls_deserialize(&mut padded_plaintext.as_slice())
92            .map_err(TlsCodecError::deserialize("TargetedMessageContent"))?
93            .into();
94
95        let sender_credential_with_key = CredentialWithKey {
96            credential: sender.credential.clone(),
97            signature_key: sender.signature_key.clone().into(),
98        };
99        let pki_env = crypto_provider.authentication_service().pki_env().await;
100        let identity = sender_credential_with_key
101            .extract_identity(*cipher_suite, pki_env.as_deref())
102            .await
103            .map_err(RecursiveError::mls_credential("extracting identity"))?;
104        let sender_client_id = sender
105            .credential
106            .identity()
107            .try_into()
108            .map_err(RecursiveError::mls_client("client id from credential"))?;
109
110        let tx = self
111            .tx_context
112            .inner()
113            .await
114            .map_err(RecursiveError::transaction("getting inner context"))?;
115        let tx = tx.transaction();
116        tx.save(TargetedMessageRxCounter {
117            conversation_id: self.id.to_bytes(),
118            sender: message.sender().u32(),
119            epoch: group_epoch,
120            count: message.counter.into(),
121        })
122        .await
123        .map_err(KeystoreError::wrap("persisting TargetedMessageRxCounter"))?;
124
125        let decrypted_bytes = DecryptedBytes {
126            plaintext,
127            sender_client_id,
128            identity,
129        };
130
131        match policy {
132            TargetedMessagePolicy::Transient => Ok(DecryptedMessage::TransientTargeted(decrypted_bytes)),
133            TargetedMessagePolicy::Persisted => Ok(DecryptedMessage::PersistedTargeted(decrypted_bytes)),
134        }
135    }

137    /// Get the HPKE context and private key from the live group state or load it from the database if the message comes
138    /// from a past epoch.
139    async fn load_hpke_decryption_data(
140        &self,
141        mls_group: &MlsGroupState,
142        crypto_provider: &CryptoProvider,
143        cipher_suite: &CipherSuite,
144        message: &TargetedMessage,
145        policy: TargetedMessagePolicy,
146    ) -> Result<(HpkeContextData, HpkePrivateKey), Error> {
147        if message.epoch == mls_group.epoch() {
148            let context = TargetedMessageContext::new(
149                policy,
150                mls_group.export_group_context(),
151                message.sender(),
152                message.recipient,
153            );
154            let context_data = extract_hpke_context_data(crypto_provider, cipher_suite, &context, mls_group)?;
155            let decryption_key = self.load_decryption_key(mls_group).await?;
156            return Ok((context_data, decryption_key));
157        }
158
159        let epoch_distance = mls_group.epoch().as_u64() - message.epoch.as_u64();
160        if epoch_distance > MAX_PAST_EPOCHS as u64 {
161            return Err(Error::MessageEpochTooOld);
162        }
163
164        let database = self.database().await?;
165        let key = TntSecretPkRef::new(mls_group.group_id().as_slice(), message.epoch.as_u64());
166        let secret = database
167            .get_borrowed::<TntSecret>(key)
168            .await
169            .map_err(KeystoreError::wrap("loading tnt secret"))?
170            .ok_or(Error::MlsGroupInvalidState("tnt secret is missing"))?;
171        let group_context = GroupContext::tls_deserialize(&mut secret.group_context.as_slice())
172            .map_err(TlsCodecError::deserialize("TntSecret GroupContext"))?;
173        let context = TargetedMessageContext::new(policy, &group_context, message.sender(), message.recipient);
174        let info = context
175            .tls_serialize_detached()
176            .map_err(TlsCodecError::serialize("TargetedMessageContext"))?;
177        let psk_id = PskId::new(mls_group.group_id().clone(), message.epoch)
178            .tls_serialize_detached()
179            .map_err(TlsCodecError::serialize("PskId"))?;
180        let decryption_key = core_crypto_keystore::deser(&secret.hpke_private_key)
181            .map_err(KeystoreError::wrap("deserializing tnt hpke private key"))?;
182        let context_data = HpkeContextData {
183            info,
184            psk_id,
185            psk: secret.targeted_message_psk.clone(),
186        };
187        Ok((context_data, decryption_key))
188    }

190    pub(in crate::mls::conversation) async fn create_tnt_secret(&self, mls_group: &MlsGroup) -> Result<TntSecret> {
191        let crypto_provider = self.crypto_provider().await?;
192        let hpke_private_key = self.load_decryption_key(mls_group).await?;
193        let targeted_message_psk = derive_targeted_message_psk(&crypto_provider, &self.cipher_suite(), mls_group)?;
194
195        Ok(TntSecret {
196            conversation_id: mls_group.group_id().to_vec(),
197            epoch: mls_group.epoch().as_u64(),
198            hpke_private_key: core_crypto_keystore::ser(&hpke_private_key)
199                .map_err(KeystoreError::wrap("serializing tnt hpke private key"))?,
200            group_context: mls_group
201                .export_group_context()
202                .tls_serialize_detached()
203                .map_err(TlsCodecError::serialize("TntSecret GroupContext"))?,
204            targeted_message_psk,
205        })
206    }

208    // Ideally, openmls would offer this API for us, but it doesn't. So we load the data we need from the database.
209    async fn load_decryption_key(&self, mls_group: &MlsGroup) -> Result<HpkePrivateKey> {
210        let kp_ref = StoredEpochEncryptionKeypairPkRef::new(
211            mls_group.group_id().as_slice(),
212            mls_group.own_leaf_index().u32(),
213            mls_group.epoch().as_u64(),
214        );
215
216        let database = self.database().await?;
217
218        let stored_keypairs = database
219            .get_borrowed::<StoredEpochEncryptionKeypair>(kp_ref)
220            .await
221            .map_err(KeystoreError::wrap("loading epoch encryption keypairs"))?
222            .ok_or(Error::MlsGroupInvalidState("epoch encryption keypairs are missing"))?;
223
224        let keypairs = core_crypto_keystore::deser::<PersistedEpochEncryptionKeypairs>(&stored_keypairs.keypairs)
225            .map_err(KeystoreError::wrap("deserializing epoch encryption key"))?;
226        let own_encryption_key = mls_group
227            .own_leaf_node()
228            .ok_or(Error::MlsGroupInvalidState("own leaf node is missing"))?
229            .encryption_key();
230
231        keypairs
232            .0
233            .into_iter()
234            .find(|keypair| keypair.public_key.as_slice() == own_encryption_key.as_slice())
235            .map(|keypair| keypair.private_key.key)
236            .ok_or(Error::MlsGroupInvalidState("own leaf encryption keypair is missing"))
237    }
