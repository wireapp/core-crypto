1//! These methods are specialized for performing certain entity-specific queries.
2
3use std::{borrow::Cow, sync::Arc};
4
5use super::dynamic_dispatch::EntityId;
6use crate::{
7    CryptoKeystoreResult,
8    entities::{MlsPendingMessage, PersistedMlsGroup},
9    traits::{BorrowPrimaryKey, KeyType as _},
10    transaction::Transaction,
11};

13impl Transaction {
14    pub(crate) async fn child_groups(
15        &self,
16        entity: PersistedMlsGroup,
17        persisted_records: impl IntoIterator<Item = PersistedMlsGroup>,
18    ) -> CryptoKeystoreResult<Vec<PersistedMlsGroup>> {
19        // First get all raw groups from the cache, then filter by their parent id
20        let cached_records = self.find_all_in_cache::<PersistedMlsGroup>().await;
21        let cached_records = cached_records
22            .iter()
23            .filter(|maybe_child| {
24                maybe_child
25                    .parent_id
26                    .as_deref()
27                    .map(|parent_id| parent_id == entity.borrow_primary_key().bytes().as_ref())
28                    .unwrap_or_default()
29            })
30            .map(Arc::as_ref)
31            .map(Cow::Borrowed);
32
33        let persisted_records = persisted_records.into_iter().map(Cow::Owned);
34
35        Ok(self.merge_records(cached_records, persisted_records).await)
36    }
37
38    pub(crate) async fn remove_pending_messages_by_conversation_id(&self, conversation_id: impl AsRef<[u8]> + Send) {
39        let conversation_id = conversation_id.as_ref();
40
41        {
42            let mut cache_guard = self.cache.write().await;
43            cache_guard.retain(|_entity_id, entity| {
44                let Some(pending_message) = entity.downcast::<MlsPendingMessage>() else {
45                    return true;
46                };
47                pending_message.foreign_id != conversation_id
48            });
49        }
50
51        let mut deleted_set = self.deleted.write().await;
52        deleted_set.insert(
53            EntityId::from_key::<MlsPendingMessage>(conversation_id.into())
54                .expect("mls pending messages are proper entities which can be parsed"),
55        );
56    }
57
58    pub(crate) async fn find_pending_messages_by_conversation_id(
59        &self,
60        conversation_id: &[u8],
61        persisted_records: impl IntoIterator<Item = MlsPendingMessage>,
62    ) -> CryptoKeystoreResult<Vec<MlsPendingMessage>> {
63        let persisted_records = persisted_records.into_iter().map(Cow::Owned);
64
65        let cached_records = self.find_all_in_cache::<MlsPendingMessage>().await;
66        let cached_records = cached_records
67            .iter()
68            .filter(|pending_message| pending_message.foreign_id == conversation_id)
69            .map(Arc::as_ref)
70            .map(Cow::Borrowed);
71
72        let merged_records = self.merge_records(cached_records, persisted_records).await;
73        Ok(merged_records)
74    }
