68/// Try to extract the relevant data from the v5 credential and signature keypair to determine whether they correspond
69/// to each other. If succeeding, and they do match, return Ok(Some) with the new credential, return Ok(None) if they
70/// don't, and return an Error if the data extraction fails.
71pub(crate) fn migrate_to_new_credential(
72    v5_credential: &V5Credential,
73    stored_keypair: &StoredSignatureKeypair,
74) -> CryptoKeystoreResult<Option<V6Credential>> {
75    let mls_keypair = SignatureKeyPair::tls_deserialize_exact(&stored_keypair.keypair)
76        .map_err(|e| CryptoKeystoreError::MigrationFailed(format!("Deserializing keypair: {e}")))?;
77
78    if !v5_credential_matches_signature_key(v5_credential, stored_keypair, &mls_keypair)? {
79        return Ok(None);
80    }
81
82    let new_credential = V6Credential {
83        session_id: v5_credential.id.clone(),
84        credential: v5_credential.credential.clone(),
85        created_at: v5_credential.created_at,
86        signature_scheme: stored_keypair.signature_scheme,
87        private_key: mls_keypair.private().to_owned(),
88        public_key: stored_keypair.pk.clone(),
89    };
90
91    Ok(Some(new_credential))
92}

203pub(crate) fn v5_credential_matches_signature_key(
204    v5_credential: &V5Credential,
205    stored_keypair: &StoredSignatureKeypair,
206    signature_key: &SignatureKeyPair,
207) -> CryptoKeystoreResult<bool> {
208    let mls_credential = MlsCredential::tls_deserialize_exact(&v5_credential.credential)
209        .map_err(|e| CryptoKeystoreError::MigrationFailed(format!("deserializing stored credential: {e}")))?;
210
211    match mls_credential.mls_credential() {
212        MlsCredentialType::Basic(_) => {
213            let keypair_identity = &stored_keypair.credential_id;
214            if keypair_identity != mls_credential.identity() {
215                return Ok(false);
216            }
217        }
218
219        MlsCredentialType::X509(cert) => {
220            let certificate_bytes = cert
221                .certificates
222                .first()
223                .ok_or(CryptoKeystoreError::MigrationFailed("No leaf certificate".into()))?;
224            let certificate_inner = x509_cert::Certificate::from_der(certificate_bytes.as_slice())
225                .map_err(|e| CryptoKeystoreError::MigrationFailed(format!("decoding x509 certificate: {e}")))?;
226            let credential_public_key = certificate_inner
227                .public_key()
228                .map_err(|e| CryptoKeystoreError::MigrationFailed(format!("extracting public key from cert: {e}")))?;
229
230            if signature_key.public() != credential_public_key {
231                return Ok(false);
232            }
233        }
234    }
235
236    Ok(true)
237}

239#[derive(Default)]
240struct CiphersuiteOccurences {
241    /// MLS_128_DHKEMX25519_AES128GCM_SHA256_Ed25519
242    ed25519_aes: u32,
243    /// MLS_128_DHKEMX25519_CHACHA20POLY1305_SHA256_Ed25519
244    ed25519_chacha: u32,
245    /// MLS_256_DHKEMX448_AES256GCM_SHA512_Ed448
246    ed448_aes: u32,
247    /// MLS_256_DHKEMX448_CHACHA20POLY1305_SHA512_Ed448
248    ed448_chacha: u32,
249}
250
251impl CiphersuiteOccurences {
252    fn of(&self, ciphersuite: u16) -> Option<u32> {
253        match ciphersuite.try_into().ok()? {
254            Ciphersuite::MLS_128_DHKEMX25519_AES128GCM_SHA256_Ed25519 => self.ed25519_aes.into(),
255            Ciphersuite::MLS_128_DHKEMP256_AES128GCM_SHA256_P256 => None,
256            Ciphersuite::MLS_128_DHKEMX25519_CHACHA20POLY1305_SHA256_Ed25519 => self.ed25519_chacha.into(),
257            Ciphersuite::MLS_256_DHKEMX448_AES256GCM_SHA512_Ed448 => self.ed448_aes.into(),
258            Ciphersuite::MLS_256_DHKEMP521_AES256GCM_SHA512_P521 => None,
259            Ciphersuite::MLS_256_DHKEMX448_CHACHA20POLY1305_SHA512_Ed448 => self.ed448_chacha.into(),
260            Ciphersuite::MLS_256_DHKEMP384_AES256GCM_SHA384_P384 => None,
261        }
262    }
263}

265/// Count occurences of ciphersuites ambiguous with regard to the signature scheme.
266fn count_ciphersuite_occurences(
267    persisted_mls_groups: impl IntoIterator<Item = LegacyPersistedMlsGroup>,
268) -> CryptoKeystoreResult<CiphersuiteOccurences> {
269    let mut occurences = CiphersuiteOccurences::default();
270
271    for mls_group in persisted_mls_groups {
272        let mls_group = deser::<MlsGroup>(&mls_group.state)?;
273        match mls_group.ciphersuite() {
274            Ciphersuite::MLS_128_DHKEMX25519_AES128GCM_SHA256_Ed25519 => occurences.ed25519_aes += 1,
275            Ciphersuite::MLS_128_DHKEMX25519_CHACHA20POLY1305_SHA256_Ed25519 => occurences.ed25519_chacha += 1,
276            Ciphersuite::MLS_256_DHKEMX448_AES256GCM_SHA512_Ed448 => occurences.ed448_aes += 1,
277            Ciphersuite::MLS_256_DHKEMX448_CHACHA20POLY1305_SHA512_Ed448 => occurences.ed448_chacha += 1,
278            _ => {}
279        }
280    }
281
282    Ok(occurences)
283}

285/// Make a function which inverts [`Credential::signature_algorithm`](https://github.com/wireapp/openmls/blob/c9cde17076508968c9cbead5728454f0a1f60c4f/traits/src/types.rs#L472-L474)
286///
287/// We need to map a signature scheme to a ciphersuite in our migrations. Most of these mappings are unambiguous.
288/// However, we still need to consider the possibility that users are using unexpected signature schemes.
289///
290/// The strategy here is to perform an analysis of the existing conversations, counting the ciphersuites actually in
291/// use. For each ambiguous signature scheme, there are these possibilities to consider:
292///
293/// | Ciphersuite A | Ciphersuite B | Outcome |
294/// |---------------|---------------|---------|
295/// |             0 |             0 | Ciphersuite unused, drop this credential |
296/// |      non-zero |             0 | As all conversations use ciphersuite A, it is safe to uniquely map the credential to this ciphersuite |
297/// |             0 |      non-zero | As all conversations use ciphersuite B, it is safe to uniquely map the credential to this ciphersuite |
298/// |    non-zero n |    non-zero n | We arbitrarily choose the ciphersuite with smaller canonical number, breaking half the user conversations |
299/// |    non-zero a |    non-zero b | We choose the ciphersuite with the greater population, breaking those conversations with the lesser |
300///
301/// However, we expect the cases where both A and B have non-zero counts to be very rare.
302pub(crate) fn make_ciphersuite_for_signature_scheme(
303    persisted_mls_groups: impl IntoIterator<Item = LegacyPersistedMlsGroup>,
304) -> CryptoKeystoreResult<impl Fn(u16) -> Option<u16>> {
305    let occurences = count_ciphersuite_occurences(persisted_mls_groups)?;
306
307    let ed25519 = match (occurences.ed25519_aes, occurences.ed25519_chacha) {
308        (0, 0) => None,
309        // other 0 cases are degenerate and handled below
310        (a, b) if a >= b => Some(Ciphersuite::MLS_128_DHKEMX25519_AES128GCM_SHA256_Ed25519.into()),
311        _ => Some(Ciphersuite::MLS_128_DHKEMX25519_CHACHA20POLY1305_SHA256_Ed25519.into()),
312    };
313
314    let ed448 = match (occurences.ed448_aes, occurences.ed448_chacha) {
315        (0, 0) => None,
316        // other 0 cases are degenerate and handled below
317        (a, b) if a >= b => Some(Ciphersuite::MLS_256_DHKEMX448_AES256GCM_SHA512_Ed448.into()),
318        _ => Some(Ciphersuite::MLS_256_DHKEMX448_CHACHA20POLY1305_SHA512_Ed448.into()),
319    };
320
321    let ciphersuite_for_signature_scheme = move |signature_scheme: u16| -> Option<u16> {
322        let signature_scheme = SignatureScheme::try_from(signature_scheme).ok()?;
323        match signature_scheme {
324            SignatureScheme::ECDSA_SECP256R1_SHA256 => {
325                Some(Ciphersuite::MLS_128_DHKEMP256_AES128GCM_SHA256_P256.into())
326            }
327            SignatureScheme::ECDSA_SECP384R1_SHA384 => {
328                Some(Ciphersuite::MLS_256_DHKEMP384_AES256GCM_SHA384_P384.into())
329            }
330            SignatureScheme::ECDSA_SECP521R1_SHA512 => {
331                Some(Ciphersuite::MLS_256_DHKEMP521_AES256GCM_SHA512_P521.into())
332            }
333            SignatureScheme::ED25519 => ed25519,
334            SignatureScheme::ED448 => ed448,
335        }
336    };
337    Ok(ciphersuite_for_signature_scheme)
338}

340/// Make a function that determines the least used ciphersuite dependeing on usages in the given mls groups.
341///
342/// Behavioral notes of the resulting function:
343/// * Ciphersuites that are not considered will cause `None` to be returned.
344/// * Only ciphersuites ambiguous w.r.t. their signature scheme will be considered (see [CiphersuiteOccurences]).
345/// * If both ciphersuites have an occurence of 0, `None` is returned.
346/// * If both ciphersuites have equal occurence, the numerically higher ciphersuite is returned.
347pub(crate) fn make_least_used_ciphersuite(
348    persisted_mls_groups: impl IntoIterator<Item = LegacyPersistedMlsGroup>,
349) -> CryptoKeystoreResult<impl Fn(u16, u16) -> Option<u16>> {
350    let occurences = count_ciphersuite_occurences(persisted_mls_groups)?;
351
352    let least_used_ciphersuite = move |ciphersuite_a: u16, ciphersuite_b: u16| -> Option<u16> {
353        let occurence_a = occurences.of(ciphersuite_a);
354        let occurence_b = occurences.of(ciphersuite_b);
355
356        match (occurence_a, occurence_b) {
357            // If one of the occurences is None, it means that the ciphersuites aren't both instances of an ambiguous
358            // pair of ciphersuites. If both have an occurence of 0, we cannot determine a least used ciphersuite,
359            // either.
360            (None, _) | (_, None) | (Some(0), Some(0)) => return None,
361            (Some(a), Some(b)) if a < b => {
362                return Some(ciphersuite_a);
363            }
364            // If both credentials have equal occurence, let the below if-clause handle this.
365            (Some(a), Some(b)) if a == b => {}
366            _ => {
367                return Some(ciphersuite_b);
368            }
369        }
370
371        // This is reached when both credentials have equal occurence. Take the one with the numerically
372        // higher ciphersuite in this case.
373        if ciphersuite_a > ciphersuite_b {
374            Some(ciphersuite_a)
375        } else {
376            Some(ciphersuite_b)
377        }
378    };
379
380    Ok(least_used_ciphersuite)
381}

383pub(crate) fn detect_duplicate_credentials(
384    creds: &[StoredCredentialV36],
385) -> Vec<(&StoredCredentialV36, &StoredCredentialV36)> {
386    let mut duplicates = Vec::new();
387
388    for (i, a) in creds.iter().enumerate() {
389        for b in creds.iter().skip(i + 1) {
390            if a.public_key == b.public_key {
391                duplicates.push((a, b));
392            }
393        }
394    }
395    duplicates
396}
