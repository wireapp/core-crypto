1use std::ops::Deref;
2
3use async_lock::RwLockReadGuardArc;
4
5use crate::transaction_context::{Error, Result, TransactionContext, TransactionContextInner};
6
7struct InnerGuard {
8    guard: RwLockReadGuardArc<Option<TransactionContextInner>>,
9}
10
11impl Deref for InnerGuard {
12    type Target = TransactionContextInner;
13
14    fn deref(&self) -> &Self::Target {
15        self.guard
16            .deref()
17            .as_ref()
18            .expect("we ensure at construction that the guard is valid")
19        // ... and it can't go invalid because we still hold that guard
20    }
21}
22
23impl TransactionContext {
24    /// Get a guard which derefs to [`TransactionContextInner`], or produce an appropriate error.
25    pub(crate) fn inner(&self) -> Result<impl Deref<Target = TransactionContextInner>> {
26        // Access to inner is designed to be immutable, concurrent, and fast.
27        // There is exactly one place where we take a write guard: `take_inner`, below, where we invalidate the context.
28        // If we can't immediately acquire a read guard, we'll never be able to (unless the future containing
29        // the `take_inner` is cancelled, which is a niche enough edge case I'm not going to worry about it).
30        //
31        // NOTE: there is no structural guarantee that there is exactly one place which acquires a write lock.
32        // We depend on convention to enforce the precondition underlying this logic.
33        let guard = self.inner.try_read_arc().ok_or(Error::InvalidTransactionContext)?;
34        match *guard {
35            Some(_) => Ok(InnerGuard { guard }),
36            None => Err(Error::InvalidTransactionContext),
37        }
38    }
39
40    /// Take the [`TransactionContextInner`] from self, leaving it in an invalid state.
41    pub(super) async fn take_inner(&self) -> Result<TransactionContextInner> {
42        let mut guard = self.inner.write().await;
43        guard.take().ok_or(Error::InvalidTransactionContext)
44    }
45}

47#[cfg(test)]
48mod tests {
49    use core::time::Duration;
50
51    use smol::future::{or, poll_once};
52
53    use crate::{CoreCrypto, Database};
54
55    /// Nesting [`TransactionContext::inner`] must not deadlock against
56    /// [`TransactionContext::take_inner`].
57    ///
58    /// `async_lock::RwLock` is write-preferring: the write future sets its writer bit while the
59    /// existing readers are still live, and new readers park as soon as that bit is set. So a task
60    /// which holds an [`InnerGuard`] across an await point and then re-enters `inner()` cannot get
61    /// the second guard once another task has asked for the write lock in between — and that writer
62    /// is itself waiting for the first guard to drop.
63    ///
64    /// Real code has this shape: `remove_credential` holds a guard across
65    /// `remove_key_packages_for`, which holds one across `remove_key_package`, which takes a third;
66    /// `check_credentials` does the same across `clean_up_irrelevant_crls`. The writer is any
67    /// concurrent `finish` or `abort` on a retained context.
68    ///
69    /// Whichever side wins, both must terminate: the nesting caller either gets its second guard or
70    /// gets [`Error::InvalidTransactionContext`], and the writer either takes the inner or reports
71    /// that it was already taken.
72    #[macro_rules_attribute::apply(smol_macros::test)]
73    async fn nested_inner_calls_with_take_inner_do_not_deadlock() {
74        let core_crypto = CoreCrypto::new(Database::open_in_memory().unwrap());
75        let context = core_crypto.new_transaction().await.unwrap();
76
77        // The nesting caller: take a guard, await something, then take the guard again.
78        let mut nested = Box::pin({
79            let context = context.clone();
80            async move {
81                let _outer_guard = context.inner().expect("outer guard is valid");
82                smol::future::yield_now().await;
83                context.inner().map(|_| ())
84            }
85        });
86
87        // The writer: `finish`/`abort` from a retained context or a callback-spawned task.
88        let mut taker = Box::pin({
89            let context = context.clone();
90            async move { context.take_inner().await.map(|_| ()) }
91        });
92
93        // Force the interleaving: the nesting caller parks holding the outer guard, then the writer
94        // parks behind it, setting the writer bit.
95        assert!(
96            poll_once(nested.as_mut()).await.is_none(),
97            "the nesting caller should suspend while holding the outer guard"
98        );
99        assert!(
100            poll_once(taker.as_mut()).await.is_none(),
101            "the writer should park behind the outer guard"
102        );
103
104        // Now let both of them run. Each must reach some conclusion, successful or not.
105        let timed_out = or(
106            async {
107                let _ = futures_util::future::join(nested, taker).await;
108                false
109            },
110            async {
111                smol::Timer::after(Duration::from_secs(1)).await;
112                true
113            },
114        )
115        .await;
116
117        assert!(
118            !timed_out,
119            "nesting `inner()` deadlocked against a concurrent `take_inner()`"
120        );
121    }
122}
